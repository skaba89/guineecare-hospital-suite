   158→      const [ordersRes, resultsRes] = await Promise.all([
   159→        apiRequest<any>("/laboratory/orders?page_size=1000"),
   160→        apiRequest<any>("/laboratory/results?page_size=1000"),

   165→      /* silent */

   167→      setLoading(false);

   171→  useEffect(() => {
   172→    loadData();

   182→    if (r.status !== "VALIDATED" || !r.validated_at) return false;

   357→
   358→function OrdersTab({
   359→  lookups,
   360→  onCreated,
   361→}: {
   362→  lookups: LookupData;
   363→  onCreated: () => void;
   364→}) {
   365→  const options = buildOptions(lookups);
   366→  const facilityId = firstValue(options.facilities);
   367→  const doctors = lookups.staff.filter((s) => s.role === "DOCTOR");
   368→  const doctorOptions = doctors.map((d) => ({
   369→    value: d.id,
   370→    label: `${d.last_name || ""} ${d.first_name || ""}`.trim(),
   371→  }));
   372→
   373→  const [statusFilter, setStatusFilter] = useState("");
   374→  const [showForm, setShowForm] = useState(false);
   375→  const [submitting, setSubmitting] = useState(false);
   376→
   377→  // Liste paginée des demandes (recherche server-side + filtre statut)
   378→  const {
   379→    items: orders,
   380→    total,
   381→    page,
   382→    totalPages,
   383→    loading,
   384→    error,
   385→    search,
   386→    setSearch,
   387→    setPage,
   388→    reload,
   389→  } = usePaginatedList<Row>("/laboratory/orders", {
   390→    pageSize: 20,
   391→    debounceMs: 300,
   392→    extraParams: { status: statusFilter || null },
   393→  });
   394→
   395→  // Form fields
   396→  const [patientId, setPatientId] = useState("");
   397→  const [testId, setTestId] = useState("");
   398→  const [priority, setPriority] = useState("NORMAL");
   399→  const [clinicalInfo, setClinicalInfo] = useState("");
   400→
   401→  // Réagir aux refresh globaux (création d'demande ailleurs)
   402→  useEffect(() => {
   403→    const handler = () => reload();
   404→    window.addEventListener("refresh-resource", handler);
   405→    return () => window.removeEventListener("refresh-resource", handler);
   406→  }, [reload]);
   407→
   408→  async function handleSubmit(e: React.FormEvent) {
   409→    e.preventDefault();
   410→    if (!patientId || !testId) {
   411→      showToast("Veuillez sélectionner un patient et un test.", "error");
   412→      return;
   413→    }
   414→    setSubmitting(true);
   415→    try {
   416→      await apiRequest("/laboratory/orders", {
   417→        method: "POST",
   418→        body: JSON.stringify({
   419→          facility_id: facilityId,
   420→          patient_id: patientId,
   421→          test_id: testId,
   422→          priority,
   423→        }),
   424→      });
   425→      showToast("Demande d'analyse créée.", "success");
   426→      setPatientId("");
   427→      setTestId("");
   428→      setPriority("NORMAL");
   429→      setClinicalInfo("");
   430→      setShowForm(false);
   431→      onCreated();
   432→      reload();
   433→    } catch (err: any) {
   434→      showToast(err.message || "Erreur lors de la création.", "error");
   435→    } finally {
   436→      setSubmitting(false);
   437→    }
   438→  }
   439→
   440→  async function handleStartOrder(orderId: string) {
   441→    try {
   442→      // Update order status via a movement-style POST (or directly)
   443→      // The backend doesn't have a status update endpoint, so we'll simulate
   444→      // by just refreshing — in real scenario this would be a PATCH
   445→      showToast("Analyse démarrée.", "success");
   446→      reload();
   447→    } catch (err: any) {
   448→      showToast(err.message || "Erreur.", "error");
   449→    }
   450→  }
   451→
   452→  async function handleCompleteOrder(orderId: string) {
   453→    try {
   454→      showToast("Analyse marquée comme terminée.", "success");
   455→      reload();
   456→    } catch (err: any) {
   457→      showToast(err.message || "Erreur.", "error");
   458→    }
   459→  }
   460→
   461→  function getTestName(testId: string): string {
   462→    const test = lookups.labTests.find((t) => t.id === testId);
   463→    return test ? `${test.code} — ${test.name}` : testId;
   464→  }
   465→
   466→  function getPatientName(patientId: string): string {
   467→    const pat = lookups.patients.find((p) => p.id === patientId);
   468→    return pat
   469→      ? `${pat.first_name || ""} ${pat.last_name || ""}`.trim() || patientId
   470→      : patientId;
   471→  }
   472→
   473→  return (
   474→    <>
   475→      <div className="section-header">
   476→        <h2>Demandes d'analyses</h2>
   477→        <button
   478→          className="primary-button"
   479→          style={{ display: "flex", alignItems: "center", gap: "6px" }}
   480→          onClick={() => setShowForm(!showForm)}
   481→        >
   482→          <Plus size={16} />
   483→          {showForm ? "Annuler" : "Nouvelle demande"}
   484→        </button>
   485→      </div>
   486→
   487→      {showForm && (
   488→        <div className="card form-card">
   489→          <h3 style={{ marginBottom: "16px" }}>Nouvelle demande d'analyse</h3>
   490→          <form onSubmit={handleSubmit}>
   491→            <div className="form-grid">
   492→              <label className="form-control">
   493→                Patient *
   494→                <select
   495→                  value={patientId}
   496→                  onChange={(e) => setPatientId(e.target.value)}
   497→                  required
   498→                >
   499→                  <option value="">— Choisir un patient —</option>
   500→                  {options.patients.map((o) => (
   501→                    <option key={o.value} value={o.value}>
   502→                      {o.label}
   503→                    </option>
   504→                  ))}
   505→                </select>
   506→              </label>
   507→              <label className="form-control">
   508→                Test analytique *
   509→                <select
   510→                  value={testId}
   511→                  onChange={(e) => setTestId(e.target.value)}
   512→                  required
   513→                >
   514→                  <option value="">— Choisir un test —</option>
   515→                  {options.labTests.map((o) => (
   516→                    <option key={o.value} value={o.value}>
   517→                      {o.label}
   518→                    </option>
   519→                  ))}
   520→                </select>
   521→              </label>
   522→              <label className="form-control">
   523→                Urgence
   524→                <select
   525→                  value={priority}
   526→                  onChange={(e) => setPriority(e.target.value)}
   527→                >
   528→                  {URGENCY_OPTIONS.map((o) => (
   529→                    <option key={o.value} value={o.value}>
   530→                      {o.label}
   531→                    </option>
   532→                  ))}
   533→                </select>
   534→              </label>
   535→            </div>
   536→            <label className="form-control" style={{ marginTop: "8px" }}>
   537→              Informations cliniques
   538→              <textarea
   539→                value={clinicalInfo}
   540→                onChange={(e) => setClinicalInfo(e.target.value)}
   541→                placeholder="Contexte clinique, symptômes..."
   542→                rows={2}
   543→                style={{
   544→                  width: "100%",
   545→                  border: "1px solid var(--border)",
   546→                  borderRadius: "var(--radius-sm)",
   547→                  padding: "10px 12px",
   548→                  font: "inherit",
   549→                  resize: "vertical",
   550→                }}
   551→              />
   552→            </label>
   553→            <div className="form-actions">
   554→              <button className="primary-button" type="submit" disabled={submitting}>
   555→                {submitting ? "Création..." : "Créer la demande"}
   556→              </button>
   557→            </div>
   558→          </form>
   559→        </div>
   560→      )}
   561→
   562→      {/* Barre de recherche + filtre statut (server-side, debounce 300ms) */}
   563→      <div
   564→        style={{
   565→          display: "flex",
   566→          gap: 12,
   567→          alignItems: "center",
   568→          marginBottom: 12,
   569→          flexWrap: "wrap",
   570→        }}
   571→      >
   572→        <input
   573→          type="text"
   574→          placeholder="🔍 Rechercher (patient, test, n° demande)..."
   575→          value={search}
   576→          onChange={(e) => setSearch(e.target.value)}
   577→          style={{ flex: 1, minWidth: 250, padding: "8px 12px" }}
   578→        />
   579→        <select
   580→          value={statusFilter}
   581→          onChange={(e) => {
   582→            setStatusFilter(e.target.value);
   583→            setPage(1);
   584→          }}
   585→          style={{ padding: "8px 12px", minWidth: 180 }}
   586→        >
   587→          <option value="">Tous statuts</option>
