    98→  const { hasPermission } = useAuth();

   101→

   103→

   132→function StaffTab({ lookups, canManage, onCreated }: { lookups: LookupData; canManage: boolean; onCreated?: () => void }) {
   133→  const [staff, setStaff] = useState<Row[]>([]);
   134→  const [loading, setLoading] = useState(true);
   135→  const [error, setError] = useState("");
   136→  const [showForm, setShowForm] = useState(false);
   137→  const [submitting, setSubmitting] = useState(false);
   138→  const [search, setSearch] = useState("");
   139→  const [roleFilter, setRoleFilter] = useState("");
   140→  const [statusFilter, setStatusFilter] = useState("");
   141→
   142→  // Form fields
   143→  const [firstName, setFirstName] = useState("");
   144→  const [lastName, setLastName] = useState("");
   145→  const [role, setRole] = useState("MEDECIN");
   146→  const [specialty, setSpecialty] = useState("");
   147→  const [phone, setPhone] = useState("");
   148→  const [email, setEmail] = useState("");
   149→  const [contractType, setContractType] = useState("CDI");
   150→  const [salaryGrade, setSalaryGrade] = useState("");
   151→  const [hireDate, setHireDate] = useState("");
   152→
   153→  const loadStaff = useCallback(async () => {
   154→    setLoading(true);
   155→    setError("");
   156→    try {
   157→      const params = new URLSearchParams();
   158→      if (search) params.set("search", search);
   159→      if (roleFilter) params.set("profession", roleFilter);
   160→      if (statusFilter) params.set("status", statusFilter);
   161→      const qs = params.toString();
   162→      const payload = await apiRequest<any>(`/personnel/staff${qs ? `?${qs}` : ""}`);
   163→      setStaff(Array.isArray(payload.data) ? payload.data : []);
   164→    } catch {
   165→      setError("Impossible de charger le personnel.");
   166→    } finally {
   167→      setLoading(false);
   168→    }
   169→  }, [search, roleFilter, statusFilter]);
   170→
   171→  useEffect(() => {
   172→    loadStaff();
   173→    const handler = () => loadStaff();
   174→    window.addEventListener("refresh-resource", handler);
   175→    return () => window.removeEventListener("refresh-resource", handler);
   176→  }, [loadStaff]);
   177→
   178→  async function handleSubmit(e: React.FormEvent) {
   179→    e.preventDefault();
   180→    if (!firstName.trim() || !lastName.trim()) return;
   181→    setSubmitting(true);
   182→    try {
   183→      await apiRequest("/personnel/staff", {
   184→        method: "POST",
   185→        body: JSON.stringify({
   186→          first_name: firstName.trim(),
   187→          last_name: lastName.trim(),
   188→          role,
   189→          specialty: specialty.trim() || undefined,
   190→          phone: phone.trim() || undefined,
   191→          email: email.trim() || undefined,
   192→          contract_type: contractType || undefined,
   193→          salary_grade: salaryGrade.trim() || undefined,
   194→          hire_date: hireDate || undefined,
   195→        }),
   196→      });
   197→      setFirstName(""); setLastName(""); setRole("MEDECIN"); setSpecialty("");
   198→      setPhone(""); setEmail(""); setContractType("CDI"); setSalaryGrade(""); setHireDate("");
   199→      setShowForm(false);
   200→      loadStaff();
   201→      onCreated?.();
   202→      showToast("Membre du personnel ajouté.", "success");
   203→    } catch {
   204→      showToast("Erreur lors de l'ajout.", "error");
   205→    } finally {
   206→      setSubmitting(false);
   207→    }
   208→  }
   209→
   210→  return (
   211→    <>
   212→      <div className="section-header">
   213→        <div style={{ display: "flex", gap: "12px", alignItems: "center", flexWrap: "wrap" }}>
   214→          <h2 style={{ margin: 0 }}>Personnel</h2>
   215→          <label className="toolbar-control" style={{ marginBottom: 0 }}>
   216→            <input type="text" placeholder="Rechercher..." value={search} onChange={(e) => setSearch(e.target.value)} style={{ minWidth: "180px" }} />
   217→          </label>
   218→          <label className="toolbar-control" style={{ marginBottom: 0 }}>
   219→            <select value={roleFilter} onChange={(e) => setRoleFilter(e.target.value)} style={{ minWidth: "160px" }}>
   220→              <option value="">Tous les rôles</option>
   221→              {STAFF_ROLE_OPTIONS.map((o) => (<option key={o.value} value={o.value}>{o.label}</option>))}
   222→            </select>
   223→          </label>
   224→          <label className="toolbar-control" style={{ marginBottom: 0 }}>
   225→            <select value={statusFilter} onChange={(e) => setStatusFilter(e.target.value)} style={{ minWidth: "140px" }}>
   226→              <option value="">Tous les statuts</option>
   227→              {STATUS_OPTIONS.map((o) => (<option key={o.value} value={o.value}>{o.label}</option>))}
   228→            </select>
   229→          </label>
   230→        </div>
   231→        {canManage && (
   232→          <button className="primary-button" onClick={() => setShowForm(!showForm)}>
   233→            {showForm ? "Annuler" : "Ajouter"}
   234→          </button>
   235→        )}
   236→      </div>
   237→
   238→      {showForm && (
   239→        <div className="card form-card">
   240→          <h3>Nouveau membre du personnel</h3>
   241→          <form onSubmit={handleSubmit} className="form-grid">
   242→            <label className="form-control">Prénom<input type="text" value={firstName} onChange={(e) => setFirstName(e.target.value)} required /></label>
   243→            <label className="form-control">Nom<input type="text" value={lastName} onChange={(e) => setLastName(e.target.value)} required /></label>
   244→            <label className="form-control">Profession
   245→              <select value={role} onChange={(e) => setRole(e.target.value)}>
   246→                {STAFF_ROLE_OPTIONS.map((o) => (<option key={o.value} value={o.value}>{o.label}</option>))}
   247→              </select>
   248→            </label>
   249→            <label className="form-control">Spécialité<input type="text" value={specialty} onChange={(e) => setSpecialty(e.target.value)} placeholder="Ex: Chirurgie" /></label>
   250→            <label className="form-control">Téléphone<input type="tel" value={phone} onChange={(e) => setPhone(e.target.value)} /></label>
   251→            <label className="form-control">Email<input type="email" value={email} onChange={(e) => setEmail(e.target.value)} /></label>
   252→            <label className="form-control">Type contrat
   253→              <select value={contractType} onChange={(e) => setContractType(e.target.value)}>
   254→                {CONTRACT_TYPE_OPTIONS.map((o) => (<option key={o.value} value={o.value}>{o.label}</option>))}
   255→              </select>
   256→            </label>
   257→            <label className="form-control">Échelle salariale<input type="text" value={salaryGrade} onChange={(e) => setSalaryGrade(e.target.value)} placeholder="Ex: A5" /></label>
   258→            <label className="form-control">Date d'embauche<input type="date" value={hireDate} onChange={(e) => setHireDate(e.target.value)} /></label>
   259→            <div className="form-actions">
   260→              <button className="primary-button" type="submit" disabled={submitting}>{submitting ? "Ajout..." : "Ajouter"}</button>
   261→            </div>
   262→          </form>
   263→        </div>
   264→      )}
   265→
   266→      {loading ? (
   267→        <div className="card" style={{ textAlign: "center", padding: "32px" }}><div className="spinner" /></div>
   268→      ) : error ? (
   269→        <p style={{ color: "crimson" }}>{error}</p>
   270→      ) : staff.length === 0 ? (
   271→        <div className="card"><p className="muted">Aucun membre du personnel trouvé.</p></div>
   272→      ) : (
   273→        <div className="card">
   274→          <div className="table-wrapper">
   275→            <table className="table">
   276→              <thead>
   277→                <tr><th>N° Employé</th><th>Nom</th><th>Prénom</th><th>Profession</th><th>Spécialité</th><th>Statut</th><th>Contrat</th><th>Téléphone</th></tr>
   278→              </thead>
   279→              <tbody>
   280→                {staff.map((s) => (
   281→                  <tr key={s.id}>
   282→                    <td style={{ fontFamily: "monospace", fontSize: "12px" }}>{s.employee_number || "—"}</td>
   283→                    <td style={{ fontWeight: 600 }}>{s.last_name || "—"}</td>
   284→                    <td>{s.first_name || "—"}</td>
   285→                    <td>
   286→                      <span className={`badge ${roleBadge[s.profession || s.role] || "badge-gray"}`}>
   287→                        {roleLabel[s.profession || s.role] || s.profession || s.role}
   288→                      </span>
   289→                    </td>
   290→                    <td>{s.specialty || "—"}</td>
   291→                    <td>
   292→                      <span className={`badge ${statusBadge[s.status] || "badge-gray"}`}>
   293→                        {STATUS_OPTIONS.find((o) => o.value === s.status)?.label || s.status}
   294→                      </span>
   295→                    </td>
   296→                    <td>{contractTypeLabel[s.contract_type] || s.contract_type || "—"}</td>
   297→                    <td>{s.phone || "—"}</td>
   298→                  </tr>
   299→                ))}
   300→              </tbody>
   301→            </table>
   302→          </div>
   303→        </div>
   304→      )}
   305→    </>
   306→  );
   307→}

   309→/* ─── On-Call Tab ────────────────────────────────────────────── */

   311→function OnCallTab({ lookups, canManage }: { lookups: LookupData; canManage: boolean }) {
   312→  const [onCallList, setOnCallList] = useState<Row[]>([]);
   313→  const [loading, setLoading] = useState(true);
   314→  const [error, setError] = useState("");
   315→  const [showForm, setShowForm] = useState(false);
   316→  const [submitting, setSubmitting] = useState(false);
   317→  const [staffId, setStaffId] = useState("");
   318→  const [onCallDate, setOnCallDate] = useState("");
   319→  const [shift, setShift] = useState("DAY");
   320→  const [staffOptions, setStaffOptions] = useState<{ value: string; label: string }[]>([]);
   321→
   322→  const loadOnCall = useCallback(async () => {
   323→    setLoading(true);
   324→    setError("");
   325→    try {
   326→      const payload = await apiRequest<any>("/personnel/on-call?page_size=1000");
   327→      setOnCallList(Array.isArray(payload.data) ? payload.data : []);
   328→    } catch {
   329→      setError("Impossible de charger le planning de garde.");
   330→    } finally {
   331→      setLoading(false);
   332→    }
   333→  }, []);
   334→
   335→  const loadStaffForDropdown = useCallback(async () => {
   336→    try {
   337→      const payload = await apiRequest<any>("/personnel/staff?page_size=1000");
   338→      const list = Array.isArray(payload.data) ? payload.data : [];
   339→      setStaffOptions(
   340→        list.map((s: Row) => ({
   341→          value: s.id,
   342→          label: `${s.last_name || ""} ${s.first_name || ""} — ${roleLabel[s.profession || s.role] || s.profession || s.role || ""}`.trim(),
   343→        }))
   344→      );
   345→    } catch { /* silent */ }
   346→  }, []);
   347→
   348→  useEffect(() => {
   349→    loadOnCall();
   350→    const handler = () => loadOnCall();
   351→    window.addEventListener("refresh-resource", handler);
   352→    return () => window.removeEventListener("refresh-resource", handler);
   353→  }, [loadOnCall]);
   354→
   355→  useEffect(() => {
   356→    if (showForm) loadStaffForDropdown();
   357→  }, [showForm, loadStaffForDropdown]);
   358→
   359→  async function handleSubmit(e: React.FormEvent) {
   360→    e.preventDefault();
   361→    if (!staffId || !onCallDate) return;
   362→    setSubmitting(true);
   363→    try {
   364→      await apiRequest("/personnel/on-call", {
   365→        method: "POST",
   366→        body: JSON.stringify({ staff_id: staffId, on_call_date: onCallDate, shift }),
   367→      });
   368→      setStaffId(""); setOnCallDate(""); setShift("DAY"); setShowForm(false);
   369→      loadOnCall();
   370→      showToast("Garde planifiée.", "success");
   371→    } catch {
   372→      showToast("Erreur lors de la planification.", "error");
   373→    } finally {
   374→      setSubmitting(false);
   375→    }
   376→  }
   377→
   378→  function getStaffName(sid: string): string {
   379→    const staffMember = lookups.staff.find((s) => s.id === sid);
   380→    if (!staffMember) return "Inconnu";
   381→    return `${staffMember.first_name || ""} ${staffMember.last_name || ""}`.trim() || staffMember.employee_number || "N/A";
   382→  }
   383→
   384→  const byDate = onCallList.reduce<Record<string, Row[]>>((acc, item) => {
   385→    const dateKey = item.on_call_date ? item.on_call_date.toString().split("T")[0] : "—";
   386→    if (!acc[dateKey]) acc[dateKey] = [];
   387→    acc[dateKey].push(item);
   388→    return acc;
   389→  }, {});
   390→
   391→  const sortedDates = Object.keys(byDate).sort((a, b) => a.localeCompare(b));
   392→
   393→  return (
   394→    <>
   395→      <div className="section-header">
   396→        <h2>Planning de garde</h2>
   397→        {canManage && (
   398→          <button className="primary-button" onClick={() => setShowForm(!showForm)}>
   399→            {showForm ? "Annuler" : "Planifier une garde"}
   400→          </button>
   401→        )}
   402→      </div>
   403→
   404→      {showForm && (
   405→        <div className="card form-card">
   406→          <h3>Planifier une garde</h3>
   407→          <form onSubmit={handleSubmit} className="form-grid">
   408→            <label className="form-control">Personnel
   409→              <select value={staffId} onChange={(e) => setStaffId(e.target.value)} required>
   410→                <option value="">-- Choisir --</option>
   411→                {staffOptions.map((o) => (<option key={o.value} value={o.value}>{o.label}</option>))}
   412→              </select>
   413→            </label>
   414→            <label className="form-control">Date de garde<input type="date" value={onCallDate} onChange={(e) => setOnCallDate(e.target.value)} required /></label>
   415→            <label className="form-control">Tranche horaire
   416→              <select value={shift} onChange={(e) => setShift(e.target.value)}>
   417→                <option value="DAY">Jour</option><option value="NIGHT">Nuit</option><option value="FULL_DAY">24h</option>
   418→              </select>
   419→            </label>
   420→            <div className="form-actions">
   421→              <button className="primary-button" type="submit" disabled={submitting}>{submitting ? "Planification..." : "Planifier"}</button>
   422→            </div>
   423→          </form>
   424→        </div>
   425→      )}
   426→
   427→      {loading ? (
   428→        <div className="card" style={{ textAlign: "center", padding: "32px" }}><div className="spinner" /></div>
   429→      ) : error ? (
   430→        <p style={{ color: "crimson" }}>{error}</p>
   431→      ) : onCallList.length === 0 ? (
   432→        <div className="card"><p className="muted">Aucune garde planifiée.</p></div>
   433→      ) : (
   434→        sortedDates.map((date) => (
   435→          <div className="card" key={date}>
   436→            <h3 style={{ marginBottom: "12px" }}>
   437→              {new Date(date).toLocaleDateString("fr-FR", { weekday: "long", year: "numeric", month: "long", day: "numeric" })}
   438→            </h3>
   439→            <div className="table-wrapper">
   440→              <table className="table">
   441→                <thead><tr><th>Personnel</th><th>Tranche</th></tr></thead>
   442→                <tbody>
   443→                  {byDate[date].map((item) => (
   444→                    <tr key={item.id}>
   445→                      <td style={{ fontWeight: 600 }}>{getStaffName(item.staff_id)}</td>
   446→                      <td>
   447→                        <span className={`badge ${shiftBadge[item.shift_type || item.shift] || "badge-gray"}`}>
   448→                          {shiftLabel[item.shift_type || item.shift] || item.shift_type || item.shift}
   449→                        </span>
   450→                      </td>
   451→                    </tr>
   452→                  ))}
   453→                </tbody>
   454→              </table>
   455→            </div>
   456→          </div>
   457→        ))
   458→      )}
   459→    </>
   460→  );
   461→}
