   143→   143→function TableauTab({
   144→   144→  lookups,
   145→   145→  onCreated,
   146→   146→}: {
   147→   147→  lookups: LookupData;
   148→   148→  onCreated: () => void;
   149→   149→}) {
   150→   150→  const [admissions, setAdmissions] = useState<Row[]>([]);
   151→   151→  const [loading, setLoading] = useState(true);
   152→   152→  const [statusFilter, setStatusFilter] = useState("OPEN");
   153→   153→  const [deptFilter, setDeptFilter] = useState("");
   154→   154→  const [dateFrom, setDateFrom] = useState("");
   155→   155→  const [dateTo, setDateTo] = useState("");
   156→   156→  const [refreshKey, setRefreshKey] = useState(0);
   157→   157→
   158→   158→  const options = buildOptions(lookups);
   159→   159→
   160→   160→  const loadAdmissions = useCallback(async () => {
   161→   161→    setLoading(true);
   162→   162→    try {
   163→   163→      const params = new URLSearchParams();
   164→   164→      if (statusFilter) params.set("search", statusFilter);
   165→   165→      const qs = params.toString();
   166→   166→      const payload = await apiRequest<any>(`/admissions${qs ? `?${qs}&page_size=1000` : "?page_size=1000"}`);
   167→   167→      const data: Row[] = Array.isArray(payload.data) ? payload.data : [];
   168→   168→      setAdmissions(data);
   169→   169→    } catch {
   170→   170→      /* silent */
   171→   171→    } finally {
   172→   172→      setLoading(false);
   173→   173→    }
   174→   174→  }, [statusFilter]);
   175→   175→
   176→   176→  useEffect(() => {
   177→   177→    loadAdmissions();
   178→   178→  }, [loadAdmissions, refreshKey]);
   179→   179→
   180→   180→  useEffect(() => {
   181→   181→    const handler = () => setRefreshKey((k) => k + 1);
   182→   182→    window.addEventListener("refresh-resource", handler);
   183→   183→    return () => window.removeEventListener("refresh-resource", handler);
   184→   184→  }, []);
   185→   185→
   186→   186→  function handleRefresh() {
   187→   187→    setRefreshKey((k) => k + 1);
   188→   188→    onCreated();
   189→   189→  }
   190→   190→
   191→   191→  /* ── Resolve names ─────────────────────────────── */
   192→   192→  function getPatientName(patientId: string): string {
   193→   193→    const p = lookups.patients.find((x) => x.id === patientId);
   194→   194→    if (!p) return "Inconnu";
   195→   195→    return `${p.first_name || ""} ${p.last_name || ""}`.trim() || p.patient_number || "N/A";
   196→   196→  }
   197→   197→
   198→   198→  function getDeptName(deptId: string | null): string {
   199→   199→    if (!deptId) return "—";
   200→   200→    const d = lookups.departments.find((x) => x.id === deptId);
   201→   201→    return d ? `${d.code || ""} - ${d.name || d.id}`.trim() : "—";
   202→   202→  }
   203→   203→
   204→   204→  function getFacilityName(facilityId: string): string {
   205→   205→    const f = lookups.facilities.find((x) => x.id === facilityId);
   206→   206→    return f ? f.name || f.id : "—";
   207→   207→  }
   208→   208→
   209→   209→  /* ── Compute KPIs ──────────────────────────────── */
   210→   210→  const now = new Date();
   211→   211→  const todayStr = now.toDateString();
   212→   212→
   213→   213→  const kpis = useMemo(() => {
   214→   214→    const active = admissions.filter(
   215→   215→      (a) => a.status === "OPEN" || a.status === "ACTIVE"
   216→   216→    );
   217→   217→    const todayAdmissions = admissions.filter((a) => {
   218→   218→      if (!a.admitted_at) return false;
   219→   219→      return new Date(a.admitted_at).toDateString() === todayStr;
   220→   220→    });
   221→   221→    const hospitalized = admissions.filter(
   222→   222→      (a) =>
   223→   223→        (a.status === "OPEN" || a.status === "ACTIVE") &&
   224→   224→        a.admission_type === "HOSPITALIZATION"
   225→   225→    );
   226→   226→    const plannedDischarges = admissions.filter(
   227→   227→      (a) => a.status === "CLOSED" || a.status === "DISCHARGED"
   228→   228→    );
   229→   229→    return {
   230→   230→      activeCount: active.length,
   231→   231→      todayCount: todayAdmissions.length,
   232→   232→      hospCount: hospitalized.length,
   233→   233→      dischargeCount: plannedDischarges.length,
   234→   234→    };
   235→   235→  }, [admissions, todayStr]);
   236→   236→
   237→   237→  /* ── Apply filters ─────────────────────────────── */
   238→   238→  const filtered = useMemo(() => {
   239→   239→    let rows = admissions.filter(
   240→   240→      (a) => a.status === "OPEN" || a.status === "ACTIVE"
   241→   241→    );
   242→   242→    if (statusFilter === "ALL") {
   243→   243→      rows = admissions;
   244→   244→    } else if (statusFilter === "CLOSED") {
   245→   245→      rows = admissions.filter(
   246→   246→        (a) => a.status === "CLOSED" || a.status === "DISCHARGED"
   247→   247→      );
   248→   248→    }
   249→   249→    if (deptFilter) {
   250→   250→      rows = rows.filter((a) => a.department_id === deptFilter);
   251→   251→    }
   252→   252→    if (dateFrom) {
   253→   253→      rows = rows.filter(
   254→   254→        (a) => a.admitted_at && new Date(a.admitted_at) >= new Date(dateFrom)
   255→   255→      );
   256→   256→    }
   257→   257→    if (dateTo) {
   258→   258→      const to = new Date(dateTo);
   259→   259→      to.setHours(23, 59, 59, 999);
   260→   260→      rows = rows.filter(
   261→   261→        (a) => a.admitted_at && new Date(a.admitted_at) <= to
   262→   262→      );
   263→   263→    }
   264→   264→    return rows;
   265→   265→  }, [admissions, statusFilter, deptFilter, dateFrom, dateTo]);
   266→   266→
   267→   267→  /* ── Color coding helpers ──────────────────────── */

   272→   272→    if (diffHours < 24) {
   273→   273→      return (

   285→   285→        />

   287→   287→    }
   288→   288→    const diffDays = diffHours / 24;
   289→   289→    if (diffDays > 7) {
   290→   290→      return (
   291→   291→        <span
   292→   292→          style={{
   293→   293→            display: "inline-block",
   294→   294→            width: "8px",
   295→   295→            height: "8px",

   297→   297→            background: "#f59e0b",
   298→   298→            marginRight: "6px",
   299→   299→            flexShrink: 0,
   300→   300→          }}
   301→   301→          title="Séjour long (> 7 jours)"
   302→   302→        />
   303→   303→      );
   304→   304→    }
   305→   305→    return null;
   306→   306→  }

   321→   321→      {/* ── KPI Cards ──────────────────────────────────── */}
   322→   322→      <div

   325→   325→          gridTemplateColumns: "repeat(auto-fit, minmax(200px, 1fr))",

   332→   332→          label="Admissions actives"

   336→   336→        />
   337→   337→        <KpiCard

   340→   340→          value={kpis.todayCount}
   341→   341→          color="#2563eb"
   342→   342→          bgColor="#eff6ff"
   343→   343→        />
   344→   344→        <KpiCard
   345→   345→          icon={<BedDouble size={20} />}
   346→   346→          label="En hospitalisation"
   347→   347→          value={kpis.hospCount}
   348→   348→          color="#7c3aed"
   349→   349→          bgColor="#f5f3ff"

   360→   360→      {/* ── Filters ────────────────────────────────────── */}
   361→   361→      <div
   362→   362→        className="card"
   363→   363→        style={{
   364→   364→          marginBottom: "16px",
   365→   365→          display: "flex",
   366→   366→          gap: "12px",

   379→   379→            <option value="OPEN">Actives</option>
   380→   380→            <option value="CLOSED">Sortis</option>

   441→   441→      {loading ? (
