   585→  }, [loadIncidents]);

   587→  async function handleSubmit(e: React.FormEvent) {
   588→    e.preventDefault();
   589→    if (!description) return;
   590→    setSubmitting(true);
   591→    try {
   592→      await apiRequest("/quality/incidents", {
   593→        method: "POST",
   594→        body: JSON.stringify({
   595→          incident_date: incidentDate || undefined,
   596→          incident_type: incidentType,
   597→          severity,
   598→          description: description.trim(),
   599→          immediate_actions: immediateActions.trim() || undefined,
   600→        }),
   601→      });
   602→      setIncidentDate("");
   603→      setIncidentType("OTHER");
   604→      setSeverity("MINOR");
   605→      setDescription("");
   606→      setImmediateActions("");
   607→      setShowForm(false);
   608→      loadIncidents();
   609→      showToast("Événement indésirable signalé avec succès.", "success");
   610→    } catch {
   611→      showToast("Erreur lors du signalement.", "error");
   612→    } finally {
   613→      setSubmitting(false);
   614→    }
   615→  }

   689→                placeholder="Actions immédiates prises..."

   691→              />
   692→            </label>
