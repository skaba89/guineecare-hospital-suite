   268→        </div>

   319→          <div className="table-wrapper">
   320→            <table className="table">
   321→              <thead>
   322→                <tr>
   323→                  <th>Produit</th>
   324→                  <th>Catégorie</th>
   325→                  <th>Forme</th>

   338→                    </td>
   339→                  </tr>

   341→                  filtered.map((item: Row) => {
   342→                    const status = getStockStatus(item);
   343→                    return (
   344→                      <tr
   345→                        key={item.id}
   346→                        style={{
   347→                          background:
   348→                            status === "epuise"
   349→                              ? "var(--danger-light)"
   350→                              : status === "bas"
   351→                              ? "var(--warning-light)"
   352→                              : "inherit",
   353→                        }}
   354→                      >

   360→                              style={{ fontWeight: 400, marginLeft: "8px", fontSize: "12px" }}

   457→/* ═════════════════════════════════════════════════════════════════
   458→   Dispensation Tab
   459→   ═════════════════════════════════════════════════════════════════ */
   460→
   461→interface DispensationLine {
   462→  id: string;
   463→  product_id: string;
   464→  quantity: number;
   465→}
   466→
   467→function DispensationTab({
   468→  lookups,
   469→  onCreated,
   470→}: {
   471→  lookups: LookupData;
   472→  onCreated: () => void;
   473→}) {
   474→  const options = buildOptions(lookups);
   475→  const facilityId = firstValue(options.facilities);
   476→  const doctors = lookups.staff.filter((s) => s.role === "DOCTOR");
   477→  const doctorOptions = doctors.map((d) => ({
   478→    value: d.id,
   479→    label: `${d.last_name || ""} ${d.first_name || ""} — ${d.role || ""}`.trim(),
   480→  }));
   481→
   482→  const [patientId, setPatientId] = useState("");
   483→  const [prescription, setPrescription] = useState("");
   484→  const [prescriberId, setPrescriberId] = useState("");
   485→  const [lines, setLines] = useState<DispensationLine[]>([]);
   486→  const [submitting, setSubmitting] = useState(false);
   487→
   488→  const [stock, setStock] = useState<Row[]>([]);
   489→  const [recentDispensations, setRecentDispensations] = useState<Row[]>([]);
   490→
   491→  const loadStock = useCallback(async () => {
   492→    try {
   493→      const res = await apiRequest<any>("/pharmacy/stock?page_size=1000");
   494→      setStock(Array.isArray(res.data) ? res.data : []);
   495→    } catch {
   496→      /* silent */
   497→    }
   498→  }, []);
   499→
   500→  const loadMovements = useCallback(async () => {
   501→    try {
   502→      const res = await apiRequest<any>("/pharmacy/stock/movements?page_size=1000");
   503→      const movements = Array.isArray(res.data) ? res.data : [];
   504→      setRecentDispensations(movements.filter((m: Row) => m.movement_type === "OUT"));
   505→    } catch {
   506→      /* silent */
   507→    }
   508→  }, []);
   509→
   510→  useEffect(() => {
   511→    loadStock();
   512→    loadMovements();
   513→    const handler = () => {
   514→      loadStock();
   515→      loadMovements();
   516→    };
   517→    window.addEventListener("refresh-resource", handler);
   518→    return () => window.removeEventListener("refresh-resource", handler);
   519→  }, [loadStock, loadMovements]);
   520→
   521→  function addLine() {
   522→    setLines([
   523→      ...lines,
   524→      { id: `line-${Date.now()}`, product_id: "", quantity: 1 },
   525→    ]);
   526→  }
   527→
   528→  function removeLine(lineId: string) {
   529→    setLines(lines.filter((l) => l.id !== lineId));
   530→  }
   531→
   532→  function updateLine(lineId: string, field: "product_id" | "quantity", value: string | number) {
   533→    setLines(
   534→      lines.map((l) => (l.id === lineId ? { ...l, [field]: value } : l))
   535→    );
   536→  }
   537→
   538→  function getStockForProduct(productId: string): number {
   539→    const item = stock.find((s) => s.product_id === productId);
   540→    return item?.quantity_available ?? 0;
   541→  }
   542→
   543→  async function handleDispense(e: React.FormEvent) {
   544→    e.preventDefault();
   545→    if (!patientId || lines.length === 0) {
   546→      showToast("Veuillez sélectionner un patient et au moins un produit.", "error");
   547→      return;
   548→    }
   549→
   550→    const invalidLines = lines.filter(
   551→      (l) => !l.product_id || l.quantity <= 0
   552→    );
   553→    if (invalidLines.length > 0) {
   554→      showToast("Veuillez remplir toutes les lignes de prescription.", "error");
   555→      return;
   556→    }
   557→
   558→    setSubmitting(true);
   559→    try {
   560→      for (const line of lines) {
   561→        await apiRequest("/pharmacy/stock/movements", {
   562→          method: "POST",
   563→          body: JSON.stringify({
   564→            facility_id: facilityId,
   565→            product_id: line.product_id,
   566→            movement_type: "OUT",
   567→            quantity: line.quantity,
   568→            reason: `Dispensation — Patient: ${patientId} — Prescripteur: ${prescriberId || "N/A"} — Ordonnance: ${prescription || "N/A"}`,
   569→          }),
   570→        });
   571→      }
   572→      showToast("Dispensation enregistrée avec succès.", "success");
   573→      setPatientId("");
   574→      setPrescription("");
   575→      setPrescriberId("");
   576→      setLines([]);
   577→      onCreated();
   578→      loadStock();
   579→      loadMovements();
   580→    } catch (err: any) {
   581→      showToast(err.message || "Erreur lors de la dispensation.", "error");
   582→    } finally {
   583→      setSubmitting(false);
   584→    }
   585→  }
   586→
   587→  return (
   588→    <>
   589→      {/* Dispensation Form */}
   590→      <div className="card form-card">
