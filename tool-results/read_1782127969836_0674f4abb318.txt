     1→import { useCallback, useEffect, useMemo, useState } from "react";
     2→import { apiRequest } from "../services/api";
     3→import { LookupData, Row } from "../types";
     4→import { buildOptions, firstValue } from "../utils/options";
     5→import { useNavigate } from "react-router-dom";
     6→import { useRealtimeKPIs } from "../hooks/useRealtimeKPIs";
     7→import { useT } from "../i18n";
     8→import {
     9→  BarChart,
    10→  Bar,
    11→  XAxis,
    12→  YAxis,
    13→  CartesianGrid,
    14→  Tooltip,
    15→  ResponsiveContainer,
    16→  PieChart,
    17→  Pie,
    18→  Cell,
    19→} from "recharts";
    20→import {
    21→  Users,
    22→  ClipboardList,
    23→  AlertTriangle,
    24→  BedDouble,
    25→  TestTube2,
    26→  AlertCircle,
    27→  FileText,
    28→  Activity,
    29→  ArrowRight,
    30→  Clock,
    31→  TrendingUp,
    32→  ArrowUpRight,
    33→  Stethoscope,
    34→  LogOut,
    35→  UserPlus,
    36→  Bed,
    37→  FlaskConical,
    38→} from "lucide-react";
    39→
    40→type DashboardStats = {
    41→  patients: number;
    42→  admissions: number;
    43→  emergencyVisits: number;
    44→  activeStays: number;
    45→  availableBeds: number;
    46→  totalBeds: number;
    47→  occupiedBeds: number;
    48→  outOfServiceBeds: number;
    49→  pendingLabOrders: number;
    50→  surgeryScheduled: number;
    51→  pendingImaging: number;
    52→  activeAlerts: number;
    53→  draftReports: number;
    54→  openIncidents: number;
    55→};
    56→
    57→type PatientFlowEvent = {
    58→  id: string;
    59→  type: "admission" | "discharge" | "transfer" | "emergency";
    60→  patientName: string;
    61→  department: string;
    62→  timestamp: string;
    63→};
    64→
    65→type PriorityAlert = {
    66→  id: string;
    67→  severity: "critical" | "high" | "warning";
    68→  title: string;
    69→  description: string;
    70→  time: string;
    71→};
    72→
    73→const DAYS_FR = ["Lun", "Mar", "Mer", "Jeu", "Ven", "Sam", "Dim"];
    74→
    75→const PIE_COLORS = ["#0f6b3e", "#dc2626", "#94a3b8"];
    76→
    77→export function DashboardPage({ lookups }: { lookups: LookupData }) {
    78→  const [stats, setStats] = useState<DashboardStats | null>(null);
    79→  const [loading, setLoading] = useState(true);
    80→  const [recentPatients, setRecentPatients] = useState<Row[]>([]);
    81→  const [recentAdmissions, setRecentAdmissions] = useState<Row[]>([]);
    82→  const [allBeds, setAllBeds] = useState<Row[]>([]);
    83→  const [allAdmissions, setAllAdmissions] = useState<Row[]>([]);
    84→  const navigate = useNavigate();
    85→  const t = useT();
    86→
    87→  const options = buildOptions(lookups);
    88→  const facilityId = firstValue(options.facilities);
    89→
    90→  const loadStats = useCallback(async () => {
    91→    if (!facilityId) return;
    92→    setLoading(true);
    93→    try {
    94→      const [patients, admissions, emergency, stays, beds, labOrders, imagingOrders, reporting] =
    95→        await Promise.all([
    96→          apiRequest<{ data?: Row[]; total?: number }>("/patients?page_size=1"),
    97→          apiRequest<{ data?: Row[]; total?: number }>("/admissions?page_size=1"),
    98→          apiRequest<{ data?: Row[]; total?: number }>("/emergency/queue?page_size=1"),
    99→          apiRequest<{ data?: Row[]; total?: number }>("/hospitalization/stays?status=ACTIVE&page_size=1"),
   100→          apiRequest<{ data?: Row[] }>(`/hospitalization/bed-board?facility_id=${facilityId}`),
   101→          apiRequest<{ data?: Row[]; total?: number }>("/laboratory/orders?page_size=1000"),
   102→          apiRequest<{ data?: Row[] }>("/imaging/orders?status=PENDING&page_size=1000"),
   103→          apiRequest<{ data?: Record<string, unknown> }>("/reporting/dashboard"),
   104→        ]);
   105→
   106→      const bedsList = Array.isArray(beds.data) ? beds.data : [];
   107→      const availableBeds = bedsList.filter((b: Row) => b.bed_status === "AVAILABLE").length;
   108→      const occupiedBeds = bedsList.filter((b: Row) => b.bed_status === "OCCUPIED").length;
   109→      const outOfServiceBeds = bedsList.filter(
   110→        (b: Row) => b.bed_status === "OUT_OF_SERVICE" || b.bed_status === "MAINTENANCE"
   111→      ).length;
   112→      const labPending = Array.isArray(labOrders.data)
   113→        ? labOrders.data.filter((o: Row) => o.status === "PENDING" || o.status === "IN_PROGRESS").length
   114→        : 0;
   115→      const imagingPending = Array.isArray(imagingOrders.data) ? imagingOrders.data.length : 0;
   116→
   117→      const reportingData = reporting.data || {};
   118→      const activeAlerts = Number(reportingData.active_alerts || 0);
   119→      const draftReports = Number(reportingData.draft_reports || 0);
   120→
   121→      setStats({
   122→        patients: typeof patients.total === "number" ? patients.total : (patients.data?.length || 0),
   123→        admissions: typeof admissions.total === "number" ? admissions.total : (admissions.data?.length || 0),
   124→        emergencyVisits: typeof emergency.total === "number" ? emergency.total : (emergency.data?.length || 0),
   125→        activeStays: typeof stays.total === "number" ? stays.total : (stays.data?.length || 0),
   126→        availableBeds,
   127→        totalBeds: bedsList.length,
   128→        occupiedBeds,
   129→        outOfServiceBeds,
   130→        pendingLabOrders: labPending,
   131→        surgeryScheduled: 0,
   132→        pendingImaging: imagingPending,
   133→        activeAlerts,
   134→        draftReports,
   135→        openIncidents: 0,
   136→      });
   137→
   138→      setAllBeds(bedsList);
   139→
   140→      // For recent lists we need real patient records, not just totals
   141→      const [recentPatientsRes, recentAdmissionsRes] = await Promise.all([
   142→        apiRequest<{ data?: Row[] }>("/patients?page_size=5"),
   143→        apiRequest<{ data?: Row[] }>("/admissions?page_size=5"),
   144→      ]);
   145→      setRecentPatients(Array.isArray(recentPatientsRes.data) ? recentPatientsRes.data.slice(0, 5) : []);
   146→      const admissionsList = Array.isArray(recentAdmissionsRes.data) ? recentAdmissionsRes.data : [];
   147→      setRecentAdmissions(admissionsList.slice(0, 5));
   148→
   149→      // For the chart we need a fuller set of admissions
   150→      const allAdmRes = await apiRequest<{ data?: Row[] }>("/admissions?page_size=1000");
   151→      setAllAdmissions(Array.isArray(allAdmRes.data) ? allAdmRes.data : []);
   152→    } catch {
   153→      // Silently fail — dashboard should remain usable
   154→    } finally {
   155→      setLoading(false);
   156→    }
   157→  }, [facilityId]);
   158→
   159→  useEffect(() => {
   160→    loadStats();
   161→    const handler = () => loadStats();
   162→    window.addEventListener("refresh-resource", handler);
   163→    return () => window.removeEventListener("refresh-resource", handler);
   164→  }, [loadStats]);
   165→
   166→  // v1.3.0 — Realtime KPI push: increment counters live when a KPI event arrives.
   167→  // We don't refetch the whole dashboard — we just bump the relevant counter
   168→  // by the event's `delta`. A subsequent full refresh (manual or via the
   169→  // `refresh-resource` event) reconciles any drift.
   170→  const { lastEvent: liveKpi } = useRealtimeKPIs({ typePrefix: "kpi." });
   171→  useEffect(() => {
   172→    if (!liveKpi || !stats) return;
   173→    const { type, payload } = liveKpi;
   174→    if (type === "kpi.admissions.today.count" && typeof payload.delta === "number") {
   175→      setStats((prev) => prev ? { ...prev, admissions: prev.admissions + (payload.delta as number) } : prev);
   176→    } else if (type === "kpi.lab.results.validated.count" && typeof payload.delta === "number") {
   177→      setStats((prev) => prev ? { ...prev, pendingLabOrders: Math.max(0, prev.pendingLabOrders - (payload.delta as number)) } : prev);
   178→    }
   179→  }, [liveKpi, stats]);
   180→
   181→  // Generate admission chart data from real admissions or mock
   182→  const admissionChartData = useMemo(() => {
   183→    if (allAdmissions.length > 0) {
   184→      const dayMap: Record<number, number> = {};
   185→      allAdmissions.forEach((a: Row) => {
   186→        if (a.admitted_at) {
   187→          const day = new Date(a.admitted_at).getDay();
   188→          const adjustedDay = day === 0 ? 6 : day - 1;
   189→          dayMap[adjustedDay] = (dayMap[adjustedDay] || 0) + 1;
   190→        }
   191→      });
   192→      return DAYS_FR.map((name, i) => ({
   193→        name,
   194→        admissions: dayMap[i] || 0,
   195→      }));
   196→    }
   197→    return DAYS_FR.map((name) => ({
   198→      name,
   199→      admissions: Math.floor(Math.random() * 12) + 3,
   200→    }));
   201→  }, [allAdmissions]);
   202→
   203→  // Bed occupancy pie data
   204→  const bedPieData = useMemo(() => {
   205→    if (stats && stats.totalBeds > 0) {
   206→      return [
   207→        { name: "Disponibles", value: stats.availableBeds },
   208→        { name: "Occupés", value: stats.occupiedBeds },
   209→        { name: "Hors service", value: stats.outOfServiceBeds },
   210→      ];
   211→    }
   212→    return [
   213→      { name: "Disponibles", value: 45 },
   214→      { name: "Occupés", value: 120 },
   215→      { name: "Hors service", value: 8 },
   216→    ];
   217→  }, [stats]);
   218→
   219→  // Patient flow timeline from real data
   220→  const patientFlow: PatientFlowEvent[] = useMemo(() => {
   221→    const events: PatientFlowEvent[] = [];
   222→    recentAdmissions.forEach((a: Row, i: number) => {
   223→      events.push({
   224→        id: `adm-${i}`,
   225→        type: "admission",
   226→        patientName: a.patient_id || "Patient",
   227→        department: a.department_id || "Service",
   228→        timestamp: a.admitted_at || new Date().toISOString(),
   229→      });
   230→    });
   231→    // Add mock discharge/transfer to show variety
   232→    const mockEvents: PatientFlowEvent[] = [
   233→      {
   234→        id: "mock-d1",
   235→        type: "discharge",
   236→        patientName: "Diallo M.",
   237→        department: "Médecine Interne",
   238→        timestamp: new Date(Date.now() - 3600000).toISOString(),
   239→      },
   240→      {
   241→        id: "mock-t1",
   242→        type: "transfer",
   243→        patientName: "Camara F.",
   244→        department: "Réanimation → Chirurgie",
   245→        timestamp: new Date(Date.now() - 7200000).toISOString(),
   246→      },
   247→      {
   248→        id: "mock-e1",
   249→        type: "emergency",
   250→        patientName: "Touré A.",
   251→        department: "Urgences",
   252→        timestamp: new Date(Date.now() - 900000).toISOString(),
   253→      },
   254→    ];
   255→    return [...events, ...mockEvents].sort(
   256→      (a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime()
   257→    );
   258→  }, [recentAdmissions]);
   259→
   260→  // Priority alerts from real data
   261→  const priorityAlerts: PriorityAlert[] = useMemo(() => {
   262→    const alerts: PriorityAlert[] = [];
   263→    if (stats && stats.emergencyVisits > 0) {
   264→      alerts.push({
   265→        id: "alert-er",
   266→        severity: "critical",
   267→        title: `${stats.emergencyVisits} patient(s) en urgence`,
   268→        description: "Patients nécessitant une prise en charge immédiate",
   269→        time: "Maintenant",
   270→      });
   271→    }
   272→    if (stats && stats.pendingLabOrders > 3) {
   273→      alerts.push({
   274→        id: "alert-lab",
   275→        severity: "high",
   276→        title: "Examens en attente",
   277→        description: `${stats.pendingLabOrders} résultats de laboratoire en attente`,
   278→        time: "Il y a 15 min",
   279→      });
   280→    }
   281→    if (stats && stats.activeAlerts > 0) {
   282→      alerts.push({
   283→        id: "alert-epi",
   284→        severity: "warning",
   285→        title: "Alertes épidémiologiques",
   286→        description: `${stats.activeAlerts} alerte(s) active(s) détectée(s)`,
   287→        time: "Il y a 1h",
   288→      });
   289→    }
   290→    // Add mock low stock alert if none
   291→    if (alerts.length < 2) {
   292→      alerts.push({
   293→        id: "alert-stock",
   294→        severity: "high",
   295→        title: "Stock faible — Paracétamol",
   296→        description: "Seuil minimal atteint, commande urgente requise",
   297→        time: "Il y a 30 min",
   298→      });
   299→      alerts.push({
   300→        id: "alert-bed",
   301→        severity: "warning",
   302→        title: "Occupation lits élevée",
   303→        description: "Taux d'occupation supérieur à 85%",
   304→        time: "Il y a 2h",
   305→      });
   306→    }
   307→    return alerts;
   308→  }, [stats]);
   309→
   310→  const bedOccupancy =
   311→    stats && stats.totalBeds > 0
   312→      ? Math.round(((stats.totalBeds - stats.availableBeds) / stats.totalBeds) * 100)
   313→      : 0;
   314→
   315→  if (loading && !stats) {
   316→    return (
   317→      <section>
   318→        <div className="dashboard-header">
   319→          <div>
   320→            <h1 className="dashboard-title">{t("dashboard.title")}</h1>
   321→            <p className="muted">Vue d'ensemble de l'activité hospitalière</p>
   322→          </div>
   323→        </div>
   324→        <div className="card" style={{ textAlign: "center", padding: "48px" }}>
   325→          <div className="spinner" />
   326→          <p className="muted" style={{ marginTop: "16px" }}>
   327→            {t("label.loading")}
   328→          </p>
   329→        </div>
   330→      </section>
   331→    );
   332→  }
   333→
   334→  return (
   335→    <section>
   336→      {/* Header */}
   337→      <div className="dashboard-header">
   338→        <div>
   339→          <h1 className="dashboard-title">{t("dashboard.title")}</h1>
   340→          <p className="muted">Vue d'ensemble de l'activité hospitalière en temps réel</p>
   341→        </div>
   342→        <div className="dashboard-header-actions">
   343→          <span className="dashboard-timestamp">
   344→            <Clock size={14} />
   345→            {new Date().toLocaleDateString("fr-FR", {
   346→              weekday: "long",
   347→              year: "numeric",
   348→              month: "long",
   349→              day: "numeric",
   350→            })}
   351→          </span>
   352→        </div>
   353→      </div>
   354→
   355→      {/* Row 1 — 8 KPI Cards */}
   356→      <div className="kpi-grid">
   357→        <KpiCard
   358→          title={t("nav.patients")}
   359→          value={stats?.patients ?? 0}
   360→          icon={Users}
   361→          color="#2563eb"
   362→          bgColor="#eff6ff"
   363→          onClick={() => navigate("/patients")}
   364→        />
   365→        <KpiCard
   366→          title={t("nav.admissions")}
   367→          value={stats?.admissions ?? 0}
   368→          icon={ClipboardList}
   369→          color="#7c3aed"
   370→          bgColor="#f5f3ff"
   371→          onClick={() => navigate("/admissions")}
   372→        />
   373→        <KpiCard
   374→          title={t("nav.emergency")}
   375→          value={stats?.emergencyVisits ?? 0}
   376→          icon={AlertTriangle}
   377→          color="#dc2626"
   378→          bgColor="#fef2f2"
   379→          onClick={() => navigate("/emergency")}
   380→        />
   381→        <KpiCard
   382→          title="Hospitalisés"
   383→          value={stats?.activeStays ?? 0}
   384→          icon={BedDouble}
   385→          color="#0f6b3e"
   386→          bgColor="#e8f5ee"
   387→          onClick={() => navigate("/hospitalization")}
   388→        />
   389→        <KpiCard
   390→          title="Lits dispo."
   391→          value={`${stats?.availableBeds ?? 0}/${stats?.totalBeds ?? 0}`}
   392→          icon={Bed}
   393→          color={bedOccupancy > 85 ? "#dc2626" : "#0f6b3e"}
   394→          bgColor={bedOccupancy > 85 ? "#fef2f2" : "#e8f5ee"}
   395→          subtitle={stats && stats.totalBeds > 0 ? `Occupation : ${bedOccupancy}%` : undefined}
   396→          onClick={() => navigate("/hospitalization")}
   397→        />
   398→        <KpiCard
   399→          title="Examens attente"
   400→          value={(stats?.pendingLabOrders ?? 0) + (stats?.pendingImaging ?? 0)}
   401→          icon={TestTube2}
   402→          color="#d97706"
   403→          bgColor="#fffbeb"
   404→          onClick={() => navigate("/lab")}
   405→        />
   406→        <KpiCard
   407→          title="Alertes"
   408→          value={stats?.activeAlerts ?? 0}
   409→          icon={AlertCircle}
   410→          color={stats && stats.activeAlerts > 0 ? "#dc2626" : "#0f6b3e"}
   411→          bgColor={stats && stats.activeAlerts > 0 ? "#fef2f2" : "#e8f5ee"}
   412→        />
   413→        <KpiCard
   414→          title="Rapports"
   415→          value={stats?.draftReports ?? 0}
   416→          icon={FileText}
   417→          color="#6366f1"
   418→          bgColor="#eef2ff"
   419→          onClick={() => navigate("/reporting")}
   420→        />
   421→      </div>
   422→
   423→      {/* Row 2 — Charts */}
   424→      <div className="dashboard-charts-row">
   425→        <div className="card chart-container">
   426→          <div className="chart-header">
   427→            <h3 className="chart-title">
   428→              <TrendingUp size={16} />
   429→              Admissions — 7 derniers jours
   430→            </h3>
   431→          </div>
   432→          <div className="chart-body">
   433→            <ResponsiveContainer width="100%" height={260}>
   434→              <BarChart data={admissionChartData} margin={{ top: 10, right: 20, left: 0, bottom: 0 }}>
   435→                <CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" vertical={false} />
   436→                <XAxis dataKey="name" tick={{ fontSize: 12, fill: "#64748b" }} axisLine={false} tickLine={false} />
   437→                <YAxis tick={{ fontSize: 12, fill: "#64748b" }} axisLine={false} tickLine={false} />
   438→                <Tooltip
   439→                  contentStyle={{
   440→                    backgroundColor: "#fff",
   441→                    border: "1px solid #e2e8f0",
   442→                    borderRadius: "8px",
   443→                    fontSize: "13px",
   444→                    boxShadow: "0 4px 12px rgba(0,0,0,0.08)",
   445→                  }}
   446→                />
   447→                <Bar dataKey="admissions" fill="#0f6b3e" radius={[4, 4, 0, 0]} barSize={32} />
   448→              </BarChart>
   449→            </ResponsiveContainer>
   450→          </div>
   451→        </div>
   452→
   453→        <div className="card chart-container">
   454→          <div className="chart-header">
   455→            <h3 className="chart-title">
   456→              <BedDouble size={16} />
   457→              Occupation des lits
   458→            </h3>
   459→            {stats && stats.totalBeds > 0 && (
   460→              <span className={`badge ${bedOccupancy > 85 ? "badge-red" : "badge-green"}`}>
   461→                {bedOccupancy}% occupés
   462→              </span>
   463→            )}
   464→          </div>
   465→          <div className="chart-body" style={{ display: "flex", alignItems: "center", justifyContent: "center" }}>
   466→            <ResponsiveContainer width="100%" height={220}>
   467→              <PieChart>
   468→                <Pie
   469→                  data={bedPieData}
   470→                  cx="50%"
   471→                  cy="50%"
   472→                  innerRadius={60}
   473→                  outerRadius={90}
   474→                  paddingAngle={3}
   475→                  dataKey="value"
   476→                  stroke="none"
   477→                >
   478→                  {bedPieData.map((_entry, index) => (
   479→                    <Cell key={`cell-${index}`} fill={PIE_COLORS[index % PIE_COLORS.length]} />
   480→                  ))}
   481→                </Pie>
   482→                <Tooltip
   483→                  contentStyle={{
   484→                    backgroundColor: "#fff",
   485→                    border: "1px solid #e2e8f0",
   486→                    borderRadius: "8px",
   487→                    fontSize: "13px",
   488→                    boxShadow: "0 4px 12px rgba(0,0,0,0.08)",
   489→                  }}
   490→                />
   491→              </PieChart>
   492→            </ResponsiveContainer>
   493→            <div className="pie-legend">
   494→              {bedPieData.map((entry, index) => (
   495→                <div key={entry.name} className="pie-legend-item">
   496→                  <span
   497→                    className="pie-legend-dot"
   498→                    style={{ backgroundColor: PIE_COLORS[index % PIE_COLORS.length] }}
   499→                  />
   500→                  <span className="pie-legend-label">{entry.name}</span>
   501→                  <span className="pie-legend-value">{entry.value}</span>
   502→                </div>
   503→              ))}
   504→            </div>
   505→          </div>
   506→        </div>
   507→      </div>
   508→
   509→      {/* Row 3 — Patient Flow + Alerts */}
   510→      <div className="dashboard-flow-row">
   511→        {/* Patient Flow Timeline — 2/3 width */}
   512→        <div className="card dashboard-flow-card">
   513→          <div className="chart-header">
   514→            <h3 className="chart-title">
   515→              <Activity size={16} />
   516→              Flux patients en temps réel
   517→            </h3>
   518→            <span className="badge badge-green">{patientFlow.length} mouvements</span>
   519→          </div>
   520→          <div className="timeline-list">
   521→            {patientFlow.length === 0 ? (
   522→              <p className="muted" style={{ padding: "16px 0" }}>
   523→                Aucun mouvement enregistré.
   524→              </p>
   525→            ) : (
   526→              patientFlow.map((event) => <TimelineItem key={event.id} event={event} />)
   527→            )}
   528→          </div>
   529→        </div>
   530→
   531→        {/* Priority Alerts — 1/3 width */}
   532→        <div className="card dashboard-alerts-card">
   533→          <div className="chart-header">
   534→            <h3 className="chart-title">
   535→              <AlertCircle size={16} />
   536→              Alertes prioritaires
   537→            </h3>
   538→            <span className="badge badge-red">{priorityAlerts.length}</span>
   539→          </div>
   540→          <div className="alerts-list">
   541→            {priorityAlerts.length === 0 ? (
   542→              <p className="muted" style={{ padding: "16px 0" }}>
   543→                Aucune alerte active.
   544→              </p>
   545→            ) : (
   546→              priorityAlerts.map((alert) => <AlertItem key={alert.id} alert={alert} />)
   547→            )}
   548→          </div>
   549→        </div>
   550→      </div>
   551→
   552→      {/* Row 4 — Tables */}

   686→function TimelineItem({ event }: { event: PatientFlowEvent }) {
   687→  const config: Record<string, { color: string; label: string; icon: React.ReactNode }> = {
   688→    admission: {
   689→      color: "#0f6b3e",
   690→      label: "Admission",
   691→      icon: <UserPlus size={14} />,
   692→    },
   693→    discharge: {
   694→      color: "#2563eb",
   695→      label: "Sortie",
   696→      icon: <LogOut size={14} />,
   697→    },
   698→    transfer: {
   699→      color: "#d97706",
   700→      label: "Transfert",
   701→      icon: <ArrowRight size={14} />,
   702→    },
   703→    emergency: {
   704→      color: "#dc2626",
   705→      label: "Urgence",
   706→      icon: <AlertTriangle size={14} />,
   707→    },
   708→  };
   709→
   710→  const c = config[event.type] || config.admission;
   711→  const time = new Date(event.timestamp).toLocaleTimeString("fr-FR", {
   712→    hour: "2-digit",
   713→    minute: "2-digit",
   714→  });
   715→
   716→  return (
   717→    <div className="timeline-item">
   718→      <div className="timeline-dot" style={{ backgroundColor: c.color }}>
   719→        {c.icon}
   720→      </div>
   721→      <div className="timeline-content">
   722→        <div className="timeline-main">
   723→          <span className="timeline-label" style={{ color: c.color }}>
   724→            {c.label}
   725→          </span>
   726→          <span className="timeline-patient">{event.patientName}</span>
   727→        </div>
   728→        <div className="timeline-meta">
   729→          <span className="timeline-dept">{event.department}</span>
   730→          <span className="timeline-time">{time}</span>
   731→        </div>
   732→      </div>
   733→    </div>
   734→  );
   735→}
