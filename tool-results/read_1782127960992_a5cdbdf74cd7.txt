   144→
   145→function TableauTab({
   146→  lookups,
   147→  onCreated,
   148→}: {
   149→  lookups: LookupData;
   150→  onCreated: () => void;
   151→}) {
   152→  // Filtres serveur (appliqués via usePaginatedList.extraParams)
   153→  const [statusFilter, setStatusFilter] = useState("OPEN");
   154→  const [deptFilter, setDeptFilter] = useState("");
   155→  const [typeFilter, setTypeFilter] = useState("");
   156→  const [dateFrom, setDateFrom] = useState("");
   157→  const [dateTo, setDateTo] = useState("");
   158→
   159→  // Données KPI (chargées séparément, indépendantes des filtres/pagination)
   160→  const [kpiData, setKpiData] = useState<Row[]>([]);
   161→
   162→  const options = buildOptions(lookups);
   163→
   164→  // Liste paginée des admissions (recherche server-side + filtres serveur).
   165→  // NOTE: le filtre statut est passé via extraParams.status (et NON via search),
   166→  // ce qui corrige l'ancien bug où params.set("search", statusFilter) filtrait
   167→  // sur admission_type au lieu du statut réel.
   168→  const {
   169→    items: admissions,
   170→    total,
   171→    page,
   172→    totalPages,
   173→    loading,
   174→    error,
   175→    search,
   176→    setSearch,
   177→    setPage,
   178→    reload,
   179→  } = usePaginatedList<Row>("/admissions", {
   180→    pageSize: 20,
   181→    debounceMs: 300,
   182→    extraParams: {
   183→      status: statusFilter || null,
   184→      department_id: deptFilter || null,
   185→      admission_type: typeFilter || null,
   186→      date_from: dateFrom || null,
   187→      date_to: dateTo || null,
   188→    },
   189→  });
   190→
   191→  // Charger toutes les admissions pour les KPIs (comptages globaux,
   192→  // indépendants des filtres/pagination de la table)
   193→  async function loadKpis() {
   194→    try {
   195→      const payload = await apiRequest<any>("/admissions?page_size=1000");
   196→      setKpiData(Array.isArray(payload.data) ? payload.data : []);
   197→    } catch {
   198→      /* silent */
   199→    }
   200→  }
   201→
   202→  useEffect(() => {
   203→    loadKpis();
   204→  }, []);
   205→
   206→  // Réagir aux refresh globaux (création/sortie d'admission ailleurs)
   207→  useEffect(() => {
   208→    const handler = () => {
   209→      reload();
   210→      loadKpis();
   211→    };
   212→    window.addEventListener("refresh-resource", handler);
   213→    return () => window.removeEventListener("refresh-resource", handler);
   214→  }, [reload]);
   215→
   216→  function handleRefresh() {
   217→    reload();
   218→    loadKpis();
   219→    onCreated();
   220→  }
   221→
   222→  function handleResetFilters() {
   223→    setStatusFilter("OPEN");
   224→    setDeptFilter("");
   225→    setTypeFilter("");
   226→    setDateFrom("");
   227→    setDateTo("");
   228→    setSearch("");
   229→    setPage(1);
   230→  }
   231→
   232→  /* ── Resolve names ─────────────────────────────── */
   233→  function getPatientName(patientId: string): string {
   234→    const p = lookups.patients.find((x) => x.id === patientId);
   235→    if (!p) return "Inconnu";
   236→    return `${p.first_name || ""} ${p.last_name || ""}`.trim() || p.patient_number || "N/A";
   237→  }
   238→
   239→  function getDeptName(deptId: string | null): string {
   240→    if (!deptId) return "—";
   241→    const d = lookups.departments.find((x) => x.id === deptId);
   242→    return d ? `${d.code || ""} - ${d.name || d.id}`.trim() : "—";
   243→  }
   244→
   245→  function getFacilityName(facilityId: string): string {
   246→    const f = lookups.facilities.find((x) => x.id === facilityId);
   247→    return f ? f.name || f.id : "—";
   248→  }
   249→
   250→  /* ── Compute KPIs ──────────────────────────────── */
   251→  const now = new Date();
   252→  const todayStr = now.toDateString();
   253→
   254→  const kpis = useMemo(() => {
   255→    const active = kpiData.filter(
   256→      (a) => a.status === "OPEN" || a.status === "ACTIVE"
   257→    );
   258→    const todayAdmissions = kpiData.filter((a) => {
   259→      if (!a.admitted_at) return false;
   260→      return new Date(a.admitted_at).toDateString() === todayStr;
   261→    });
   262→    const hospitalized = kpiData.filter(
   263→      (a) =>
   264→        (a.status === "OPEN" || a.status === "ACTIVE") &&
   265→        a.admission_type === "HOSPITALIZATION"
   266→    );
   267→    const plannedDischarges = kpiData.filter(
   268→      (a) => a.status === "CLOSED" || a.status === "DISCHARGED"
   269→    );
   270→    return {
   271→      activeCount: active.length,
   272→      todayCount: todayAdmissions.length,
   273→      hospCount: hospitalized.length,
   274→      dischargeCount: plannedDischarges.length,
   275→    };
   276→  }, [kpiData, todayStr]);
   277→
   278→  /* ── Apply filters ── (filtrage server-side via usePaginatedList.extraParams) */
   279→
   280→  /* ── Color coding helpers ──────────────────────── */
   281→  function getStayIndicator(admission: Row): React.ReactNode {
   282→    if (!admission.admitted_at) return null;
   283→    const admittedDate = new Date(admission.admitted_at);
   284→    const diffHours = (now.getTime() - admittedDate.getTime()) / (1000 * 60 * 60);
   285→    if (diffHours < 24) {
   286→      return (
   287→        <span
   288→          style={{
   289→            display: "inline-block",
   290→            width: "8px",
   291→            height: "8px",
   292→            borderRadius: "50%",
   293→            background: "#16a34a",
   294→            marginRight: "6px",
   295→            flexShrink: 0,
   296→          }}
   297→          title="Admission récente (< 24h)"
   298→        />
   299→      );
   300→    }
   301→    const diffDays = diffHours / 24;
   302→    if (diffDays > 7) {
   303→      return (
   304→        <span
   305→          style={{
   306→            display: "inline-block",
   307→            width: "8px",
   308→            height: "8px",
   309→            borderRadius: "50%",
   310→            background: "#f59e0b",
   311→            marginRight: "6px",
   312→            flexShrink: 0,
   313→          }}
   314→          title="Séjour long (> 7 jours)"
   315→        />
   316→      );
   317→    }
   318→    return null;
   319→  }
   320→
   321→  async function handleDischarge(admissionId: string) {
   322→    if (!confirm("Confirmer la sortie de ce patient ?")) return;
   323→    try {
   324→      await apiRequest(`/admissions/${admissionId}/close`, { method: "POST" });
   325→      showToast("Sortie enregistrée avec succès.", "success");
   326→      handleRefresh();
   327→    } catch {
   328→      showToast("Erreur lors de l'enregistrement de la sortie.", "error");
   329→    }
   330→  }
   331→
   332→  return (
   333→    <>
   334→      {/* ── KPI Cards ──────────────────────────────────── */}
   335→      <div
   336→        style={{
   337→          display: "grid",
   338→          gridTemplateColumns: "repeat(auto-fit, minmax(200px, 1fr))",
   339→          gap: "12px",
   340→          marginBottom: "20px",
   341→        }}
   342→      >
   343→        <KpiCard
   344→          icon={<Activity size={20} />}
   345→          label="Admissions actives"
   346→          value={kpis.activeCount}
   347→          color="var(--primary)"
   348→          bgColor="var(--primary-light)"
   349→        />
   350→        <KpiCard
   351→          icon={<CalendarCheck size={20} />}
   352→          label="Prévues aujourd'hui"
   353→          value={kpis.todayCount}
   354→          color="#2563eb"
   355→          bgColor="#eff6ff"
   356→        />
   357→        <KpiCard
   358→          icon={<BedDouble size={20} />}
   359→          label="En hospitalisation"
   360→          value={kpis.hospCount}
   361→          color="#7c3aed"
   362→          bgColor="#f5f3ff"
   363→        />
   364→        <KpiCard
   365→          icon={<LogOut size={20} />}
   366→          label="Sorties prévues"
   367→          value={kpis.dischargeCount}
   368→          color="#b45309"
   369→          bgColor="#fef3c7"
   370→        />
   371→      </div>
   372→
   373→      {/* ── Barre de recherche (server-side, debounce 300ms) ── */}
   374→      <div
   375→        style={{
   376→          display: "flex",
   377→          gap: 12,
   378→          alignItems: "center",
   379→          marginBottom: 12,
   380→          flexWrap: "wrap",
   381→        }}
   382→      >
   383→        <input
   384→          type="text"
   385→          placeholder="🔍 Rechercher (patient, numéro, motif)..."
   386→          value={search}
   387→          onChange={(e) => setSearch(e.target.value)}
   388→          style={{ flex: 1, minWidth: 250, padding: "8px 12px" }}
   389→        />
   390→      </div>
   391→
   392→      {/* ── Filters ────────────────────────────────────── */}
   393→      <div
   394→        className="card"
   395→        style={{
   396→          marginBottom: "16px",
   397→          display: "flex",
   398→          gap: "12px",
   399→          flexWrap: "wrap",
   400→          alignItems: "end",
   401→          padding: "16px",
   402→        }}
   403→      >
   404→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   405→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Statut</span>
   406→          <select
   407→            value={statusFilter}
   408→            onChange={(e) => {
   409→              setStatusFilter(e.target.value);
   410→              setPage(1);
   411→            }}
   412→            style={{ minWidth: "160px", fontSize: "13px" }}
   413→          >
   414→            <option value="">Tous statuts</option>
   415→            <option value="OPEN">Actives</option>
   416→            <option value="CLOSED">Sortis</option>
   417→          </select>
   418→        </div>
   419→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   420→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Type</span>
   421→          <select
   422→            value={typeFilter}
   423→            onChange={(e) => {
   424→              setTypeFilter(e.target.value);
   425→              setPage(1);
   426→            }}
   427→            style={{ minWidth: "160px", fontSize: "13px" }}
   428→          >
   429→            <option value="">Tous types</option>
   430→            {ADMISSION_TYPE_OPTIONS.map((o) => (
   431→              <option key={o.value} value={o.value}>
   432→                {o.label}
   433→              </option>
   434→            ))}
   435→          </select>
   436→        </div>
   437→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   438→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Service</span>
   439→          <select
   440→            value={deptFilter}
   441→            onChange={(e) => {
   442→              setDeptFilter(e.target.value);
   443→              setPage(1);
   444→            }}
   445→            style={{ minWidth: "180px", fontSize: "13px" }}
   446→          >
   447→            <option value="">Tous les services</option>
   448→            {options.departments.map((o) => (
   449→              <option key={o.value} value={o.value}>
   450→                {o.label}
   451→              </option>
   452→            ))}
   453→          </select>
   454→        </div>
   455→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   456→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Date début</span>
   457→          <input
   458→            type="date"
   459→            value={dateFrom}
   460→            onChange={(e) => {
   461→              setDateFrom(e.target.value);
   462→              setPage(1);
   463→            }}
   464→            style={{ fontSize: "13px" }}
   465→          />
   466→        </div>
   467→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   468→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Date fin</span>
   469→          <input
   470→            type="date"
   471→            value={dateTo}
   472→            onChange={(e) => {
   473→              setDateTo(e.target.value);
   474→              setPage(1);
   475→            }}
   476→            style={{ fontSize: "13px" }}
   477→          />
   478→        </div>
   479→        <button
   480→          className="btn btn-outline btn-sm"
   481→          onClick={handleResetFilters}
   482→          style={{ height: "38px" }}
   483→        >
   484→          <X size={14} />
   485→          Réinitialiser
   486→        </button>
   487→        <button
   488→          className="btn btn-outline btn-sm"
   489→          onClick={handleRefresh}
   490→          style={{ height: "38px" }}
   491→        >
   492→          <RefreshCw size={14} />
   493→          Actualiser
   494→        </button>
   495→      </div>
   496→
   497→      {/* ── Admissions Table ───────────────────────────── */}
   498→      {loading ? (
   499→        <div className="card" style={{ textAlign: "center", padding: "32px" }}>
   500→          <div className="spinner" />
   501→          <p className="muted" style={{ marginTop: "12px" }}>
   502→            Chargement des admissions...
   503→          </p>
   504→        </div>
   505→      ) : error ? (
   506→        <div className="card" style={{ padding: "16px", color: "var(--danger)" }}>
   507→          {error}
   508→        </div>
   509→      ) : admissions.length === 0 ? (
   510→        <div className="card" style={{ textAlign: "center", padding: "32px" }}>
   511→          <p className="muted">Aucune admission trouvée.</p>
   512→        </div>
   513→      ) : (
   514→        <div className="card" style={{ padding: "0" }}>
   515→          <div className="table-wrapper" style={{ maxHeight: "520px", overflowY: "auto" }}>
   516→            <table className="data-table">
   517→              <thead>
   518→                <tr>
   519→                  <th>Patient</th>
   520→                  <th>Date d'admission</th>
   521→                  <th>Motif / Type</th>
   522→                  <th>Service</th>
   523→                  <th>Statut</th>
   524→                  <th>Établissement</th>
   525→                  <th>Actions</th>
   526→                </tr>
   527→              </thead>
   528→              <tbody>
   529→                {admissions.map((adm) => {
   530→                  const statusInfo = STATUS_MAP[adm.status] || {
   531→                    label: adm.status,
   532→                    badge: "badge-gray",
   533→                  };
   534→                  const isActive =
   535→                    adm.status === "OPEN" || adm.status === "ACTIVE";
   536→                  return (
   537→                    <tr key={adm.id}>
   538→                      <td>
   539→                        <div style={{ display: "flex", alignItems: "center" }}>
   540→                          {getStayIndicator(adm)}
   541→                          <span style={{ fontWeight: 600 }}>
   542→                            {getPatientName(adm.patient_id)}
   543→                          </span>
   544→                        </div>
   545→                      </td>
   546→                      <td style={{ whiteSpace: "nowrap", fontSize: "13px" }}>
   547→                        {adm.admitted_at
   548→                          ? new Date(adm.admitted_at).toLocaleString("fr-FR", {
   549→                              day: "2-digit",
   550→                              month: "2-digit",
   551→                              year: "numeric",
   552→                              hour: "2-digit",
   553→                              minute: "2-digit",
   554→                            })
   555→                          : "—"}
   556→                      </td>
   557→                      <td>
   558→                        <span style={{ fontWeight: 500 }}>
   559→                          {TYPE_LABELS[adm.admission_type] ||
   560→                            adm.admission_type ||
   561→                            "—"}
   562→                        </span>
   563→                      </td>
   564→                      <td style={{ fontSize: "13px" }}>
   565→                        {getDeptName(adm.department_id)}
   566→                      </td>
   567→                      <td>
   568→                        <span className={`badge ${statusInfo.badge}`}>
   569→                          {statusInfo.label}
   570→                        </span>
   571→                      </td>
   572→                      <td style={{ fontSize: "13px" }}>
   573→                        {getFacilityName(adm.facility_id)}
   574→                      </td>
   575→                      <td>
   576→                        <div style={{ display: "flex", gap: "6px" }}>
   577→                          <button
   578→                            className="btn btn-outline btn-sm"
   579→                            title="Voir dossier"
   580→                          >
   581→                            <Eye size={13} />
