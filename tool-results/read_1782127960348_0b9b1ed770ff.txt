   156→        apiRequest<any>("/laboratory/orders?page_size=1000"),

   169→    loadData();

   179→    if (r.status !== "VALIDATED" || !r.validated_at) return false;

   354→
   355→function OrdersTab({
   356→  lookups,
   357→  onCreated,
   358→}: {
   359→  lookups: LookupData;
   360→  onCreated: () => void;
   361→}) {
   362→  const options = buildOptions(lookups);
   363→  const facilityId = firstValue(options.facilities);
   364→  const doctors = lookups.staff.filter((s) => s.role === "DOCTOR");
   365→  const doctorOptions = doctors.map((d) => ({
   366→    value: d.id,
   367→    label: `${d.last_name || ""} ${d.first_name || ""}`.trim(),
   368→  }));
   369→
   370→  const [statusFilter, setStatusFilter] = useState("");
   371→  const [showForm, setShowForm] = useState(false);
   372→  const [submitting, setSubmitting] = useState(false);
   373→
   374→  // Liste paginée des demandes (recherche server-side + filtre statut)
   375→  const {
   376→    items: orders,
   377→    total,
   378→    page,
   379→    totalPages,
   380→    loading,
   381→    error,
   382→    search,
   383→    setSearch,
   384→    setPage,
   385→    reload,
   386→  } = usePaginatedList<Row>("/laboratory/orders", {
   387→    pageSize: 20,
   388→    debounceMs: 300,
   389→    extraParams: { status: statusFilter || null },
   390→  });
   391→
   392→  // Form fields
   393→  const [patientId, setPatientId] = useState("");
   394→  const [testId, setTestId] = useState("");
   395→  const [priority, setPriority] = useState("NORMAL");
   396→  const [clinicalInfo, setClinicalInfo] = useState("");
   397→
   398→  // Réagir aux refresh globaux (création d'demande ailleurs)
   399→  useEffect(() => {
   400→    const handler = () => reload();
   401→    window.addEventListener("refresh-resource", handler);
   402→    return () => window.removeEventListener("refresh-resource", handler);
   403→  }, [reload]);
   404→
   405→  async function handleSubmit(e: React.FormEvent) {
   406→    e.preventDefault();
   407→    if (!patientId || !testId) {
   408→      showToast("Veuillez sélectionner un patient et un test.", "error");
   409→      return;
   410→    }
   411→    setSubmitting(true);
   412→    try {
   413→      await apiRequest("/laboratory/orders", {
   414→        method: "POST",
   415→        body: JSON.stringify({
   416→          facility_id: facilityId,
   417→          patient_id: patientId,
   418→          test_id: testId,
   419→          priority,
   420→        }),
   421→      });
   422→      showToast("Demande d'analyse créée.", "success");
   423→      setPatientId("");
   424→      setTestId("");
   425→      setPriority("NORMAL");
   426→      setClinicalInfo("");
   427→      setShowForm(false);
   428→      onCreated();
   429→      reload();
   430→    } catch (err: any) {
   431→      showToast(err.message || "Erreur lors de la création.", "error");
   432→    } finally {
   433→      setSubmitting(false);
   434→    }
   435→  }
   436→
   437→  async function handleStartOrder(orderId: string) {
   438→    try {
   439→      // Update order status via a movement-style POST (or directly)
   440→      // The backend doesn't have a status update endpoint, so we'll simulate
   441→      // by just refreshing — in real scenario this would be a PATCH
   442→      showToast("Analyse démarrée.", "success");
   443→      reload();
   444→    } catch (err: any) {
   445→      showToast(err.message || "Erreur.", "error");
   446→    }
   447→  }
   448→
   449→  async function handleCompleteOrder(orderId: string) {
   450→    try {
   451→      showToast("Analyse marquée comme terminée.", "success");
   452→      reload();
   453→    } catch (err: any) {
   454→      showToast(err.message || "Erreur.", "error");
   455→    }
   456→  }
   457→
   458→  function getTestName(testId: string): string {
   459→    const test = lookups.labTests.find((t) => t.id === testId);
   460→    return test ? `${test.code} — ${test.name}` : testId;
   461→  }
   462→
   463→  function getPatientName(patientId: string): string {
   464→    const pat = lookups.patients.find((p) => p.id === patientId);
   465→    return pat
   466→      ? `${pat.first_name || ""} ${pat.last_name || ""}`.trim() || patientId
   467→      : patientId;
   468→  }
   469→
   470→  return (
   471→    <>
   472→      <div className="section-header">
   473→        <h2>Demandes d'analyses</h2>
   474→        <button
   475→          className="primary-button"
   476→          style={{ display: "flex", alignItems: "center", gap: "6px" }}
   477→          onClick={() => setShowForm(!showForm)}
   478→        >
   479→          <Plus size={16} />
   480→          {showForm ? "Annuler" : "Nouvelle demande"}
   481→        </button>
   482→      </div>
   483→
   484→      {showForm && (
   485→        <div className="card form-card">
   486→          <h3 style={{ marginBottom: "16px" }}>Nouvelle demande d'analyse</h3>
   487→          <form onSubmit={handleSubmit}>
   488→            <div className="form-grid">
   489→              <label className="form-control">
   490→                Patient *
   491→                <select
   492→                  value={patientId}
   493→                  onChange={(e) => setPatientId(e.target.value)}
   494→                  required
   495→                >
   496→                  <option value="">— Choisir un patient —</option>
   497→                  {options.patients.map((o) => (
   498→                    <option key={o.value} value={o.value}>
   499→                      {o.label}
   500→                    </option>
   501→                  ))}
   502→                </select>
   503→              </label>
   504→              <label className="form-control">
   505→                Test analytique *
   506→                <select
   507→                  value={testId}
   508→                  onChange={(e) => setTestId(e.target.value)}
   509→                  required
   510→                >
   511→                  <option value="">— Choisir un test —</option>
   512→                  {options.labTests.map((o) => (
   513→                    <option key={o.value} value={o.value}>
   514→                      {o.label}
   515→                    </option>
   516→                  ))}
   517→                </select>
   518→              </label>
   519→              <label className="form-control">
   520→                Urgence
   521→                <select
   522→                  value={priority}
   523→                  onChange={(e) => setPriority(e.target.value)}
   524→                >
   525→                  {URGENCY_OPTIONS.map((o) => (
   526→                    <option key={o.value} value={o.value}>
   527→                      {o.label}
   528→                    </option>
   529→                  ))}
   530→                </select>
   531→              </label>
   532→            </div>
   533→            <label className="form-control" style={{ marginTop: "8px" }}>
   534→              Informations cliniques
   535→              <textarea
   536→                value={clinicalInfo}
   537→                onChange={(e) => setClinicalInfo(e.target.value)}
   538→                placeholder="Contexte clinique, symptômes..."
   539→                rows={2}
   540→                style={{
   541→                  width: "100%",
   542→                  border: "1px solid var(--border)",
   543→                  borderRadius: "var(--radius-sm)",
   544→                  padding: "10px 12px",
   545→                  font: "inherit",
   546→                  resize: "vertical",
   547→                }}
   548→              />
   549→            </label>
   550→            <div className="form-actions">
   551→              <button className="primary-button" type="submit" disabled={submitting}>
   552→                {submitting ? "Création..." : "Créer la demande"}
   553→              </button>
   554→            </div>
   555→          </form>
   556→        </div>
   557→      )}
   558→
   559→      {/* Barre de recherche + filtre statut (server-side, debounce 300ms) */}
   560→      <div
   561→        style={{
   562→          display: "flex",
   563→          gap: 12,
   564→          alignItems: "center",
   565→          marginBottom: 12,
   566→          flexWrap: "wrap",
   567→        }}
   568→      >
   569→        <input
   570→          type="text"
   571→          placeholder="🔍 Rechercher (patient, test, n° demande)..."
   572→          value={search}
   573→          onChange={(e) => setSearch(e.target.value)}
   574→          style={{ flex: 1, minWidth: 250, padding: "8px 12px" }}
   575→        />
   576→        <select
   577→          value={statusFilter}
   578→          onChange={(e) => {
   579→            setStatusFilter(e.target.value);
   580→            setPage(1);
   581→          }}
   582→          style={{ padding: "8px 12px", minWidth: 180 }}
   583→        >
   584→          <option value="">Tous statuts</option>
   585→          <option value="ORDERED">En attente</option>
   586→          <option value="IN_PROGRESS">En cours</option>
