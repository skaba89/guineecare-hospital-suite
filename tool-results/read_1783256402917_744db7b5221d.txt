   107→/* ─── Dashboard Tab ──────────────────────────────────────────── */
   108→
   109→function DashboardTab({ lookups }: { lookups: LookupData }) {
   110→  const [dashboard, setDashboard] = useState<Row | null>(null);
   111→  const [loading, setLoading] = useState(true);
   112→  const [error, setError] = useState("");
   113→
   114→  const loadDashboard = useCallback(async () => {
   115→    setLoading(true);
   116→    setError("");
   117→    try {
   118→      const payload = await apiRequest<any>("/reporting/dashboard");
   119→      setDashboard(payload.data || null);
   120→    } catch {
   121→      setError("Impossible de charger le tableau de bord.");
   122→    } finally {
   123→      setLoading(false);
   124→    }
   125→  }, []);
   126→
   127→  useEffect(() => {
   128→    loadDashboard();
   129→    const handler = () => loadDashboard();
   130→    window.addEventListener("refresh-resource", handler);
   131→    return () => window.removeEventListener("refresh-resource", handler);
   132→  }, [loadDashboard]);
   133→
   134→  if (loading) {
   135→    return (
   136→      <div className="card" style={{ textAlign: "center", padding: "32px" }}>
   137→        <div className="spinner" />
   138→        <p className="muted" style={{ marginTop: "12px" }}>Chargement du tableau de bord...</p>
   139→      </div>
   140→    );
   141→  }
   142→
   143→  if (error) {
   144→    return <p style={{ color: "crimson" }}>{error}</p>;
   145→  }
   146→
   147→  const cards = [
   148→    { label: "Rapports en brouillon", value: dashboard?.draft_reports ?? 0, color: "#6b7280" },
   149→    { label: "Rapports soumis", value: dashboard?.submitted_reports ?? 0, color: "#2563eb" },
   150→    { label: "Rapports validés", value: dashboard?.validated_reports ?? 0, color: "#047857" },
   151→    { label: "Alertes actives", value: dashboard?.active_alerts ?? 0, color: "#b91c1c" },
   152→  ];
   153→
   154→  return (
   155→    <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(220px, 1fr))", gap: "16px" }}>
   156→      {cards.map((card) => (
   157→        <div className="card" key={card.label} style={{ textAlign: "center" }}>
   158→          <div style={{ fontSize: "32px", fontWeight: 700, color: card.color }}>{card.value}</div>
   159→          <div className="muted" style={{ marginTop: "8px", fontSize: "14px" }}>{card.label}</div>
   160→        </div>
   161→      ))}
   162→    </div>
   163→  );
   164→}
