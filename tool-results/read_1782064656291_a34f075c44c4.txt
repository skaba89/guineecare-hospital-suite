   142→
   143→function TableauTab({
   144→  lookups,
   145→  onCreated,
   146→}: {
   147→  lookups: LookupData;
   148→  onCreated: () => void;
   149→}) {
   150→  const [admissions, setAdmissions] = useState<Row[]>([]);
   151→  const [loading, setLoading] = useState(true);
   152→  const [statusFilter, setStatusFilter] = useState("OPEN");
   153→  const [deptFilter, setDeptFilter] = useState("");
   154→  const [dateFrom, setDateFrom] = useState("");
   155→  const [dateTo, setDateTo] = useState("");
   156→  const [refreshKey, setRefreshKey] = useState(0);
   157→
   158→  const options = buildOptions(lookups);
   159→
   160→  const loadAdmissions = useCallback(async () => {
   161→    setLoading(true);
   162→    try {
   163→      const params = new URLSearchParams();
   164→      if (statusFilter) params.set("search", statusFilter);
   165→      const qs = params.toString();
   166→      const payload = await apiRequest<any>(`/admissions${qs ? `?${qs}&page_size=1000` : "?page_size=1000"}`);
   167→      const data: Row[] = Array.isArray(payload.data) ? payload.data : [];
   168→      setAdmissions(data);
   169→    } catch {
   170→      /* silent */
   171→    } finally {
   172→      setLoading(false);
   173→    }
   174→  }, [statusFilter]);
   175→
   176→  useEffect(() => {
   177→    loadAdmissions();
   178→  }, [loadAdmissions, refreshKey]);
   179→
   180→  useEffect(() => {
   181→    const handler = () => setRefreshKey((k) => k + 1);
   182→    window.addEventListener("refresh-resource", handler);
   183→    return () => window.removeEventListener("refresh-resource", handler);
   184→  }, []);
   185→
   186→  function handleRefresh() {
   187→    setRefreshKey((k) => k + 1);
   188→    onCreated();
   189→  }
   190→
   191→  /* ── Resolve names ─────────────────────────────── */
   192→  function getPatientName(patientId: string): string {
   193→    const p = lookups.patients.find((x) => x.id === patientId);
   194→    if (!p) return "Inconnu";
   195→    return `${p.first_name || ""} ${p.last_name || ""}`.trim() || p.patient_number || "N/A";
   196→  }
   197→
   198→  function getDeptName(deptId: string | null): string {
   199→    if (!deptId) return "—";
   200→    const d = lookups.departments.find((x) => x.id === deptId);
   201→    return d ? `${d.code || ""} - ${d.name || d.id}`.trim() : "—";
   202→  }
   203→
   204→  function getFacilityName(facilityId: string): string {
   205→    const f = lookups.facilities.find((x) => x.id === facilityId);
   206→    return f ? f.name || f.id : "—";
   207→  }
   208→
   209→  /* ── Compute KPIs ──────────────────────────────── */
   210→  const now = new Date();
   211→  const todayStr = now.toDateString();
   212→
   213→  const kpis = useMemo(() => {
   214→    const active = admissions.filter(
   215→      (a) => a.status === "OPEN" || a.status === "ACTIVE"
   216→    );
   217→    const todayAdmissions = admissions.filter((a) => {
   218→      if (!a.admitted_at) return false;
   219→      return new Date(a.admitted_at).toDateString() === todayStr;
   220→    });
   221→    const hospitalized = admissions.filter(
   222→      (a) =>
   223→        (a.status === "OPEN" || a.status === "ACTIVE") &&
   224→        a.admission_type === "HOSPITALIZATION"
   225→    );
   226→    const plannedDischarges = admissions.filter(
   227→      (a) => a.status === "CLOSED" || a.status === "DISCHARGED"
   228→    );
   229→    return {
   230→      activeCount: active.length,
   231→      todayCount: todayAdmissions.length,
   232→      hospCount: hospitalized.length,
   233→      dischargeCount: plannedDischarges.length,
   234→    };
   235→  }, [admissions, todayStr]);
   236→
   237→  /* ── Apply filters ─────────────────────────────── */
   238→  const filtered = useMemo(() => {
   239→    let rows = admissions.filter(
   240→      (a) => a.status === "OPEN" || a.status === "ACTIVE"
   241→    );
   242→    if (statusFilter === "ALL") {
   243→      rows = admissions;
   244→    } else if (statusFilter === "CLOSED") {
   245→      rows = admissions.filter(
   246→        (a) => a.status === "CLOSED" || a.status === "DISCHARGED"
   247→      );
   248→    }
   249→    if (deptFilter) {
   250→      rows = rows.filter((a) => a.department_id === deptFilter);
   251→    }
   252→    if (dateFrom) {
   253→      rows = rows.filter(
   254→        (a) => a.admitted_at && new Date(a.admitted_at) >= new Date(dateFrom)
   255→      );
   256→    }
   257→    if (dateTo) {
   258→      const to = new Date(dateTo);
   259→      to.setHours(23, 59, 59, 999);
   260→      rows = rows.filter(
   261→        (a) => a.admitted_at && new Date(a.admitted_at) <= to
   262→      );
   263→    }
   264→    return rows;
   265→  }, [admissions, statusFilter, deptFilter, dateFrom, dateTo]);
   266→
   267→  /* ── Color coding helpers ──────────────────────── */
   268→  function getStayIndicator(admission: Row): React.ReactNode {
   269→    if (!admission.admitted_at) return null;
   270→    const admittedDate = new Date(admission.admitted_at);
   271→    const diffHours = (now.getTime() - admittedDate.getTime()) / (1000 * 60 * 60);
   272→    if (diffHours < 24) {
   273→      return (
   274→        <span
   275→          style={{
   276→            display: "inline-block",
   277→            width: "8px",
   278→            height: "8px",
   279→            borderRadius: "50%",
   280→            background: "#16a34a",
   281→            marginRight: "6px",
   282→            flexShrink: 0,
   283→          }}
   284→          title="Admission récente (< 24h)"
   285→        />
   286→      );
   287→    }
   288→    const diffDays = diffHours / 24;
   289→    if (diffDays > 7) {
   290→      return (
   291→        <span
   292→          style={{
   293→            display: "inline-block",
   294→            width: "8px",
   295→            height: "8px",
   296→            borderRadius: "50%",
   297→            background: "#f59e0b",
   298→            marginRight: "6px",
   299→            flexShrink: 0,
   300→          }}
   301→          title="Séjour long (> 7 jours)"
   302→        />
   303→      );
   304→    }
   305→    return null;
   306→  }
   307→
   308→  async function handleDischarge(admissionId: string) {
   309→    if (!confirm("Confirmer la sortie de ce patient ?")) return;
   310→    try {
   311→      await apiRequest(`/admissions/${admissionId}/close`, { method: "POST" });
   312→      showToast("Sortie enregistrée avec succès.", "success");
   313→      handleRefresh();
   314→    } catch {
   315→      showToast("Erreur lors de l'enregistrement de la sortie.", "error");
   316→    }
   317→  }
   318→
   319→  return (
   320→    <>
   321→      {/* ── KPI Cards ──────────────────────────────────── */}
   322→      <div
   323→        style={{
   324→          display: "grid",
   325→          gridTemplateColumns: "repeat(auto-fit, minmax(200px, 1fr))",
   326→          gap: "12px",
   327→          marginBottom: "20px",
   328→        }}
   329→      >
   330→        <KpiCard
   331→          icon={<Activity size={20} />}
   332→          label="Admissions actives"
   333→          value={kpis.activeCount}
   334→          color="var(--primary)"
   335→          bgColor="var(--primary-light)"
   336→        />
   337→        <KpiCard
   338→          icon={<CalendarCheck size={20} />}
   339→          label="Prévues aujourd'hui"
   340→          value={kpis.todayCount}
   341→          color="#2563eb"
   342→          bgColor="#eff6ff"
   343→        />
   344→        <KpiCard
   345→          icon={<BedDouble size={20} />}
   346→          label="En hospitalisation"
   347→          value={kpis.hospCount}
   348→          color="#7c3aed"
   349→          bgColor="#f5f3ff"
   350→        />
   351→        <KpiCard
   352→          icon={<LogOut size={20} />}
   353→          label="Sorties prévues"
   354→          value={kpis.dischargeCount}
   355→          color="#b45309"
   356→          bgColor="#fef3c7"
   357→        />
   358→      </div>
   359→
   360→      {/* ── Filters ────────────────────────────────────── */}
   361→      <div
   362→        className="card"
   363→        style={{
   364→          marginBottom: "16px",
   365→          display: "flex",
   366→          gap: "12px",
   367→          flexWrap: "wrap",
   368→          alignItems: "end",
   369→          padding: "16px",
   370→        }}
   371→      >
   372→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   373→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Statut</span>
   374→          <select
   375→            value={statusFilter}
   376→            onChange={(e) => setStatusFilter(e.target.value)}
   377→            style={{ minWidth: "160px", fontSize: "13px" }}
   378→          >
   379→            <option value="OPEN">Actives</option>
   380→            <option value="CLOSED">Sortis</option>
   381→            <option value="ALL">Tous</option>
   382→          </select>
   383→        </div>
   384→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   385→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Service</span>
   386→          <select
   387→            value={deptFilter}
   388→            onChange={(e) => setDeptFilter(e.target.value)}
   389→            style={{ minWidth: "180px", fontSize: "13px" }}
   390→          >
   391→            <option value="">Tous les services</option>
   392→            {options.departments.map((o) => (
   393→              <option key={o.value} value={o.value}>
   394→                {o.label}
   395→              </option>
   396→            ))}
   397→          </select>
   398→        </div>
   399→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   400→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Date début</span>
   401→          <input
   402→            type="date"
   403→            value={dateFrom}
   404→            onChange={(e) => setDateFrom(e.target.value)}
   405→            style={{ fontSize: "13px" }}
   406→          />
   407→        </div>
   408→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   409→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Date fin</span>
   410→          <input
   411→            type="date"
   412→            value={dateTo}
   413→            onChange={(e) => setDateTo(e.target.value)}
   414→            style={{ fontSize: "13px" }}
   415→          />
   416→        </div>
   417→        <button
   418→          className="btn btn-outline btn-sm"
   419→          onClick={() => {
   420→            setStatusFilter("OPEN");
   421→            setDeptFilter("");
   422→            setDateFrom("");
   423→            setDateTo("");
   424→          }}
   425→          style={{ height: "38px" }}
   426→        >
   427→          <X size={14} />
   428→          Réinitialiser
   429→        </button>
   430→        <button
   431→          className="btn btn-outline btn-sm"
   432→          onClick={handleRefresh}
   433→          style={{ height: "38px" }}
   434→        >
   435→          <RefreshCw size={14} />
   436→          Actualiser
   437→        </button>
   438→      </div>
   439→
   440→      {/* ── Admissions Table ───────────────────────────── */}
   441→      {loading ? (
   442→        <div className="card" style={{ textAlign: "center", padding: "32px" }}>
   443→          <div className="spinner" />
   444→          <p className="muted" style={{ marginTop: "12px" }}>
   445→            Chargement des admissions...
   446→          </p>
   447→        </div>
   448→      ) : filtered.length === 0 ? (
   449→        <div className="card" style={{ textAlign: "center", padding: "32px" }}>
   450→          <p className="muted">Aucune admission trouvée.</p>
   451→        </div>
   452→      ) : (
   453→        <div className="card" style={{ padding: "0" }}>
   454→          <div className="table-wrapper" style={{ maxHeight: "520px", overflowY: "auto" }}>
   455→            <table className="data-table">
   456→              <thead>
   457→                <tr>
   458→                  <th>Patient</th>
   459→                  <th>Date d'admission</th>
   460→                  <th>Motif / Type</th>
   461→                  <th>Service</th>
   462→                  <th>Statut</th>
   463→                  <th>Établissement</th>
   464→                  <th>Actions</th>
   465→                </tr>
   466→              </thead>
   467→              <tbody>
   468→                {filtered.map((adm) => {
   469→                  const statusInfo = STATUS_MAP[adm.status] || {
   470→                    label: adm.status,
   471→                    badge: "badge-gray",
   472→                  };
   473→                  const isActive =
   474→                    adm.status === "OPEN" || adm.status === "ACTIVE";
   475→                  return (
   476→                    <tr key={adm.id}>
   477→                      <td>
   478→                        <div style={{ display: "flex", alignItems: "center" }}>
   479→                          {getStayIndicator(adm)}
   480→                          <span style={{ fontWeight: 600 }}>
   481→                            {getPatientName(adm.patient_id)}
   482→                          </span>
   483→                        </div>
   484→                      </td>
   485→                      <td style={{ whiteSpace: "nowrap", fontSize: "13px" }}>
   486→                        {adm.admitted_at
   487→                          ? new Date(adm.admitted_at).toLocaleString("fr-FR", {
   488→                              day: "2-digit",
   489→                              month: "2-digit",
   490→                              year: "numeric",
   491→                              hour: "2-digit",
   492→                              minute: "2-digit",
   493→                            })
   494→                          : "—"}
   495→                      </td>
   496→                      <td>
   497→                        <span style={{ fontWeight: 500 }}>
   498→                          {TYPE_LABELS[adm.admission_type] ||
   499→                            adm.admission_type ||
   500→                            "—"}
   501→                        </span>
   502→                      </td>
   503→                      <td style={{ fontSize: "13px" }}>
   504→                        {getDeptName(adm.department_id)}
   505→                      </td>
   506→                      <td>
   507→                        <span className={`badge ${statusInfo.badge}`}>
   508→                          {statusInfo.label}
   509→                        </span>
   510→                      </td>
   511→                      <td style={{ fontSize: "13px" }}>
   512→                        {getFacilityName(adm.facility_id)}
   513→                      </td>
   514→                      <td>
   515→                        <div style={{ display: "flex", gap: "6px" }}>
   516→                          <button
   517→                            className="btn btn-outline btn-sm"
   518→                            title="Voir dossier"
   519→                          >
   520→                            <Eye size={13} />
   521→                            Dossier
   522→                          </button>
   523→                          {isActive && (
   524→                            <button
   525→                              className="btn btn-sm"
   526→                              style={{
   527→                                background: "#fef3c7",
   528→                                color: "#92400e",
   529→                                border: "1px solid #fcd34d",
   530→                              }}
   531→                              onClick={() => handleDischarge(adm.id)}
   532→                              title="Programmer sortie"
   533→                            >
   534→                              <ClipboardCheck size={13} />
   535→                              Sortie
   536→                            </button>
   537→                          )}
   538→                        </div>
   539→                      </td>
   540→                    </tr>
   541→                  );
   542→                })}
   543→              </tbody>
   544→            </table>
   545→          </div>
   546→          <div
   547→            style={{
   548→              padding: "12px 16px",
   549→              borderTop: "1px solid var(--border-light)",
   550→              fontSize: "13px",
   551→              color: "var(--muted)",
   552→              fontWeight: 600,
   553→            }}
   554→          >
   555→            {filtered.length} admission{filtered.length > 1 ? "s" : ""} affichée
   556→            {filtered.length > 1 ? "s" : ""}
   557→          </div>
   558→        </div>
   559→      )}
   560→    </>
   561→  );
   562→}

   564→/* ═══════════════════════════════════════════════════════════════════

   567→
   568→function NouvelleAdmissionTab({
   569→  lookups,
   570→  onCreated,
   571→}: {
   572→  lookups: LookupData;
   573→  onCreated: () => void;
   574→}) {
   575→  const options = buildOptions(lookups);
   576→  const doctors = lookups.staff.filter(
   577→    (s) => s.role === "DOCTOR" || s.role === "PHYSICIAN"
   578→  );
   579→
   580→  const [facilityId, setFacilityId] = useState(firstValue(options.facilities));
