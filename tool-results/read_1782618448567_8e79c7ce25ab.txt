   423→interface InvoiceLine {
   424→  id: string;
   425→  tariff_item_id: string;
   426→  quantity: number;
   427→  unit_price: number;
   428→}
   429→
   430→function InvoicesTab({
   431→  lookups,
   432→  onCreated,
   433→}: {
   434→  lookups: LookupData;
   435→  onCreated: () => void;
   436→}) {
   437→  const t = useT();
   438→  const options = buildOptions(lookups);
   439→  const facilityId = firstValue(options.facilities);
   440→
   441→  // Tarifs chargés en plein (petite table de référence pour le dropdown du formulaire)
   442→  const [tariffs, setTariffs] = useState<Row[]>([]);
   443→  const [showForm, setShowForm] = useState(false);
   444→  const [submitting, setSubmitting] = useState(false);
   445→
   446→  // Liste paginée des factures (recherche server-side + filtre statut)
   447→  const [statusFilter, setStatusFilter] = useState<string>("");
   448→  const {
   449→    items: invoices,
   450→    total,
   451→    page,
   452→    totalPages,
   453→    loading,
   454→    error,
   455→    search,
   456→    setSearch,
   457→    setPage,
   458→    reload,
   459→  } = usePaginatedList<Row>("/billing/invoices", {
   460→    pageSize: 20,
   461→    debounceMs: 300,
   462→    extraParams: { status: statusFilter || null },
   463→  });
   464→
   465→  // Charger les tarifs (petite table de référence) une seule fois
   466→  useEffect(() => {
   467→    async function loadTariffs() {
   468→      try {
   469→        const tariffsRes = await apiRequest<any>("/billing/tariffs?page_size=1000");
   470→        setTariffs(Array.isArray(tariffsRes.data) ? tariffsRes.data : []);
   471→      } catch {
   472→        // silent — le form ne sera pas bloqué
   473→      }
   474→    }
   475→    loadTariffs();
   476→  }, []);
   477→
   478→  // Réagir aux refresh globaux (création d'facture ailleurs)
   479→  useEffect(() => {
   480→    const handler = () => reload();
   481→    window.addEventListener("refresh-resource", handler);
   482→    return () => window.removeEventListener("refresh-resource", handler);
   483→  }, [reload]);
   484→
   485→  // Form fields
   486→  const [patientId, setPatientId] = useState("");
   487→  const [invoiceLines, setInvoiceLines] = useState<InvoiceLine[]>([]);
   488→  const [invoiceNumber, setInvoiceNumber] = useState("");
   489→  const [description, setDescription] = useState("");
   490→
   491→  function addLine() {
   492→    setInvoiceLines([
   493→      ...invoiceLines,
   494→      { id: `line-${Date.now()}`, tariff_item_id: "", quantity: 1, unit_price: 0 },
   495→    ]);
   496→  }
   497→
   498→  function removeLine(lineId: string) {
   499→    setInvoiceLines(invoiceLines.filter((l) => l.id !== lineId));
   500→  }
   501→
   502→  function updateLine(
   503→    lineId: string,
   504→    field: "tariff_item_id" | "quantity",
   505→    value: string | number
   506→  ) {
   507→    setInvoiceLines(
   508→      invoiceLines.map((l) => {
   509→        if (l.id !== lineId) return l;
   510→        if (field === "tariff_item_id") {
   511→          const tariff = tariffs.find((t) => t.id === value);
   512→          return {
   513→            ...l,
   514→            tariff_item_id: value as string,
   515→            unit_price: tariff?.unit_price || 0,
   516→          };
   517→        }
   518→        return { ...l, quantity: Number(value) || 1 };
   519→      })
   520→    );
   521→  }
   522→
   523→  const totalAmount = invoiceLines.reduce(
   524→    (sum, l) => sum + l.unit_price * l.quantity,
   525→    0
   526→  );
   527→
   528→  function generateInvoiceNumber(): string {
   529→    const now = new Date();
   530→    const y = now.getFullYear();
   531→    const m = String(now.getMonth() + 1).padStart(2, "0");
   532→    const seq = String(total + 1).padStart(4, "0");
   533→    return `FAC-${y}${m}-${seq}`;
   534→  }
   535→
   536→  async function handleSubmit(e: React.FormEvent) {
   537→    e.preventDefault();
   538→    if (!patientId || invoiceLines.length === 0) {
   539→      showToast("Veuillez sélectionner un patient et ajouter au moins une ligne.", "error");
   540→      return;
   541→    }
   542→    const invalidLines = invoiceLines.filter(
   543→      (l) => !l.tariff_item_id || l.quantity <= 0
   544→    );
   545→    if (invalidLines.length > 0) {
   546→      showToast("Veuillez remplir toutes les lignes de facturation.", "error");
   547→      return;
   548→    }
   549→
   550→    setSubmitting(true);
   551→    try {
   552→      const invNum = invoiceNumber || generateInvoiceNumber();
   553→      await apiRequest("/billing/invoices", {
   554→        method: "POST",
   555→        body: JSON.stringify({
   556→          facility_id: facilityId,
   557→          patient_id: patientId,
   558→          invoice_number: invNum,
   559→          description: description || undefined,
   560→          net_amount: totalAmount,
   561→        }),
   562→      });
   563→      showToast("Facture créée avec succès.", "success");
   564→      setPatientId("");
   565→      setInvoiceLines([]);
   566→      setInvoiceNumber("");
   567→      setDescription("");
   568→      setShowForm(false);
   569→      onCreated();
   570→      reload();
   571→    } catch (err: any) {
   572→      showToast(err.message || "Erreur lors de la création.", "error");
   573→    } finally {
   574→      setSubmitting(false);
   575→    }
   576→  }
   577→
   578→  const tariffOptions = tariffs.map((t) => ({
   579→    value: t.id,
