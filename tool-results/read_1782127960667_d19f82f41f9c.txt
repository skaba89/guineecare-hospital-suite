   261→            <div className="kpi-card-value">
   262→              {estimatedValue.toLocaleString("fr-FR")} GNF
   263→            </div>

   266→        </div>

   310→        </div>
   311→      ) : error ? (

   316→        <div className="card" style={{ padding: 0, overflow: "hidden" }}>
   317→          <div className="table-wrapper">
   318→            <table className="table">
   319→              <thead>
   320→                <tr>
   321→                  <th>Produit</th>
   322→                  <th>Catégorie</th>
   323→                  <th>Forme</th>

   337→                  </tr>

   339→                  filtered.map((item: Row) => {
   340→                    const status = getStockStatus(item);
   341→                    return (
   342→                      <tr
   343→                        key={item.id}
   344→                        style={{
   345→                          background:
   346→                            status === "epuise"
   347→                              ? "var(--danger-light)"
   348→                              : status === "bas"
   349→                              ? "var(--warning-light)"
   350→                              : "inherit",
   351→                        }}
   352→                      >

   357→                              className="muted"
   358→                              style={{ fontWeight: 400, marginLeft: "8px", fontSize: "12px" }}

   387→                              : "Épuisé"}

   455→/* ═════════════════════════════════════════════════════════════════
   456→   Dispensation Tab
   457→   ═════════════════════════════════════════════════════════════════ */
   458→
   459→interface DispensationLine {
   460→  id: string;
   461→  product_id: string;
   462→  quantity: number;
   463→}
   464→
   465→function DispensationTab({
   466→  lookups,
   467→  onCreated,
   468→}: {
   469→  lookups: LookupData;
   470→  onCreated: () => void;
   471→}) {
   472→  const options = buildOptions(lookups);
   473→  const facilityId = firstValue(options.facilities);
   474→  const doctors = lookups.staff.filter((s) => s.role === "DOCTOR");
   475→  const doctorOptions = doctors.map((d) => ({
   476→    value: d.id,
   477→    label: `${d.last_name || ""} ${d.first_name || ""} — ${d.role || ""}`.trim(),
   478→  }));
   479→
   480→  const [patientId, setPatientId] = useState("");
   481→  const [prescription, setPrescription] = useState("");
   482→  const [prescriberId, setPrescriberId] = useState("");
   483→  const [lines, setLines] = useState<DispensationLine[]>([]);
   484→  const [submitting, setSubmitting] = useState(false);
   485→
   486→  const [stock, setStock] = useState<Row[]>([]);
   487→  const [recentDispensations, setRecentDispensations] = useState<Row[]>([]);
   488→
   489→  const loadStock = useCallback(async () => {
   490→    try {
   491→      const res = await apiRequest<any>("/pharmacy/stock?page_size=1000");
   492→      setStock(Array.isArray(res.data) ? res.data : []);
   493→    } catch {
   494→      /* silent */
   495→    }
   496→  }, []);
   497→
   498→  const loadMovements = useCallback(async () => {
   499→    try {
   500→      const res = await apiRequest<any>("/pharmacy/stock/movements?page_size=1000");
   501→      const movements = Array.isArray(res.data) ? res.data : [];
   502→      setRecentDispensations(movements.filter((m: Row) => m.movement_type === "OUT"));
   503→    } catch {
   504→      /* silent */
   505→    }
   506→  }, []);
   507→
   508→  useEffect(() => {
   509→    loadStock();
   510→    loadMovements();
   511→    const handler = () => {
   512→      loadStock();
   513→      loadMovements();
   514→    };
   515→    window.addEventListener("refresh-resource", handler);
   516→    return () => window.removeEventListener("refresh-resource", handler);
   517→  }, [loadStock, loadMovements]);
   518→
   519→  function addLine() {
   520→    setLines([
   521→      ...lines,
   522→      { id: `line-${Date.now()}`, product_id: "", quantity: 1 },
   523→    ]);
   524→  }
   525→
   526→  function removeLine(lineId: string) {
   527→    setLines(lines.filter((l) => l.id !== lineId));
   528→  }
   529→
   530→  function updateLine(lineId: string, field: "product_id" | "quantity", value: string | number) {
   531→    setLines(
   532→      lines.map((l) => (l.id === lineId ? { ...l, [field]: value } : l))
   533→    );
   534→  }
   535→
   536→  function getStockForProduct(productId: string): number {
   537→    const item = stock.find((s) => s.product_id === productId);
   538→    return item?.quantity_available ?? 0;
   539→  }
   540→
   541→  async function handleDispense(e: React.FormEvent) {
   542→    e.preventDefault();
   543→    if (!patientId || lines.length === 0) {
   544→      showToast("Veuillez sélectionner un patient et au moins un produit.", "error");
   545→      return;
   546→    }
   547→
   548→    const invalidLines = lines.filter(
   549→      (l) => !l.product_id || l.quantity <= 0
   550→    );
   551→    if (invalidLines.length > 0) {
   552→      showToast("Veuillez remplir toutes les lignes de prescription.", "error");
   553→      return;
   554→    }
   555→
   556→    setSubmitting(true);
   557→    try {
   558→      for (const line of lines) {
   559→        await apiRequest("/pharmacy/stock/movements", {
   560→          method: "POST",
   561→          body: JSON.stringify({
   562→            facility_id: facilityId,
   563→            product_id: line.product_id,
   564→            movement_type: "OUT",
   565→            quantity: line.quantity,
   566→            reason: `Dispensation — Patient: ${patientId} — Prescripteur: ${prescriberId || "N/A"} — Ordonnance: ${prescription || "N/A"}`,
   567→          }),
   568→        });
   569→      }
   570→      showToast("Dispensation enregistrée avec succès.", "success");
   571→      setPatientId("");
   572→      setPrescription("");
   573→      setPrescriberId("");
   574→      setLines([]);
   575→      onCreated();
   576→      loadStock();
   577→      loadMovements();
   578→    } catch (err: any) {
   579→      showToast(err.message || "Erreur lors de la dispensation.", "error");
   580→    } finally {
   581→      setSubmitting(false);
   582→    }
   583→  }
   584→
   585→  return (
   586→    <>
   587→      {/* Dispensation Form */}
   588→      <div className="card form-card">
   589→        <h3 style={{ marginBottom: "16px" }}>Nouvelle dispensation</h3>
   590→        <form onSubmit={handleDispense}>
   591→          <div className="form-grid" style={{ marginBottom: "16px" }}>
