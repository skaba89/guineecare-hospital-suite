   420→interface InvoiceLine {
   421→  id: string;
   422→  tariff_item_id: string;
   423→  quantity: number;
   424→  unit_price: number;

   426→
   427→function InvoicesTab({
   428→  lookups,
   429→  onCreated,
   430→}: {
   431→  lookups: LookupData;
   432→  onCreated: () => void;
   433→}) {
   434→  const options = buildOptions(lookups);
   435→  const facilityId = firstValue(options.facilities);
   436→
   437→  // Tarifs chargés en plein (petite table de référence pour le dropdown du formulaire)
   438→  const [tariffs, setTariffs] = useState<Row[]>([]);
   439→  const [showForm, setShowForm] = useState(false);
   440→  const [submitting, setSubmitting] = useState(false);
   441→
   442→  // Liste paginée des factures (recherche server-side + filtre statut)
   443→  const [statusFilter, setStatusFilter] = useState<string>("");
   444→  const {
   445→    items: invoices,
   446→    total,
   447→    page,
   448→    totalPages,
   449→    loading,
   450→    error,
   451→    search,
   452→    setSearch,
   453→    setPage,
   454→    reload,
   455→  } = usePaginatedList<Row>("/billing/invoices", {
   456→    pageSize: 20,
   457→    debounceMs: 300,
   458→    extraParams: { status: statusFilter || null },
   459→  });
   460→
   461→  // Charger les tarifs (petite table de référence) une seule fois
   462→  useEffect(() => {
   463→    async function loadTariffs() {
   464→      try {
   465→        const tariffsRes = await apiRequest<any>("/billing/tariffs?page_size=1000");
   466→        setTariffs(Array.isArray(tariffsRes.data) ? tariffsRes.data : []);
   467→      } catch {
   468→        // silent — le form ne sera pas bloqué
   469→      }
   470→    }
   471→    loadTariffs();
   472→  }, []);
   473→
   474→  // Réagir aux refresh globaux (création d'facture ailleurs)
   475→  useEffect(() => {
   476→    const handler = () => reload();
   477→    window.addEventListener("refresh-resource", handler);
   478→    return () => window.removeEventListener("refresh-resource", handler);
   479→  }, [reload]);
   480→
   481→  // Form fields
   482→  const [patientId, setPatientId] = useState("");
   483→  const [invoiceLines, setInvoiceLines] = useState<InvoiceLine[]>([]);
   484→  const [invoiceNumber, setInvoiceNumber] = useState("");
   485→  const [description, setDescription] = useState("");
   486→
   487→  function addLine() {
   488→    setInvoiceLines([
   489→      ...invoiceLines,
   490→      { id: `line-${Date.now()}`, tariff_item_id: "", quantity: 1, unit_price: 0 },
   491→    ]);
   492→  }
   493→
   494→  function removeLine(lineId: string) {
   495→    setInvoiceLines(invoiceLines.filter((l) => l.id !== lineId));
   496→  }
   497→
   498→  function updateLine(
   499→    lineId: string,
   500→    field: "tariff_item_id" | "quantity",
   501→    value: string | number
   502→  ) {
   503→    setInvoiceLines(
   504→      invoiceLines.map((l) => {
   505→        if (l.id !== lineId) return l;
   506→        if (field === "tariff_item_id") {
   507→          const tariff = tariffs.find((t) => t.id === value);
   508→          return {
   509→            ...l,
   510→            tariff_item_id: value as string,
   511→            unit_price: tariff?.unit_price || 0,
   512→          };
   513→        }
   514→        return { ...l, quantity: Number(value) || 1 };
   515→      })
   516→    );
   517→  }
   518→
   519→  const totalAmount = invoiceLines.reduce(
   520→    (sum, l) => sum + l.unit_price * l.quantity,
   521→    0
   522→  );
   523→
   524→  function generateInvoiceNumber(): string {
   525→    const now = new Date();
   526→    const y = now.getFullYear();
   527→    const m = String(now.getMonth() + 1).padStart(2, "0");
   528→    const seq = String(total + 1).padStart(4, "0");
   529→    return `FAC-${y}${m}-${seq}`;
   530→  }
   531→
   532→  async function handleSubmit(e: React.FormEvent) {
   533→    e.preventDefault();
   534→    if (!patientId || invoiceLines.length === 0) {
   535→      showToast("Veuillez sélectionner un patient et ajouter au moins une ligne.", "error");
   536→      return;
   537→    }
   538→    const invalidLines = invoiceLines.filter(
   539→      (l) => !l.tariff_item_id || l.quantity <= 0
   540→    );
   541→    if (invalidLines.length > 0) {
   542→      showToast("Veuillez remplir toutes les lignes de facturation.", "error");
   543→      return;
   544→    }
   545→
   546→    setSubmitting(true);
   547→    try {
   548→      const invNum = invoiceNumber || generateInvoiceNumber();
   549→      await apiRequest("/billing/invoices", {
   550→        method: "POST",
   551→        body: JSON.stringify({
   552→          facility_id: facilityId,
   553→          patient_id: patientId,
   554→          invoice_number: invNum,
   555→          description: description || undefined,
   556→          net_amount: totalAmount,
   557→        }),
   558→      });
   559→      showToast("Facture créée avec succès.", "success");
   560→      setPatientId("");
   561→      setInvoiceLines([]);
   562→      setInvoiceNumber("");
   563→      setDescription("");
   564→      setShowForm(false);
   565→      onCreated();
   566→      reload();
   567→    } catch (err: any) {
   568→      showToast(err.message || "Erreur lors de la création.", "error");
   569→    } finally {
   570→      setSubmitting(false);
   571→    }
   572→  }
   573→
   574→  const tariffOptions = tariffs.map((t) => ({
   575→    value: t.id,
   576→    label: `${t.code} — ${t.name} (${(t.unit_price || 0).toLocaleString("fr-FR")} GNF)`,
   577→  }));
   578→
