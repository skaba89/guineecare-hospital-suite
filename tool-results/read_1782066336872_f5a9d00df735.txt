    83→    83→/* Map action_name prefixes to action types */
    84→    84→function inferActionType(actionName: string): string {
    85→    85→  const lower = actionName.toLowerCase();
    86→    86→  if (lower.includes("created") || lower.includes(".create") || lower.includes("admitted"))
    87→    87→    return "CREATE";
    88→    88→  if (lower.includes("deleted") || lower.includes(".delete") || lower.includes("removed"))
    89→    89→    return "DELETE";
    90→    90→  if (lower.includes("login") || lower.includes("logged_in") || lower.includes("authenticated"))
    91→    91→    return "LOGIN";
    92→    92→  if (lower.includes("logout") || lower.includes("logged_out"))
    93→    93→    return "LOGOUT";
    94→    94→  return "UPDATE";
    95→    95→}

    97→    97→/* Map entity_type to module */
    98→    98→function inferModule(entityType: string | null): string {
    99→    99→  if (!entityType) return "AUTH";
   100→   100→  const lower = entityType.toLowerCase();
   101→   101→  if (lower.includes("patient")) return "PATIENTS";
   102→   102→  if (lower.includes("admission")) return "ADMISSIONS";
   103→   103→  if (lower.includes("emergency")) return "EMERGENCY";
   104→   104→  if (lower.includes("product") || lower.includes("dispensation") || lower.includes("pharmacy"))
   105→   105→    return "PHARMACY";
   106→   106→  if (lower.includes("lab") || lower.includes("order") || lower.includes("result"))
   107→   107→    return "LAB";
   108→   108→  if (lower.includes("invoice") || lower.includes("billing") || lower.includes("payment"))
   109→   109→    return "BILLING";
   110→   110→  if (lower.includes("stay") || lower.includes("bed") || lower.includes("room") || lower.includes("hospital"))
   111→   111→    return "HOSPITALIZATION";
   112→   112→  if (lower.includes("maternity") || lower.includes("delivery"))
   113→   113→    return "MATERNITY";
   114→   114→  if (lower.includes("staff") || lower.includes("personnel"))
   115→   115→    return "PERSONNEL";
   116→   116→  if (lower.includes("imaging") || lower.includes("radiology"))
   117→   117→    return "IMAGING";
   118→   118→  if (lower.includes("surgery") || lower.includes("operation"))
   119→   119→    return "SURGERY";
   120→   120→  if (lower.includes("quality") || lower.includes("audit"))
   121→   121→    return "QUALITY";
   122→   122→  if (lower.includes("report"))
   123→   123→    return "REPORTING";
   124→   124→  return "AUTH";
   125→   125→}
