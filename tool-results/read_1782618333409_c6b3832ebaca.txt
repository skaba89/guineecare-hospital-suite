   146→
   147→function TableauTab({
   148→  lookups,
   149→  onCreated,
   150→}: {
   151→  lookups: LookupData;
   152→  onCreated: () => void;
   153→}) {
   154→  const t = useT();
   155→  // Filtres serveur (appliqués via usePaginatedList.extraParams)
   156→  const [statusFilter, setStatusFilter] = useState("OPEN");
   157→  const [deptFilter, setDeptFilter] = useState("");
   158→  const [typeFilter, setTypeFilter] = useState("");
   159→  const [dateFrom, setDateFrom] = useState("");
   160→  const [dateTo, setDateTo] = useState("");
   161→
   162→  // Données KPI (chargées séparément, indépendantes des filtres/pagination)
   163→  const [kpiData, setKpiData] = useState<Row[]>([]);
   164→
   165→  const options = buildOptions(lookups);
   166→
   167→  // Liste paginée des admissions (recherche server-side + filtres serveur).
   168→  // NOTE: le filtre statut est passé via extraParams.status (et NON via search),
   169→  // ce qui corrige l'ancien bug où params.set("search", statusFilter) filtrait
   170→  // sur admission_type au lieu du statut réel.
   171→  const {
   172→    items: admissions,
   173→    total,
   174→    page,
   175→    totalPages,
   176→    loading,
   177→    error,
   178→    search,
   179→    setSearch,
   180→    setPage,
   181→    reload,
   182→  } = usePaginatedList<Row>("/admissions", {
   183→    pageSize: 20,
   184→    debounceMs: 300,
   185→    extraParams: {
   186→      status: statusFilter || null,
   187→      department_id: deptFilter || null,
   188→      admission_type: typeFilter || null,
   189→      date_from: dateFrom || null,
   190→      date_to: dateTo || null,
   191→    },
   192→  });
   193→
   194→  // Charger toutes les admissions pour les KPIs (comptages globaux,
   195→  // indépendants des filtres/pagination de la table)
   196→  async function loadKpis() {
   197→    try {
   198→      const payload = await apiRequest<any>("/admissions?page_size=1000");
   199→      setKpiData(Array.isArray(payload.data) ? payload.data : []);
   200→    } catch {
   201→      /* silent */
   202→    }
   203→  }
   204→
   205→  useEffect(() => {
   206→    loadKpis();
   207→  }, []);
   208→
   209→  // Réagir aux refresh globaux (création/sortie d'admission ailleurs)
   210→  useEffect(() => {
   211→    const handler = () => {
   212→      reload();
   213→      loadKpis();
   214→    };
   215→    window.addEventListener("refresh-resource", handler);
   216→    return () => window.removeEventListener("refresh-resource", handler);
   217→  }, [reload]);
   218→
   219→  function handleRefresh() {
   220→    reload();
   221→    loadKpis();
   222→    onCreated();
   223→  }
   224→
   225→  function handleResetFilters() {
   226→    setStatusFilter("OPEN");
   227→    setDeptFilter("");
   228→    setTypeFilter("");
   229→    setDateFrom("");
   230→    setDateTo("");
   231→    setSearch("");
   232→    setPage(1);
   233→  }
   234→
   235→  /* ── Resolve names ─────────────────────────────── */
   236→  function getPatientName(patientId: string): string {
   237→    const p = lookups.patients.find((x) => x.id === patientId);
   238→    if (!p) return "Inconnu";
   239→    return `${p.first_name || ""} ${p.last_name || ""}`.trim() || p.patient_number || "N/A";
   240→  }
   241→
   242→  function getDeptName(deptId: string | null): string {
   243→    if (!deptId) return "—";
   244→    const d = lookups.departments.find((x) => x.id === deptId);
   245→    return d ? `${d.code || ""} - ${d.name || d.id}`.trim() : "—";
   246→  }
   247→
   248→  function getFacilityName(facilityId: string): string {
   249→    const f = lookups.facilities.find((x) => x.id === facilityId);
   250→    return f ? f.name || f.id : "—";
   251→  }
   252→
   253→  /* ── Compute KPIs ──────────────────────────────── */
   254→  const now = new Date();
   255→  const todayStr = now.toDateString();
   256→
   257→  const kpis = useMemo(() => {
   258→    const active = kpiData.filter(
   259→      (a) => a.status === "OPEN" || a.status === "ACTIVE"
   260→    );
   261→    const todayAdmissions = kpiData.filter((a) => {
   262→      if (!a.admitted_at) return false;
   263→      return new Date(a.admitted_at).toDateString() === todayStr;
   264→    });
   265→    const hospitalized = kpiData.filter(
   266→      (a) =>
   267→        (a.status === "OPEN" || a.status === "ACTIVE") &&
   268→        a.admission_type === "HOSPITALIZATION"
   269→    );
   270→    const plannedDischarges = kpiData.filter(
   271→      (a) => a.status === "CLOSED" || a.status === "DISCHARGED"
   272→    );
   273→    return {
   274→      activeCount: active.length,
   275→      todayCount: todayAdmissions.length,
   276→      hospCount: hospitalized.length,
   277→      dischargeCount: plannedDischarges.length,
   278→    };
   279→  }, [kpiData, todayStr]);
   280→
   281→  /* ── Apply filters ── (filtrage server-side via usePaginatedList.extraParams) */
   282→
   283→  /* ── Color coding helpers ──────────────────────── */
   284→  function getStayIndicator(admission: Row): React.ReactNode {
   285→    if (!admission.admitted_at) return null;
   286→    const admittedDate = new Date(admission.admitted_at);
   287→    const diffHours = (now.getTime() - admittedDate.getTime()) / (1000 * 60 * 60);
   288→    if (diffHours < 24) {
   289→      return (
   290→        <span
   291→          style={{
   292→            display: "inline-block",
   293→            width: "8px",
   294→            height: "8px",
   295→            borderRadius: "50%",
   296→            background: "#16a34a",
   297→            marginRight: "6px",
   298→            flexShrink: 0,
   299→          }}
   300→          title="Admission récente (< 24h)"
   301→        />
   302→      );
   303→    }
   304→    const diffDays = diffHours / 24;
   305→    if (diffDays > 7) {
   306→      return (
   307→        <span
   308→          style={{
   309→            display: "inline-block",
   310→            width: "8px",
   311→            height: "8px",
   312→            borderRadius: "50%",
   313→            background: "#f59e0b",
   314→            marginRight: "6px",
   315→            flexShrink: 0,
   316→          }}
   317→          title="Séjour long (> 7 jours)"
   318→        />
   319→      );
   320→    }
   321→    return null;
   322→  }
   323→
   324→  async function handleDischarge(admissionId: string) {
   325→    if (!confirm("Confirmer la sortie de ce patient ?")) return;
   326→    try {
   327→      await apiRequest(`/admissions/${admissionId}/close`, { method: "POST" });
   328→      showToast("Sortie enregistrée avec succès.", "success");
   329→      handleRefresh();
   330→    } catch {
   331→      showToast("Erreur lors de l'enregistrement de la sortie.", "error");
   332→    }
   333→  }
   334→
   335→  return (
   336→    <>
   337→      {/* ── KPI Cards ──────────────────────────────────── */}
   338→      <div
   339→        style={{
   340→          display: "grid",
   341→          gridTemplateColumns: "repeat(auto-fit, minmax(200px, 1fr))",
   342→          gap: "12px",
   343→          marginBottom: "20px",
   344→        }}
   345→      >
   346→        <KpiCard
   347→          icon={<Activity size={20} />}
   348→          label="Admissions actives"
   349→          value={kpis.activeCount}
   350→          color="var(--primary)"
   351→          bgColor="var(--primary-light)"
   352→        />
   353→        <KpiCard
   354→          icon={<CalendarCheck size={20} />}
   355→          label="Prévues aujourd'hui"
   356→          value={kpis.todayCount}
   357→          color="#2563eb"
   358→          bgColor="#eff6ff"
   359→        />
   360→        <KpiCard
   361→          icon={<BedDouble size={20} />}
   362→          label="En hospitalisation"
   363→          value={kpis.hospCount}
   364→          color="#7c3aed"
   365→          bgColor="#f5f3ff"
   366→        />
   367→        <KpiCard
   368→          icon={<LogOut size={20} />}
   369→          label="Sorties prévues"
   370→          value={kpis.dischargeCount}
   371→          color="#b45309"
   372→          bgColor="#fef3c7"
   373→        />
   374→      </div>
   375→
   376→      {/* ── Barre de recherche (server-side, debounce 300ms) ── */}
   377→      <div
   378→        style={{
   379→          display: "flex",
   380→          gap: 12,
   381→          alignItems: "center",
   382→          marginBottom: 12,
   383→          flexWrap: "wrap",
   384→        }}
   385→      >
   386→        <input
   387→          type="text"
   388→          placeholder="🔍 Rechercher (patient, numéro, motif)..."
   389→          value={search}
   390→          onChange={(e) => setSearch(e.target.value)}
   391→          style={{ flex: 1, minWidth: 250, padding: "8px 12px" }}
   392→        />
   393→      </div>
   394→
   395→      {/* ── Filters ────────────────────────────────────── */}
   396→      <div
   397→        className="card"
   398→        style={{
   399→          marginBottom: "16px",
   400→          display: "flex",
   401→          gap: "12px",
   402→          flexWrap: "wrap",
   403→          alignItems: "end",
   404→          padding: "16px",
   405→        }}
   406→      >
   407→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   408→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Statut</span>
   409→          <select
   410→            value={statusFilter}
   411→            onChange={(e) => {
   412→              setStatusFilter(e.target.value);
   413→              setPage(1);
   414→            }}
   415→            style={{ minWidth: "160px", fontSize: "13px" }}
   416→          >
   417→            <option value="">Tous statuts</option>
   418→            <option value="OPEN">Actives</option>
   419→            <option value="CLOSED">Sortis</option>
   420→          </select>
   421→        </div>
   422→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   423→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Type</span>
   424→          <select
   425→            value={typeFilter}
   426→            onChange={(e) => {
   427→              setTypeFilter(e.target.value);
   428→              setPage(1);
   429→            }}
   430→            style={{ minWidth: "160px", fontSize: "13px" }}
   431→          >
   432→            <option value="">Tous types</option>
   433→            {ADMISSION_TYPE_OPTIONS.map((o) => (
   434→              <option key={o.value} value={o.value}>
   435→                {o.label}
   436→              </option>
   437→            ))}
   438→          </select>
   439→        </div>
   440→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   441→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Service</span>
   442→          <select
   443→            value={deptFilter}
   444→            onChange={(e) => {
   445→              setDeptFilter(e.target.value);
   446→              setPage(1);
   447→            }}
   448→            style={{ minWidth: "180px", fontSize: "13px" }}
   449→          >
   450→            <option value="">Tous les services</option>
   451→            {options.departments.map((o) => (
   452→              <option key={o.value} value={o.value}>
   453→                {o.label}
   454→              </option>
   455→            ))}
   456→          </select>
   457→        </div>
   458→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   459→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Date début</span>
   460→          <input
   461→            type="date"
   462→            value={dateFrom}
   463→            onChange={(e) => {
   464→              setDateFrom(e.target.value);
   465→              setPage(1);
   466→            }}
   467→            style={{ fontSize: "13px" }}
   468→          />
   469→        </div>
   470→        <div style={{ display: "grid", gap: "6px", fontWeight: 600, fontSize: "13px" }}>
   471→          <span style={{ color: "var(--muted)", fontSize: "11px", textTransform: "uppercase", letterSpacing: "0.5px" }}>Date fin</span>
   472→          <input
   473→            type="date"
   474→            value={dateTo}
   475→            onChange={(e) => {
   476→              setDateTo(e.target.value);
   477→              setPage(1);
   478→            }}
   479→            style={{ fontSize: "13px" }}
   480→          />
   481→        </div>
   482→        <button
   483→          className="btn btn-outline btn-sm"
   484→          onClick={handleResetFilters}
   485→          style={{ height: "38px" }}
   486→        >
   487→          <X size={14} />
   488→          Réinitialiser
   489→        </button>
   490→        <button
   491→          className="btn btn-outline btn-sm"
   492→          onClick={handleRefresh}
   493→          style={{ height: "38px" }}
   494→        >
   495→          <RefreshCw size={14} />
   496→          Actualiser
   497→        </button>
   498→      </div>
   499→
   500→      {/* ── Admissions Table ───────────────────────────── */}
   501→      {loading ? (
   502→        <div className="card" style={{ textAlign: "center", padding: "32px" }}>
   503→          <div className="spinner" />
   504→          <p className="muted" style={{ marginTop: "12px" }}>
   505→            {t("label.loading")}
   506→          </p>
   507→        </div>
   508→      ) : error ? (
   509→        <div className="card" style={{ padding: "16px", color: "var(--danger)" }}>
   510→          {error}
   511→        </div>
   512→      ) : admissions.length === 0 ? (
   513→        <div className="card" style={{ textAlign: "center", padding: "32px" }}>
   514→          <p className="muted">Aucune admission trouvée.</p>
   515→        </div>
   516→      ) : (
   517→        <div className="card" style={{ padding: "0" }}>
   518→          <div className="table-wrapper" style={{ maxHeight: "520px", overflowY: "auto" }}>
   519→            <table className="data-table">
   520→              <thead>
   521→                <tr>
   522→                  <th>Patient</th>
   523→                  <th>Date d'admission</th>
   524→                  <th>Motif / Type</th>
   525→                  <th>Service</th>
   526→                  <th>Statut</th>
   527→                  <th>Établissement</th>
   528→                  <th>Actions</th>
   529→                </tr>
   530→              </thead>
   531→              <tbody>
   532→                {admissions.map((adm) => {
   533→                  const statusInfo = STATUS_MAP[adm.status] || {
   534→                    label: adm.status,
   535→                    badge: "badge-gray",
   536→                  };
   537→                  const isActive =
   538→                    adm.status === "OPEN" || adm.status === "ACTIVE";
   539→                  return (
   540→                    <tr key={adm.id}>
   541→                      <td>
   542→                        <div style={{ display: "flex", alignItems: "center" }}>
   543→                          {getStayIndicator(adm)}
   544→                          <span style={{ fontWeight: 600 }}>
   545→                            {getPatientName(adm.patient_id)}
   546→                          </span>
   547→                        </div>
   548→                      </td>
   549→                      <td style={{ whiteSpace: "nowrap", fontSize: "13px" }}>
   550→                        {adm.admitted_at
   551→                          ? new Date(adm.admitted_at).toLocaleString("fr-FR", {
   552→                              day: "2-digit",
   553→                              month: "2-digit",
   554→                              year: "numeric",
   555→                              hour: "2-digit",
   556→                              minute: "2-digit",
   557→                            })
   558→                          : "—"}
   559→                      </td>
   560→                      <td>
   561→                        <span style={{ fontWeight: 500 }}>
   562→                          {TYPE_LABELS[adm.admission_type] ||
   563→                            adm.admission_type ||
   564→                            "—"}
   565→                        </span>
   566→                      </td>
   567→                      <td style={{ fontSize: "13px" }}>
   568→                        {getDeptName(adm.department_id)}
   569→                      </td>
   570→                      <td>
   571→                        <span className={`badge ${statusInfo.badge}`}>
   572→                          {statusInfo.label}
   573→                        </span>
   574→                      </td>
   575→                      <td style={{ fontSize: "13px" }}>
   576→                        {getFacilityName(adm.facility_id)}
   577→                      </td>
   578→                      <td>
   579→                        <div style={{ display: "flex", gap: "6px" }}>
   580→                          <button
