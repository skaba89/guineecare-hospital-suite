   253→   253→def seed_default_indicators(db: Session, facility_id: str | None = None) -> int:
   254→   254→    """Insère les indicateurs prédéfinis (OMS/HAS) s'ils n'existent pas encore.
   255→   255→
   256→   256→    Args:
   257→   257→        db: session SQLAlchemy
   258→   258→        facility_id: si fourni, ne crée que pour cette facility. Si None, crée
   259→   259→                     pour toutes les facilities existantes (ou global si aucune).
   260→   260→
   261→   261→    Returns:
   262→   262→        Nombre d'indicateurs créés.
   263→   263→    """
   264→   264→    created = 0
   265→   265→    facilities = []
   266→   266→    if facility_id:
   267→   267→        facilities = [facility_id]
   268→   268→    else:
   269→   269→        from app.modules.facilities.models import Facility
   270→   270→        facilities = [f.id for f in db.query(Facility.id).all()]
   271→   271→        if not facilities:
   272→   272→            # Aucune facility — ne crée rien (les indicateurs seront seedés
   273→   273→            # ultérieurement quand une facility existera)
   274→   274→            return 0
   275→   275→
   276→   276→    for fid in facilities:
   277→   277→        for ind_data in DEFAULT_INDICATORS:
   278→   278→            existing = (
   279→   279→                db.query(QualityIndicator)
   280→   280→                .filter(QualityIndicator.facility_id == fid)
   281→   281→                .filter(QualityIndicator.code == ind_data["code"])
   282→   282→                .first()
   283→   283→            )
   284→   284→            if existing:
   285→   285→                continue
   286→   286→            ind = QualityIndicator(
   287→   287→                facility_id=fid,
   288→   288→                code=ind_data["code"],
   289→   289→                name=ind_data["name"],
   290→   290→                category=ind_data["category"],
   291→   291→                description=ind_data["description"],
   292→   292→                unit=ind_data["unit"],
   293→   293→                target_value=ind_data["target_value"],
   294→   294→                frequency=ind_data["frequency"],
   295→   295→            )
   296→   296→            db.add(ind)
   297→   297→            created += 1
   298→   298→
   299→   299→    db.commit()
   300→   300→    return created
   301→   301→

   303→   303→def seed_default_thresholds(db: Session, facility_id: str | None = None) -> int:
   304→   304→    """Insère les seuils par défaut liés aux indicateurs prédéfinis."""
   305→   305→    created = 0
   306→   306→    facilities = []
   307→   307→    if facility_id:
   308→   308→        facilities = [facility_id]
   309→   309→    else:
   310→   310→        from app.modules.facilities.models import Facility
   311→   311→        facilities = [f.id for f in db.query(Facility.id).all()]
   312→   312→        if not facilities:
   313→   313→            return 0
   314→   314→
   315→   315→    for fid in facilities:
   316→   316→        for th_data in DEFAULT_THRESHOLDS:
   317→   317→            indicator = (
   318→   318→                db.query(QualityIndicator)
   319→   319→                .filter(QualityIndicator.facility_id == fid)
   320→   320→                .filter(QualityIndicator.code == th_data["indicator_code"])
   321→   321→                .first()
   322→   322→            )
   323→   323→            if not indicator:
   324→   324→                continue
   325→   325→            existing = (
   326→   326→                db.query(QualityThreshold)
   327→   327→                .filter(QualityThreshold.facility_id == fid)
   328→   328→                .filter(QualityThreshold.indicator_id == indicator.id)
   329→   329→                .filter(QualityThreshold.comparator == th_data["comparator"])
   330→   330→                .filter(QualityThreshold.threshold_value == th_data["threshold_value"])
   331→   331→                .first()
   332→   332→            )
   333→   333→            if existing:
   334→   334→                continue
   335→   335→            th = QualityThreshold(
   336→   336→                facility_id=fid,
   337→   337→                indicator_id=indicator.id,
   338→   338→                comparator=th_data["comparator"],
   339→   339→                threshold_value=th_data["threshold_value"],
   340→   340→                severity=th_data["severity"],
   341→   341→                alert_message=th_data["alert_message"],
   342→   342→                notify_roles=th_data["notify_roles"],
   343→   343→                channels=th_data["channels"],
   344→   344→                cooldown_hours=th_data["cooldown_hours"],
   345→   345→                enabled="true",
   346→   346→            )
   347→   347→            db.add(th)
   348→   348→            created += 1
   349→   349→
   350→   350→    db.commit()
   351→   351→    return created

   358→   358→def compute_dashboard(
   359→   359→    db: Session,
   360→   360→    *,
   361→   361→    facility_id: str | None = None,
   362→   362→    department_id: str | None = None,
   363→   363→    period_start: datetime | None = None,
   364→   364→    period_end: datetime | None = None,
   365→   365→) -> dict[str, Any]:
   366→   366→    """Calcule le dashboard qualité agrégé pour une période.
   367→   367→
   368→   368→    Retourne :
   369→   369→    - `kpis` : liste des derniers KPIs mesurés (par indicateur).
   370→   370→    - `incidents` : agrégats par type/sévérité/statut.
   371→   371→    - `alerts` : alertes ouvertes récentes.
   372→   372→    - `trends` : séries temporelles sur la période (par indicateur principal).
   373→   373→    """
   374→   374→    period_end = period_end or utcnow()
   375→   375→    period_start = period_start or (period_end - timedelta(days=30))
   376→   376→
   377→   377→    # KPIs : dernières mesures par indicateur
   378→   378→    indicators_q = db.query(QualityIndicator)
   379→   379→    if facility_id:
   380→   380→        indicators_q = indicators_q.filter(QualityIndicator.facility_id == facility_id)
   381→   381→    indicators = indicators_q.order_by(QualityIndicator.category, QualityIndicator.code).all()
   382→   382→
   383→   383→    kpis = []
   384→   384→    for ind in indicators:
   385→   385→        last_measure = (
   386→   386→            db.query(QualityMeasurement)
   387→   387→            .filter(QualityMeasurement.indicator_id == ind.id)
   388→   388→            .filter(QualityMeasurement.period_start >= period_start)
   389→   389→            .filter(QualityMeasurement.period_end <= period_end)
   390→   390→            .order_by(QualityMeasurement.period_end.desc())
   391→   391→            .first()
   392→   392→        )
   393→   393→        if not last_measure:
   394→   394→            # Pas de mesure sur la période — on prend la plus récente absolue
   395→   395→            last_measure = (
   396→   396→                db.query(QualityMeasurement)
   397→   397→                .filter(QualityMeasurement.indicator_id == ind.id)
   398→   398→                .order_by(QualityMeasurement.period_end.desc())
   399→   399→                .first()
   400→   400→            )
   401→   401→        kpis.append({
   402→   402→            "indicator_id": ind.id,
   403→   403→            "indicator_code": ind.code,
   404→   404→            "indicator_name": ind.name,
   405→   405→            "category": ind.category,
   406→   406→            "unit": ind.unit,
   407→   407→            "target_value": ind.target_value,
   408→   408→            "frequency": ind.frequency,
   409→   409→            "last_value": last_measure.value if last_measure else None,
   410→   410→            "last_period_start": last_measure.period_start.isoformat() if last_measure and last_measure.period_start else None,
   411→   411→            "last_period_end": last_measure.period_end.isoformat() if last_measure and last_measure.period_end else None,
   412→   412→            "has_data": last_measure is not None,
   413→   413→        })
   414→   414→
   415→   415→    # Incidents : agrégats
   416→   416→    inc_q = db.query(IncidentReport)
   417→   417→    if facility_id:
   418→   418→        inc_q = inc_q.filter(IncidentReport.facility_id == facility_id)
   419→   419→    inc_q = inc_q.filter(IncidentReport.incident_date >= period_start).filter(IncidentReport.incident_date <= period_end)
   420→   420→    incidents_total = inc_q.count()
   421→   421→    incidents_by_type = (
   422→   422→        db.query(IncidentReport.incident_type, func.count(IncidentReport.id))
   423→   423→        .filter(IncidentReport.facility_id == facility_id if facility_id else True)
   424→   424→        .filter(IncidentReport.incident_date >= period_start)
   425→   425→        .filter(IncidentReport.incident_date <= period_end)
   426→   426→        .group_by(IncidentReport.incident_type)
   427→   427→        .all()
   428→   428→    )
   429→   429→    incidents_by_severity = (
   430→   430→        db.query(IncidentReport.severity, func.count(IncidentReport.id))
   431→   431→        .filter(IncidentReport.facility_id == facility_id if facility_id else True)
   432→   432→        .filter(IncidentReport.incident_date >= period_start)
   433→   433→        .filter(IncidentReport.incident_date <= period_end)
   434→   434→        .group_by(IncidentReport.severity)
   435→   435→        .all()
   436→   436→    )
   437→   437→    incidents_by_status = (
   438→   438→        db.query(IncidentReport.status, func.count(IncidentReport.id))
   439→   439→        .filter(IncidentReport.facility_id == facility_id if facility_id else True)
   440→   440→        .filter(IncidentReport.incident_date >= period_start)
   441→   441→        .filter(IncidentReport.incident_date <= period_end)
   442→   442→        .group_by(IncidentReport.status)
   443→   443→        .all()
   444→   444→    )
   445→   445→
   446→   446→    # Délai moyen de traitement des incidents (RESOLVED uniquement)
   447→   447→    resolved_incidents = (
   448→   448→        db.query(IncidentReport)
   449→   449→        .filter(IncidentReport.status.in_(["RESOLVED", "CLOSED"]))
   450→   450→        .filter(IncidentReport.facility_id == facility_id if facility_id else True)
   451→   451→        .filter(IncidentReport.incident_date >= period_start)
   452→   452→        .filter(IncidentReport.incident_date <= period_end)
   453→   453→        .all()
   454→   454→    )
   455→   455→    # Pour le délai, on utilise created_at → résolution (le statut RESOLVED ne
   456→   456→    # porte pas de timestamp dédié — approximation)
   457→   457→    avg_resolution_hours = None
   458→   458→    if resolved_incidents:
   459→   459→        total_hours = 0
   460→   460→        count = 0
   461→   461→        for inc in resolved_incidents:
   462→   462→            if inc.created_at:
   463→   463→                delta = (period_end - inc.created_at).total_seconds() / 3600
   464→   464→                total_hours += delta
   465→   465→                count += 1
   466→   466→        if count > 0:
   467→   467→            avg_resolution_hours = round(total_hours / count, 1)
   468→   468→
   469→   469→    # Alertes ouvertes
   470→   470→    alerts_q = db.query(QualityAlert)
   471→   471→    if facility_id:
   472→   472→        alerts_q = alerts_q.filter(QualityAlert.facility_id == facility_id)
   473→   473→    alerts_open = alerts_q.filter(QualityAlert.status == "OPEN").count()
   474→   474→    alerts_acked = alerts_q.filter(QualityAlert.status == "ACKNOWLEDGED").count()
   475→   475→    alerts_resolved = alerts_q.filter(QualityAlert.status == "RESOLVED").count()
   476→   476→    alerts_recent = (
   477→   477→        alerts_q.order_by(QualityAlert.created_at.desc())
   478→   478→        .limit(10)
   479→   479→        .all()
   480→   480→    )
   481→   481→
   482→   482→    # Tendances : séries temporelles sur la période pour les 5 premiers indicateurs
   483→   483→    trends = []
   484→   484→    for ind in indicators[:5]:
   485→   485→        measures = (
   486→   486→            db.query(QualityMeasurement)
   487→   487→            .filter(QualityMeasurement.indicator_id == ind.id)
   488→   488→            .filter(QualityMeasurement.period_start >= period_start)
   489→   489→            .filter(QualityMeasurement.period_end <= period_end)
   490→   490→            .order_by(QualityMeasurement.period_start)
   491→   491→            .all()
   492→   492→        )
   493→   493→        trends.append({
   494→   494→            "indicator_code": ind.code,
   495→   495→            "indicator_name": ind.name,
   496→   496→            "unit": ind.unit,
   497→   497→            "target_value": ind.target_value,
   498→   498→            "data_points": [
   499→   499→                {
   500→   500→                    "period_start": m.period_start.isoformat() if m.period_start else None,
   501→   501→                    "period_end": m.period_end.isoformat() if m.period_end else None,
   502→   502→                    "value": m.value,
   503→   503→                }
   504→   504→                for m in measures
   505→   505→            ],
   506→   506→        })
   507→   507→
   508→   508→    return {
   509→   509→        "period_start": period_start.isoformat(),
   510→   510→        "period_end": period_end.isoformat(),
   511→   511→        "facility_id": facility_id,
   512→   512→        "department_id": department_id,
   513→   513→        "kpis": kpis,
   514→   514→        "incidents": {
   515→   515→            "total": incidents_total,
   516→   516→            "by_type": [{"type": t, "count": int(c)} for (t, c) in incidents_by_type],
   517→   517→            "by_severity": [{"severity": s, "count": int(c)} for (s, c) in incidents_by_severity],
   518→   518→            "by_status": [{"status": s, "count": int(c)} for (s, c) in incidents_by_status],
   519→   519→            "avg_resolution_hours": avg_resolution_hours,
   520→   520→        },
   521→   521→        "alerts": {
   522→   522→            "open": alerts_open,
   523→   523→            "acknowledged": alerts_acked,
   524→   524→            "resolved": alerts_resolved,
   525→   525→            "recent": [a.to_dict() for a in alerts_recent],
   526→   526→        },
   527→   527→        "trends": trends,
   528→   528→        "thresholds_count": db.query(QualityThreshold).filter(
   529→   529→            QualityThreshold.facility_id == facility_id if facility_id else True
   530→   530→        ).count(),
   531→   531→    }

   538→   538→def check_thresholds(
   539→   539→    db: Session,
   540→   540→    *,
   541→   541→    measurement: QualityMeasurement | None = None,
   542→   542→    facility_id: str | None = None,
   543→   543→) -> list[QualityAlert]:
   544→   544→    """Évalue les seuils pour une mesure donnée (ou toutes les mesures récentes).
   545→   545→
   546→   546→    Si `measurement` est fourni, n'évalue que les thresholds de son indicateur.
   547→   547→    Sinon, évalue toutes les mesures des 7 derniers jours pour `facility_id`.
   548→   548→
   549→   549→    Pour chaque seuil franchi :
   550→   550→    1. Vérifie le cooldown (pas d'alerte similaire dans les `cooldown_hours` dernières heures).
   551→   551→    2. Lève une `QualityAlert` OPEN.
   552→   552→    3. Notifie les rôles configurés via `notify()` du module notifications.
   553→   553→    4. Déclenche un SMS si la règle de routage le permet (catégorie `quality_alert`).
   554→   554→
   555→   555→    NE LEVE JAMAIS d'exception — toute erreur de notification est journalisée.
   556→   556→    """
   557→   557→    raised: list[QualityAlert] = []
   558→   558→
   559→   559→    # Déterminer les thresholds à évaluer
   560→   560→    if measurement:
   561→   561→        thresholds_q = (
   562→   562→            db.query(QualityThreshold)
   563→   563→            .filter(QualityThreshold.indicator_id == measurement.indicator_id)
   564→   564→            .filter(QualityThreshold.enabled == "true")
   565→   565→        )
   566→   566→        if measurement.facility_id:
   567→   567→            thresholds_q = thresholds_q.filter(
   568→   568→                (QualityThreshold.facility_id == measurement.facility_id) |
   569→   569→                (QualityThreshold.facility_id.is_(None))
   570→   570→            )
   571→   571→        thresholds = thresholds_q.all()
   572→   572→        measurements_to_check = [measurement]
   573→   573→    else:
   574→   574→        # Toutes les mesures des 7 derniers jours
   575→   575→        since = utcnow() - timedelta(days=7)
   576→   576→        measures_q = (
   577→   577→            db.query(QualityMeasurement)
   578→   578→            .filter(QualityMeasurement.created_at >= since)
   579→   579→        )
   580→   580→        if facility_id:
   581→   581→            measures_q = measures_q.filter(QualityMeasurement.facility_id == facility_id)
   582→   582→        measurements_to_check = measures_q.all()
   583→   583→
   584→   584→        # Tous les thresholds enabled
   585→   585→        thresholds_q = db.query(QualityThreshold).filter(QualityThreshold.enabled == "true")
   586→   586→        if facility_id:
   587→   587→            thresholds_q = thresholds_q.filter(
   588→   588→                (QualityThreshold.facility_id == facility_id) |
   589→   589→                (QualityThreshold.facility_id.is_(None))
   590→   590→            )
   591→   591→        thresholds = thresholds_q.all()
   592→   592→
   593→   593→    # Indexer les thresholds par indicator_id pour lookup rapide
   594→   594→    thresholds_by_indicator: dict[str, list[QualityThreshold]] = {}
   595→   595→    for th in thresholds:
   596→   596→        thresholds_by_indicator.setdefault(th.indicator_id, []).append(th)
   597→   597→
   598→   598→    for m in measurements_to_check:
   599→   599→        ths = thresholds_by_indicator.get(m.indicator_id, [])
   600→   600→        for th in ths:
   601→   601→            # Vérifier le comparateur
   602→   602→            if th.comparator not in COMPARATORS:
   603→   603→                continue
   604→   604→            if not evaluate_threshold(th.comparator, m.value, th.threshold_value):
   605→   605→                continue
   606→   606→
   607→   607→            # Cooldown : pas d'alerte similaire dans la fenêtre
   608→   608→            cooldown_hours = 24
   609→   609→            if th.cooldown_hours and str(th.cooldown_hours).isdigit():
   610→   610→                cooldown_hours = int(th.cooldown_hours)
   611→   611→            cooldown_since = utcnow() - timedelta(hours=cooldown_hours)
   612→   612→            recent_alert = (
   613→   613→                db.query(QualityAlert)
   614→   614→                .filter(QualityAlert.threshold_id == th.id)
   615→   615→                .filter(QualityAlert.created_at >= cooldown_since)
   616→   616→                .filter(QualityAlert.status.in_(["OPEN", "ACKNOWLEDGED"]))
   617→   617→                .first()
   618→   618→            )
   619→   619→            if recent_alert:
   620→   620→                continue  # cooldown actif
   621→   621→
   622→   622→            # Lever l'alerte
   623→   623→            title = f"[{th.severity}] Seuil qualité franchi"
   624→   624→            message = (th.alert_message or "").replace("{{value}}", str(m.value)).replace("{{threshold}}", str(th.threshold_value))
   625→   625→
   626→   626→            alert = QualityAlert(
   627→   627→                facility_id=m.facility_id,
   628→   628→                department_id=th.department_id,
   629→   629→                threshold_id=th.id,
   630→   630→                measurement_id=m.id,
   631→   631→                indicator_id=m.indicator_id,
   632→   632→                status="OPEN",
   633→   633→                severity=th.severity,
   634→   634→                title=title,
   635→   635→                message=message,
   636→   636→                observed_value=str(m.value),
   637→   637→                threshold_value=str(th.threshold_value),
   638→   638→                comparator=th.comparator,
   639→   639→            )
   640→   640→            db.add(alert)
   641→   641→            db.commit()
   642→   642→            db.refresh(alert)
   643→   643→            raised.append(alert)
   644→   644→
   645→   645→            # Notifier (best-effort)
   646→   646→            try:
   647→   647→                _notify_alert(db, alert, th)
   648→   648→            except Exception as e:
   649→   649→                logger.warning("Alert notification failed for alert %s: %s", alert.id, e)
   650→   650→
   651→   651→    return raised

   654→   654→def _notify_alert(db: Session, alert: QualityAlert, threshold: QualityThreshold) -> None:
   655→   655→    """Envoie une notification multi-canal pour une alerte qualité.
   656→   656→
   657→   657→    - Identifie les destinataires par rôle (notify_roles) dans la facility.
   658→   658→    - Appelle `notify()` du module notifications.
   659→   659→    - Si la catégorie `quality_alert` a une règle de routage SMS, un SMS est envoyé.
   660→   660→    """
   661→   661→    from app.modules.notifications.service import notify
   662→   662→    from app.modules.users.models import User
   663→   663→
   664→   664→    if not threshold.notify_roles:
   665→   665→        return
   666→   666→
   667→   667→    roles = [r for r in threshold.notify_roles.split(",") if r]
   668→   668→    channels = [c for c in (threshold.channels or "in_app").split(",") if c]
   669→   669→
   670→   670→    # Trouver les destinataires (SUPER_ADMIN toujours + rôles configurés dans la facility)
   671→   671→    recipients_q = db.query(User).filter(User.is_active.is_(True))
   672→   672→    if alert.facility_id:
   673→   673→        recipients_q = recipients_q.filter(
   674→   674→            (User.facility_id == alert.facility_id) | (User.role == "SUPER_ADMIN")
   675→   675→        )
   676→   676→    recipients_q = recipients_q.filter(User.role.in_(roles))
   677→   677→    recipients = recipients_q.all()
   678→   678→
   679→   679→    # Si aucun destinataire direct, notifier les SUPER_ADMIN globaux
   680→   680→    if not recipients:
   681→   681→        recipients = db.query(User).filter(User.role == "SUPER_ADMIN").filter(User.is_active.is_(True)).all()
   682→   682→
   683→   683→    priority = "urgent" if alert.severity == "CRITICAL" else "high"
   684→   684→
   685→   685→    for recipient in recipients:
   686→   686→        try:
   687→   687→            notif = notify(
   688→   688→                db=db,
   689→   689→                recipient_id=recipient.id,
   690→   690→                title=alert.title,
   691→   691→                body=alert.message,
   692→   692→                category="quality_alert",
   693→   693→                priority=priority,
   694→   694→                channels=channels,
   695→   695→                facility_id=alert.facility_id,
   696→   696→                resource_type="quality_alert",
   697→   697→                resource_id=alert.id,
   698→   698→                recipient_email=getattr(recipient, "email", None),
   699→   699→                recipient_phone=getattr(recipient, "phone", None),
   700→   700→            )
   701→   701→            # Lier la notification à l'alerte (premier destinataire seulement)
   702→   702→            if not alert.notification_id:
   703→   703→                alert.notification_id = notif.id
   704→   704→                db.commit()
   705→   705→        except Exception as e:
   706→   706→            logger.warning("notify() failed for recipient %s: %s", recipient.id, e)
   707→   707→

   713→   713→def acknowledge_alert(
   714→   714→    db: Session,
   715→   715→    alert_id: str,
   716→   716→    user_id: str,
   717→   717→    assign_to: str | None = None,
   718→   718→) -> QualityAlert | None:
   719→   719→    """Marque une alerte comme prise en charge (ACKNOWLEDGED)."""
   720→   720→    alert = db.query(QualityAlert).filter(QualityAlert.id == alert_id).first()
   721→   721→    if not alert:
   722→   722→        return None
   723→   723→    alert.status = "ACKNOWLEDGED"
   724→   724→    alert.acknowledged_at = utcnow()
   725→   725→    alert.acknowledged_by = user_id
   726→   726→    if assign_to:
   727→   727→        alert.assigned_to = assign_to
   728→   728→    elif not alert.assigned_to:
   729→   729→        alert.assigned_to = user_id
   730→   730→    db.commit()
   731→   731→    db.refresh(alert)
   732→   732→    return alert
