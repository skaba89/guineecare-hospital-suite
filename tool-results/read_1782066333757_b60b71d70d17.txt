    83→/* Map action_name prefixes to action types */
    84→function inferActionType(actionName: string): string {
    85→  const lower = actionName.toLowerCase();
    86→  if (lower.includes("created") || lower.includes(".create") || lower.includes("admitted"))
    87→    return "CREATE";
    88→  if (lower.includes("deleted") || lower.includes(".delete") || lower.includes("removed"))
    89→    return "DELETE";
    90→  if (lower.includes("login") || lower.includes("logged_in") || lower.includes("authenticated"))
    91→    return "LOGIN";
    92→  if (lower.includes("logout") || lower.includes("logged_out"))
    93→    return "LOGOUT";
    94→  return "UPDATE";
    95→}
    96→
    97→/* Map entity_type to module */
    98→function inferModule(entityType: string | null): string {
    99→  if (!entityType) return "AUTH";
   100→  const lower = entityType.toLowerCase();
   101→  if (lower.includes("patient")) return "PATIENTS";
   102→  if (lower.includes("admission")) return "ADMISSIONS";
   103→  if (lower.includes("emergency")) return "EMERGENCY";
   104→  if (lower.includes("product") || lower.includes("dispensation") || lower.includes("pharmacy"))
   105→    return "PHARMACY";
   106→  if (lower.includes("lab") || lower.includes("order") || lower.includes("result"))
   107→    return "LAB";
   108→  if (lower.includes("invoice") || lower.includes("billing") || lower.includes("payment"))
   109→    return "BILLING";
   110→  if (lower.includes("stay") || lower.includes("bed") || lower.includes("room") || lower.includes("hospital"))
   111→    return "HOSPITALIZATION";
   112→  if (lower.includes("maternity") || lower.includes("delivery"))
   113→    return "MATERNITY";
   114→  if (lower.includes("staff") || lower.includes("personnel"))
   115→    return "PERSONNEL";
   116→  if (lower.includes("imaging") || lower.includes("radiology"))
   117→    return "IMAGING";
   118→  if (lower.includes("surgery") || lower.includes("operation"))
   119→    return "SURGERY";
   120→  if (lower.includes("quality") || lower.includes("audit"))
   121→    return "QUALITY";
   122→  if (lower.includes("report"))
   123→    return "REPORTING";
   124→  return "AUTH";
   125→}
