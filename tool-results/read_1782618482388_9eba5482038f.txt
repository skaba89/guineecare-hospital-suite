    92→/* ─── Rooms Tab ──────────────────────────────────────────── */
    93→
    94→function RoomsTab({ lookups }: { lookups: LookupData }) {
    95→  const [rooms, setRooms] = useState<Row[]>([]);
    96→  const [loading, setLoading] = useState(true);
    97→  const [error, setError] = useState("");
    98→  const [showForm, setShowForm] = useState(false);
    99→  const [submitting, setSubmitting] = useState(false);
   100→
   101→  // Form state
   102→  const [code, setCode] = useState("");
   103→  const [name, setName] = useState("");
   104→  const [roomType, setRoomType] = useState("");
   105→
   106→  const loadRooms = useCallback(async () => {
   107→    setLoading(true);
   108→    setError("");
   109→    try {
   110→      const payload = await apiRequest<any>("/surgery/rooms?page_size=1000");
   111→      setRooms(Array.isArray(payload.data) ? payload.data : []);
   112→    } catch {
   113→      setError("Impossible de charger les salles.");
   114→    } finally {
   115→      setLoading(false);
   116→    }
   117→  }, []);
   118→
   119→  useEffect(() => {
   120→    loadRooms();
   121→    const handler = () => loadRooms();
   122→    window.addEventListener("refresh-resource", handler);
   123→    return () => window.removeEventListener("refresh-resource", handler);
   124→  }, [loadRooms]);
   125→
   126→  // Count by status
   127→  const statusCounts = rooms.reduce<Record<string, number>>((acc, room) => {
   128→    acc[room.status] = (acc[room.status] || 0) + 1;
   129→    return acc;
   130→  }, {});
   131→
   132→  async function handleSubmit(e: React.FormEvent) {
   133→    e.preventDefault();
   134→    if (!code || !name) return;
   135→    setSubmitting(true);
   136→    try {
   137→      await apiRequest("/surgery/rooms", {
   138→        method: "POST",
   139→        body: JSON.stringify({
   140→          code: code.trim(),
   141→          name: name.trim(),
   142→          type: roomType.trim() || undefined,
   143→        }),
   144→      });
   145→      setCode("");
   146→      setName("");
   147→      setRoomType("");
   148→      setShowForm(false);
   149→      loadRooms();
   150→      showToast("Salle créée avec succès.", "success");
   151→    } catch {
   152→      showToast("Erreur lors de la création de la salle.", "error");
   153→    } finally {
   154→      setSubmitting(false);
   155→    }
   156→  }
   157→
   158→  if (loading) {
   159→    return (
   160→      <div className="card" style={{ textAlign: "center", padding: "32px" }}>
   161→        <div className="spinner" />
   162→        <p className="muted" style={{ marginTop: "12px" }}>Chargement des salles...</p>
   163→      </div>
   164→    );
   165→  }
   166→
   167→  if (error) {
   168→    return <p style={{ color: "crimson" }}>{error}</p>;
   169→  }
   170→
   171→  return (
   172→    <>
   173→      {/* Status summary */}
   174→      <div className="card" style={{ marginBottom: "18px" }}>
   175→        <div style={{ display: "flex", gap: "24px", flexWrap: "wrap", alignItems: "center" }}>
   176→          <span style={{ fontWeight: 700, fontSize: "18px" }}>{rooms.length} salle(s)</span>
   177→          {Object.entries(ROOM_STATUS_CONFIG).map(([status, config]) => (
   178→            <span key={status} style={{ display: "flex", alignItems: "center", gap: "6px" }}>
   179→              <span
   180→                style={{
   181→                  width: "12px",
   182→                  height: "12px",
   183→                  borderRadius: "4px",
   184→                  background: config.color,
   185→                  display: "inline-block",
   186→                }}
   187→              />
   188→              <span style={{ fontSize: "14px", color: "var(--muted)" }}>
   189→                {config.label}: <strong style={{ color: "var(--text)" }}>{statusCounts[status] || 0}</strong>
   190→              </span>
   191→            </span>
   192→          ))}
   193→          <button className="primary-button" style={{ marginLeft: "auto" }} onClick={() => setShowForm(!showForm)}>
   194→            {showForm ? "Annuler" : "Nouvelle salle"}
   195→          </button>
   196→        </div>
   197→      </div>
   198→
   199→      {showForm && (
   200→        <div className="card form-card">
   201→          <h3>Nouvelle salle opératoire</h3>
   202→          <form onSubmit={handleSubmit} className="form-grid">
   203→            <label className="form-control">
   204→              Code
   205→              <input
   206→                type="text"
   207→                value={code}
   208→                onChange={(e) => setCode(e.target.value)}
   209→                placeholder="Code de la salle"
   210→                required
   211→              />
   212→            </label>
   213→            <label className="form-control">
   214→              Nom
   215→              <input
   216→                type="text"
   217→                value={name}
   218→                onChange={(e) => setName(e.target.value)}
   219→                placeholder="Nom de la salle"
   220→                required
   221→              />
   222→            </label>
   223→            <label className="form-control">
   224→              Type
   225→              <input
   226→                type="text"
   227→                value={roomType}
   228→                onChange={(e) => setRoomType(e.target.value)}
   229→                placeholder="Type de salle"
   230→              />
   231→            </label>
   232→            <div className="form-actions">
   233→              <button className="primary-button" type="submit" disabled={submitting}>
   234→                {submitting ? "Création..." : "Créer la salle"}
   235→              </button>
   236→            </div>
   237→          </form>
   238→        </div>
   239→      )}
   240→
   241→      {/* Room grid */}
   242→      <div className="bed-grid" style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(200px, 1fr))", gap: "16px" }}>
   243→        {rooms.map((room) => {
   244→          const config = ROOM_STATUS_CONFIG[room.status] || ROOM_STATUS_CONFIG.AVAILABLE;
   245→          return (
   246→            <div key={room.id} className={`bed-card ${config.cssClass}`}>
   247→              <div className="bed-number" style={{ fontSize: "16px", fontWeight: 700 }}>{room.code}</div>
   248→              <div className="bed-room">{room.name}</div>
   249→              {room.type && <div className="bed-patient" style={{ fontSize: "12px" }}>{room.type}</div>}
   250→              <div style={{ fontSize: "12px", marginTop: "6px", color: config.color, fontWeight: 600 }}>
   251→                {config.label}
   252→              </div>
   253→            </div>
   254→          );
   255→        })}
   256→      </div>
   257→
   258→      {rooms.length === 0 && (
   259→        <div className="card">
   260→          <p className="muted">Aucune salle opératoire configurée.</p>
   261→        </div>
   262→      )}
   263→    </>
   264→  );
   265→}

   267→/* ─── Schedules Tab ──────────────────────────────────────────── */
   268→
   269→function SchedulesTab({ lookups }: { lookups: LookupData }) {
   270→  const [schedules, setSchedules] = useState<Row[]>([]);
   271→  const [rooms, setRooms] = useState<Row[]>([]);
   272→  const [loading, setLoading] = useState(true);
   273→  const [error, setError] = useState("");
   274→  const [showForm, setShowForm] = useState(false);
   275→  const [submitting, setSubmitting] = useState(false);
   276→
   277→  // Form state
   278→  const [patientId, setPatientId] = useState("");
   279→  const [procedureName, setProcedureName] = useState("");
   280→  const [procedureCode, setProcedureCode] = useState("");
   281→  const [laterality, setLaterality] = useState("NOT_APPLICABLE");
   282→  const [urgency, setUrgency] = useState("PLANNED");
   283→  const [operatingRoomId, setOperatingRoomId] = useState("");
   284→  const [surgeonId, setSurgeonId] = useState("");
   285→
   286→  const options = buildOptions(lookups);
   287→
   288→  function getPatientName(patientId: string): string {
   289→    const patient = lookups.patients.find((p) => p.id === patientId);
   290→    if (!patient) return "Inconnu";
   291→    return `${patient.first_name || ""} ${patient.last_name || ""}`.trim() || patient.patient_number || "N/A";
   292→  }
   293→
   294→  const loadSchedules = useCallback(async () => {
   295→    setLoading(true);
   296→    setError("");
   297→    try {
   298→      const payload = await apiRequest<any>("/surgery/schedules?page_size=1000");
   299→      setSchedules(Array.isArray(payload.data) ? payload.data : []);
   300→    } catch {
   301→      setError("Impossible de charger les programmations.");
   302→    } finally {
   303→      setLoading(false);
   304→    }
   305→  }, []);
   306→
   307→  const loadRooms = useCallback(async () => {
   308→    try {
   309→      const payload = await apiRequest<any>("/surgery/rooms?page_size=1000");
   310→      setRooms(Array.isArray(payload.data) ? payload.data : []);
   311→    } catch {
   312→      // Silently fail
   313→    }
   314→  }, []);
   315→
   316→  useEffect(() => {
   317→    loadSchedules();
   318→    const handler = () => loadSchedules();
   319→    window.addEventListener("refresh-resource", handler);
   320→    return () => window.removeEventListener("refresh-resource", handler);
   321→  }, [loadSchedules]);
   322→
   323→  useEffect(() => {
   324→    if (showForm) {
   325→      loadRooms();
   326→    }
   327→  }, [showForm, loadRooms]);
   328→
   329→  async function handleSubmit(e: React.FormEvent) {
   330→    e.preventDefault();
   331→    if (!patientId || !procedureName) return;
   332→    setSubmitting(true);
   333→    try {
   334→      await apiRequest("/surgery/schedules", {
   335→        method: "POST",
   336→        body: JSON.stringify({
   337→          patient_id: patientId,
   338→          procedure_name: procedureName.trim(),
   339→          procedure_code: procedureCode.trim() || undefined,
   340→          laterality,
   341→          urgency,
   342→          operating_room_id: operatingRoomId || undefined,
   343→          surgeon_id: surgeonId || undefined,
   344→        }),
   345→      });
   346→      setPatientId("");
   347→      setProcedureName("");
   348→      setProcedureCode("");
   349→      setLaterality("NOT_APPLICABLE");
   350→      setUrgency("PLANNED");
   351→      setOperatingRoomId("");
   352→      setSurgeonId("");
   353→      setShowForm(false);
   354→      loadSchedules();
   355→      showToast("Programmation créée avec succès.", "success");
   356→    } catch {
   357→      showToast("Erreur lors de la création de la programmation.", "error");
   358→    } finally {
   359→      setSubmitting(false);
   360→    }
   361→  }
   362→
   363→  async function handleStart(scheduleId: string) {
   364→    try {
   365→      await apiRequest(`/surgery/schedules/${scheduleId}/start`, { method: "POST" });
   366→      loadSchedules();
   367→      showToast("Intervention démarrée.", "success");
   368→    } catch {
   369→      showToast("Erreur lors du démarrage.", "error");
   370→    }
   371→  }
   372→
   373→  async function handleComplete(scheduleId: string) {
   374→    try {
   375→      await apiRequest(`/surgery/schedules/${scheduleId}/complete`, { method: "POST" });
   376→      loadSchedules();
   377→      showToast("Intervention terminée.", "success");
   378→    } catch {
   379→      showToast("Erreur lors de la complétion.", "error");
   380→    }
   381→  }
   382→
   383→  async function handleCancel(scheduleId: string) {
   384→    if (!confirm("Confirmer l'annulation de cette intervention ?")) return;
   385→    try {
   386→      await apiRequest(`/surgery/schedules/${scheduleId}/cancel`, { method: "POST" });
   387→      loadSchedules();
   388→      showToast("Intervention annulée.", "success");
   389→    } catch {
   390→      showToast("Erreur lors de l'annulation.", "error");
   391→    }
   392→  }
   393→
   394→  const roomOptions = rooms.map((r) => ({
   395→    value: r.id,
   396→    label: `${r.code} — ${r.name}`,
   397→  }));
   398→
   399→  const URGENCY_LABEL: Record<string, string> = {
   400→    PLANNED: "Planifié",
   401→    URGENT: "Urgent",
   402→    EMERGENCY: "Urgence",
   403→  };
   404→
   405→  return (
   406→    <>
   407→      <div className="section-header">
   408→        <h2 style={{ margin: 0 }}>Programmation chirurgicale</h2>
   409→        <button className="primary-button" onClick={() => setShowForm(!showForm)}>
   410→          {showForm ? "Annuler" : "Nouvelle programmation"}
   411→        </button>
   412→      </div>
   413→
   414→      {showForm && (
   415→        <div className="card form-card">
   416→          <h3>Nouvelle programmation</h3>
   417→          <form onSubmit={handleSubmit} className="form-grid">
   418→            <label className="form-control">
   419→              Patient
   420→              <select value={patientId} onChange={(e) => setPatientId(e.target.value)} required>
   421→                <option value="">-- Choisir un patient --</option>
   422→                {options.patients.map((o) => (
   423→                  <option key={o.value} value={o.value}>{o.label}</option>
   424→                ))}
   425→              </select>
   426→            </label>
   427→            <label className="form-control">
   428→              Intervention
   429→              <input
   430→                type="text"
   431→                value={procedureName}
   432→                onChange={(e) => setProcedureName(e.target.value)}
   433→                placeholder="Nom de l'intervention"
   434→                required
   435→              />
   436→            </label>
   437→            <label className="form-control">
   438→              Code intervention
   439→              <input
   440→                type="text"
   441→                value={procedureCode}
   442→                onChange={(e) => setProcedureCode(e.target.value)}
   443→                placeholder="Code (optionnel)"
   444→              />
   445→            </label>
   446→            <label className="form-control">
   447→              Latéralité
   448→              <select value={laterality} onChange={(e) => setLaterality(e.target.value)}>
   449→                {LATERALITY_OPTIONS.map((l) => (
   450→                  <option key={l.value} value={l.value}>{l.label}</option>
   451→                ))}
   452→              </select>
   453→            </label>
   454→            <label className="form-control">
   455→              Urgence
   456→              <select value={urgency} onChange={(e) => setUrgency(e.target.value)}>
   457→                {SURGICAL_URGENCY_OPTIONS.map((u) => (
   458→                  <option key={u.value} value={u.value}>{u.label}</option>
   459→                ))}
   460→              </select>
   461→            </label>
   462→            <label className="form-control">
   463→              Salle
   464→              <select value={operatingRoomId} onChange={(e) => setOperatingRoomId(e.target.value)}>
   465→                <option value="">-- Choisir une salle --</option>
   466→                {roomOptions.map((o) => (
   467→                  <option key={o.value} value={o.value}>{o.label}</option>
   468→                ))}
   469→              </select>
   470→            </label>
   471→            <label className="form-control">
   472→              Chirurgien
   473→              <select value={surgeonId} onChange={(e) => setSurgeonId(e.target.value)}>
   474→                <option value="">-- Choisir un chirurgien --</option>
   475→                {options.staff.map((o) => (
   476→                  <option key={o.value} value={o.value}>{o.label}</option>
   477→                ))}
   478→              </select>
   479→            </label>
   480→            <div className="form-actions">
   481→              <button className="primary-button" type="submit" disabled={submitting}>
   482→                {submitting ? "Création..." : "Programmer"}
   483→              </button>
   484→            </div>
   485→          </form>
   486→        </div>
   487→      )}
   488→
   489→      {loading ? (
   490→        <div className="card" style={{ textAlign: "center", padding: "32px" }}>
   491→          <div className="spinner" />
   492→        </div>
   493→      ) : error ? (
   494→        <p style={{ color: "crimson" }}>{error}</p>
   495→      ) : schedules.length === 0 ? (
   496→        <div className="card">
   497→          <p className="muted">Aucune programmation trouvée.</p>
   498→        </div>
   499→      ) : (
   500→        <div className="card">
   501→          <div className="table-wrapper">
   502→            <table className="table">
   503→              <thead>
   504→                <tr>
   505→                  <th>Date prévue</th>
   506→                  <th>Patient</th>
   507→                  <th>Intervention</th>
   508→                  <th>Chirurgien</th>
   509→                  <th>Salle</th>
   510→                  <th>Urgence</th>
   511→                  <th>Statut</th>
   512→                  <th>Actions</th>
   513→                </tr>
   514→              </thead>
   515→              <tbody>
   516→                {schedules.map((schedule) => (
   517→                  <tr key={schedule.id}>
   518→                    <td style={{ whiteSpace: "nowrap" }}>
   519→                      {schedule.scheduled_date ? new Date(schedule.scheduled_date).toLocaleDateString("fr-FR") : "—"}
   520→                    </td>
   521→                    <td style={{ fontWeight: 600 }}>{getPatientName(schedule.patient_id)}</td>
   522→                    <td>{schedule.procedure_name || "—"}</td>
   523→                    <td>{schedule.surgeon_id || "—"}</td>
   524→                    <td>{schedule.operating_room_id || "—"}</td>
   525→                    <td>
   526→                      <span className={`badge ${schedule.urgency === "EMERGENCY" ? "badge-red" : schedule.urgency === "URGENT" ? "badge-yellow" : "badge-gray"}`}>
   527→                        {URGENCY_LABEL[schedule.urgency] || schedule.urgency || "—"}
   528→                      </span>
   529→                    </td>
   530→                    <td>
   531→                      <span className={`badge ${SCHEDULE_STATUS_BADGE[schedule.status] || "badge-gray"}`}>
   532→                        {SCHEDULE_STATUS_LABEL[schedule.status] || schedule.status}
   533→                      </span>
   534→                    </td>
   535→                    <td>
   536→                      {schedule.status === "PLANNED" && (
   537→                        <button
   538→                          className="secondary-button"
   539→                          style={{ padding: "6px 14px", fontSize: "13px" }}
   540→                          onClick={() => handleStart(schedule.id)}
   541→                        >
   542→                          Démarrer
   543→                        </button>
   544→                      )}
   545→                      {schedule.status === "IN_PROGRESS" && (
   546→                        <button
   547→                          className="secondary-button"
   548→                          style={{ padding: "6px 14px", fontSize: "13px" }}
   549→                          onClick={() => handleComplete(schedule.id)}
   550→                        >
   551→                          Terminer
   552→                        </button>
   553→                      )}
   554→                      {(schedule.status === "PLANNED" || schedule.status === "IN_PROGRESS") && (
   555→                        <button
   556→                          className="secondary-button"
   557→                          style={{ padding: "6px 14px", fontSize: "13px", marginLeft: "4px", color: "crimson" }}
   558→                          onClick={() => handleCancel(schedule.id)}
   559→                        >
   560→                          Annuler
   561→                        </button>
   562→                      )}
   563→                    </td>
   564→                  </tr>
   565→                ))}
   566→              </tbody>
   567→            </table>
   568→          </div>
   569→        </div>
   570→      )}

   575→/* ─── Reports Tab ──────────────────────────────────────────── */
   576→
   577→function ReportsTab({ lookups }: { lookups: LookupData }) {
   578→  const [reports, setReports] = useState<Row[]>([]);
   579→  const [completedSchedules, setCompletedSchedules] = useState<Row[]>([]);
   580→  const [loading, setLoading] = useState(true);
   581→  const [error, setError] = useState("");
   582→  const [showForm, setShowForm] = useState(false);
   583→  const [submitting, setSubmitting] = useState(false);
   584→
   585→  // Form state
   586→  const [scheduleId, setScheduleId] = useState("");
   587→  const [operativeFindings, setOperativeFindings] = useState("");
   588→  const [procedurePerformed, setProcedurePerformed] = useState("");
   589→  const [complications, setComplications] = useState("");
   590→  const [specimens, setSpecimens] = useState("");
   591→  const [bloodLoss, setBloodLoss] = useState("");
   592→  const [anesthesiaType, setAnesthesiaType] = useState("GENERAL");
   593→
   594→  function getPatientName(patientId: string): string {
   595→    const patient = lookups.patients.find((p) => p.id === patientId);
   596→    if (!patient) return "Inconnu";
   597→    return `${patient.first_name || ""} ${patient.last_name || ""}`.trim() || patient.patient_number || "N/A";
   598→  }
   599→
   600→  const loadReports = useCallback(async () => {
   601→    setLoading(true);
   602→    setError("");
   603→    try {
   604→      const payload = await apiRequest<any>("/surgery/reports?page_size=1000");
   605→      setReports(Array.isArray(payload.data) ? payload.data : []);
   606→    } catch {
   607→      setError("Impossible de charger les comptes rendus.");
   608→    } finally {
   609→      setLoading(false);
   610→    }
   611→  }, []);
   612→
   613→  const loadCompletedSchedules = useCallback(async () => {
   614→    try {
   615→      const payload = await apiRequest<any>("/surgery/schedules?status=COMPLETED&page_size=1000");
   616→      setCompletedSchedules(Array.isArray(payload.data) ? payload.data : []);
   617→    } catch {
   618→      // Silently fail
   619→    }
   620→  }, []);
   621→
   622→  useEffect(() => {
   623→    loadReports();
   624→    const handler = () => loadReports();
   625→    window.addEventListener("refresh-resource", handler);
   626→    return () => window.removeEventListener("refresh-resource", handler);
   627→  }, [loadReports]);
   628→
   629→  useEffect(() => {
   630→    if (showForm) {
   631→      loadCompletedSchedules();
   632→    }
   633→  }, [showForm, loadCompletedSchedules]);
   634→
   635→  async function handleSubmit(e: React.FormEvent) {
   636→    e.preventDefault();
   637→    if (!scheduleId) return;
   638→    setSubmitting(true);
   639→    try {
   640→      await apiRequest("/surgery/reports", {
   641→        method: "POST",
   642→        body: JSON.stringify({
   643→          schedule_id: scheduleId,
   644→          operative_findings: operativeFindings.trim() || undefined,
   645→          procedure_performed: procedurePerformed.trim() || undefined,
   646→          complications: complications.trim() || undefined,
   647→          specimens: specimens.trim() || undefined,
   648→          blood_loss: bloodLoss.trim() || undefined,
   649→          anesthesia_type: anesthesiaType,
   650→        }),
   651→      });
   652→      setScheduleId("");
   653→      setOperativeFindings("");
   654→      setProcedurePerformed("");
   655→      setComplications("");
   656→      setSpecimens("");
   657→      setBloodLoss("");
   658→      setAnesthesiaType("GENERAL");
   659→      setShowForm(false);
   660→      loadReports();
   661→      showToast("Compte rendu créé avec succès.", "success");
   662→    } catch {
   663→      showToast("Erreur lors de la création du compte rendu.", "error");
   664→    } finally {
   665→      setSubmitting(false);
   666→    }
   667→  }
   668→
   669→  async function handleValidate(reportId: string) {
   670→    try {
   671→      await apiRequest(`/surgery/reports/${reportId}/validate`, { method: "POST" });
   672→      loadReports();
   673→      showToast("Compte rendu validé.", "success");
   674→    } catch {
   675→      showToast("Erreur lors de la validation.", "error");
   676→    }
   677→  }
   678→
   679→  const scheduleOptions = completedSchedules.map((s) => ({
   680→    value: s.id,
   681→    label: `${s.procedure_name || "Intervention"} — ${getPatientName(s.patient_id)}`,
   682→  }));
   683→
   684→  const ANESTHESIA_LABEL: Record<string, string> = {
   685→    GENERAL: "Générale",
   686→    REGIONAL: "Régionale",
   687→    LOCAL: "Locale",
   688→  };
   689→
   690→  return (
   691→    <>
   692→      <div className="section-header">
   693→        <h2 style={{ margin: 0 }}>Comptes rendus opératoires</h2>
   694→        <button className="primary-button" onClick={() => setShowForm(!showForm)}>
   695→          {showForm ? "Annuler" : "Nouveau compte rendu"}
   696→        </button>
   697→      </div>
   698→
   699→      {showForm && (
   700→        <div className="card form-card">
   701→          <h3>Nouveau compte rendu opératoire</h3>
   702→          <form onSubmit={handleSubmit} className="form-grid">
   703→            <label className="form-control">
   704→              Intervention
   705→              <select value={scheduleId} onChange={(e) => setScheduleId(e.target.value)} required>
   706→                <option value="">-- Choisir une intervention terminée --</option>
   707→                {scheduleOptions.map((o) => (
   708→                  <option key={o.value} value={o.value}>{o.label}</option>
   709→                ))}
   710→              </select>
   711→            </label>
   712→            <label className="form-control">
   713→              Type d'anesthésie
   714→              <select value={anesthesiaType} onChange={(e) => setAnesthesiaType(e.target.value)}>
   715→                {ANESTHESIA_OPTIONS.map((a) => (
   716→                  <option key={a.value} value={a.value}>{a.label}</option>
   717→                ))}
   718→              </select>
   719→            </label>
   720→            <label className="form-control">
   721→              Perte sanguine (ml)
   722→              <input
   723→                type="text"
   724→                value={bloodLoss}
   725→                onChange={(e) => setBloodLoss(e.target.value)}
   726→                placeholder="Volume en ml"
   727→              />
   728→            </label>
   729→            <label className="form-control" style={{ gridColumn: "1 / -1" }}>
   730→              Constatations per-opératoires
   731→              <textarea
   732→                value={operativeFindings}
   733→                onChange={(e) => setOperativeFindings(e.target.value)}
   734→                placeholder="Constatations..."
   735→                rows={3}
   736→              />
   737→            </label>
   738→            <label className="form-control" style={{ gridColumn: "1 / -1" }}>
   739→              Gestes réalisés
   740→              <textarea
   741→                value={procedurePerformed}
   742→                onChange={(e) => setProcedurePerformed(e.target.value)}
   743→                placeholder="Description des gestes..."
   744→                rows={2}
   745→              />
   746→            </label>
   747→            <label className="form-control" style={{ gridColumn: "1 / -1" }}>
   748→              Complications
   749→              <textarea
   750→                value={complications}
   751→                onChange={(e) => setComplications(e.target.value)}
   752→                placeholder="Complications éventuelles..."
   753→                rows={2}
   754→              />
   755→            </label>
   756→            <label className="form-control" style={{ gridColumn: "1 / -1" }}>
   757→              Pièces opératoires
   758→              <textarea
   759→                value={specimens}
   760→                onChange={(e) => setSpecimens(e.target.value)}
   761→                placeholder="Pièces envoyées..."
   762→                rows={2}
   763→              />
   764→            </label>
   765→            <div className="form-actions">
   766→              <button className="primary-button" type="submit" disabled={submitting}>
   767→                {submitting ? "Création..." : "Créer le compte rendu"}
   768→              </button>
   769→            </div>
   770→          </form>
   771→        </div>
   772→      )}
   773→
   774→      {loading ? (
   775→        <div className="card" style={{ textAlign: "center", padding: "32px" }}>
   776→          <div className="spinner" />
   777→        </div>
   778→      ) : error ? (
   779→        <p style={{ color: "crimson" }}>{error}</p>
   780→      ) : reports.length === 0 ? (
   781→        <div className="card">
   782→          <p className="muted">Aucun compte rendu trouvé.</p>
   783→        </div>
   784→      ) : (
   785→        <div className="card">
   786→          <div className="table-wrapper">
   787→            <table className="table">
   788→              <thead>
   789→                <tr>
   790→                  <th>Date</th>
   791→                  <th>Patient</th>
   792→                  <th>Intervention</th>
   793→                  <th>Constatations</th>
   794→                  <th>Complications</th>
   795→                  <th>Statut</th>
   796→                  <th>Actions</th>
   797→                </tr>
   798→              </thead>
   799→              <tbody>
   800→                {reports.map((report) => (
   801→                  <tr key={report.id}>
   802→                    <td style={{ whiteSpace: "nowrap" }}>
   803→                      {report.created_at ? new Date(report.created_at).toLocaleDateString("fr-FR") : "—"}
   804→                    </td>
   805→                    <td style={{ fontWeight: 600 }}>{getPatientName(report.patient_id)}</td>
   806→                    <td>{report.procedure_performed || report.procedure_name || "—"}</td>
   807→                    <td>{report.operative_findings || "—"}</td>
   808→                    <td>{report.complications || "—"}</td>
   809→                    <td>
   810→                      <span className={`badge ${REPORT_STATUS_BADGE[report.status] || "badge-gray"}`}>
   811→                        {REPORT_STATUS_LABEL[report.status] || report.status}
   812→                      </span>
   813→                    </td>
   814→                    <td>
   815→                      {report.status === "DRAFT" && (
   816→                        <button
   817→                          className="secondary-button"
   818→                          style={{ padding: "6px 14px", fontSize: "13px" }}
   819→                          onClick={() => handleValidate(report.id)}
   820→                        >
   821→                          Valider
   822→                        </button>
   823→                      )}
   824→                    </td>
   825→                  </tr>
   826→                ))}
   827→              </tbody>
   828→            </table>
   829→          </div>
   830→        </div>
   831→      )}
   832→    </>
   833→  );
   834→}
