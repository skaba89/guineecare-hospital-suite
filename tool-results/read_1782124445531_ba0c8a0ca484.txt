    75→    75→
    76→    76→export function DashboardPage({ lookups }: { lookups: LookupData }) {
    77→    77→  const [stats, setStats] = useState<DashboardStats | null>(null);
    78→    78→  const [loading, setLoading] = useState(true);
    79→    79→  const [recentPatients, setRecentPatients] = useState<Row[]>([]);
    80→    80→  const [recentAdmissions, setRecentAdmissions] = useState<Row[]>([]);
    81→    81→  const [allBeds, setAllBeds] = useState<Row[]>([]);
    82→    82→  const [allAdmissions, setAllAdmissions] = useState<Row[]>([]);
    83→    83→  const navigate = useNavigate();
    84→    84→
    85→    85→  const options = buildOptions(lookups);
    86→    86→  const facilityId = firstValue(options.facilities);
    87→    87→
    88→    88→  const loadStats = useCallback(async () => {
    89→    89→    if (!facilityId) return;
    90→    90→    setLoading(true);
    91→    91→    try {
    92→    92→      const [patients, admissions, emergency, stays, beds, labOrders, imagingOrders, reporting] =
    93→    93→        await Promise.all([
    94→    94→          apiRequest<{ data?: Row[]; total?: number }>("/patients?page_size=1"),
    95→    95→          apiRequest<{ data?: Row[]; total?: number }>("/admissions?page_size=1"),
    96→    96→          apiRequest<{ data?: Row[]; total?: number }>("/emergency/queue?page_size=1"),
    97→    97→          apiRequest<{ data?: Row[]; total?: number }>("/hospitalization/stays?status=ACTIVE&page_size=1"),
    98→    98→          apiRequest<{ data?: Row[] }>(`/hospitalization/bed-board?facility_id=${facilityId}`),
    99→    99→          apiRequest<{ data?: Row[]; total?: number }>("/laboratory/orders?page_size=1000"),
   100→   100→          apiRequest<{ data?: Row[] }>("/imaging/orders?status=PENDING&page_size=1000"),
   101→   101→          apiRequest<{ data?: Record<string, unknown> }>("/reporting/dashboard"),
   102→   102→        ]);
   103→   103→
   104→   104→      const bedsList = Array.isArray(beds.data) ? beds.data : [];
   105→   105→      const availableBeds = bedsList.filter((b: Row) => b.bed_status === "AVAILABLE").length;
   106→   106→      const occupiedBeds = bedsList.filter((b: Row) => b.bed_status === "OCCUPIED").length;
   107→   107→      const outOfServiceBeds = bedsList.filter(
   108→   108→        (b: Row) => b.bed_status === "OUT_OF_SERVICE" || b.bed_status === "MAINTENANCE"
   109→   109→      ).length;
   110→   110→      const labPending = Array.isArray(labOrders.data)
   111→   111→        ? labOrders.data.filter((o: Row) => o.status === "PENDING" || o.status === "IN_PROGRESS").length
   112→   112→        : 0;
   113→   113→      const imagingPending = Array.isArray(imagingOrders.data) ? imagingOrders.data.length : 0;
   114→   114→
   115→   115→      const reportingData = reporting.data || {};
   116→   116→      const activeAlerts = Number(reportingData.active_alerts || 0);
   117→   117→      const draftReports = Number(reportingData.draft_reports || 0);
   118→   118→
   119→   119→      setStats({
   120→   120→        patients: typeof patients.total === "number" ? patients.total : (patients.data?.length || 0),
   121→   121→        admissions: typeof admissions.total === "number" ? admissions.total : (admissions.data?.length || 0),
   122→   122→        emergencyVisits: typeof emergency.total === "number" ? emergency.total : (emergency.data?.length || 0),
   123→   123→        activeStays: typeof stays.total === "number" ? stays.total : (stays.data?.length || 0),
   124→   124→        availableBeds,
   125→   125→        totalBeds: bedsList.length,
   126→   126→        occupiedBeds,
   127→   127→        outOfServiceBeds,
   128→   128→        pendingLabOrders: labPending,
   129→   129→        surgeryScheduled: 0,
   130→   130→        pendingImaging: imagingPending,
   131→   131→        activeAlerts,
   132→   132→        draftReports,
   133→   133→        openIncidents: 0,
   134→   134→      });
   135→   135→
   136→   136→      setAllBeds(bedsList);
   137→   137→
   138→   138→      // For recent lists we need real patient records, not just totals
   139→   139→      const [recentPatientsRes, recentAdmissionsRes] = await Promise.all([
   140→   140→        apiRequest<{ data?: Row[] }>("/patients?page_size=5"),
   141→   141→        apiRequest<{ data?: Row[] }>("/admissions?page_size=5"),
   142→   142→      ]);
   143→   143→      setRecentPatients(Array.isArray(recentPatientsRes.data) ? recentPatientsRes.data.slice(0, 5) : []);
   144→   144→      const admissionsList = Array.isArray(recentAdmissionsRes.data) ? recentAdmissionsRes.data : [];
   145→   145→      setRecentAdmissions(admissionsList.slice(0, 5));
   146→   146→
   147→   147→      // For the chart we need a fuller set of admissions
   148→   148→      const allAdmRes = await apiRequest<{ data?: Row[] }>("/admissions?page_size=1000");
   149→   149→      setAllAdmissions(Array.isArray(allAdmRes.data) ? allAdmRes.data : []);
   150→   150→    } catch {
   151→   151→      // Silently fail — dashboard should remain usable
   152→   152→    } finally {
   153→   153→      setLoading(false);
   154→   154→    }
   155→   155→  }, [facilityId]);
   156→   156→
   157→   157→  useEffect(() => {
   158→   158→    loadStats();
   159→   159→    const handler = () => loadStats();
   160→   160→    window.addEventListener("refresh-resource", handler);
   161→   161→    return () => window.removeEventListener("refresh-resource", handler);
   162→   162→  }, [loadStats]);
   163→   163→
   164→   164→  // v1.3.0 — Realtime KPI push: increment counters live when a KPI event arrives.
   165→   165→  // We don't refetch the whole dashboard — we just bump the relevant counter
   166→   166→  // by the event's `delta`. A subsequent full refresh (manual or via the
   167→   167→  // `refresh-resource` event) reconciles any drift.
   168→   168→  const { lastEvent: liveKpi } = useRealtimeKPIs({ typePrefix: "kpi." });
   169→   169→  useEffect(() => {
   170→   170→    if (!liveKpi || !stats) return;
   171→   171→    const { type, payload } = liveKpi;
   172→   172→    if (type === "kpi.admissions.today.count" && typeof payload.delta === "number") {
   173→   173→      setStats((prev) => prev ? { ...prev, admissions: prev.admissions + (payload.delta as number) } : prev);
   174→   174→    } else if (type === "kpi.lab.results.validated.count" && typeof payload.delta === "number") {
   175→   175→      setStats((prev) => prev ? { ...prev, pendingLabOrders: Math.max(0, prev.pendingLabOrders - (payload.delta as number)) } : prev);
   176→   176→    }
   177→   177→  }, [liveKpi, stats]);
   178→   178→

   184→   184→        if (a.admitted_at) {
   185→   185→          const day = new Date(a.admitted_at).getDay();
   186→   186→          const adjustedDay = day === 0 ? 6 : day - 1;
   187→   187→          dayMap[adjustedDay] = (dayMap[adjustedDay] || 0) + 1;
   188→   188→        }
   189→   189→      });
   190→   190→      return DAYS_FR.map((name, i) => ({
   191→   191→        name,
   192→   192→        admissions: dayMap[i] || 0,
   193→   193→      }));
   194→   194→    }
   195→   195→    return DAYS_FR.map((name) => ({
   196→   196→      name,
   197→   197→      admissions: Math.floor(Math.random() * 12) + 3,
   198→   198→    }));
   199→   199→  }, [allAdmissions]);
   200→   200→
   201→   201→  // Bed occupancy pie data
   202→   202→  const bedPieData = useMemo(() => {

   204→   204→      return [
   205→   205→        { name: "Disponibles", value: stats.availableBeds },
   206→   206→        { name: "Occupés", value: stats.occupiedBeds },
   207→   207→        { name: "Hors service", value: stats.outOfServiceBeds },
   208→   208→      ];
   209→   209→    }
   210→   210→    return [
   211→   211→      { name: "Disponibles", value: 45 },
   212→   212→      { name: "Occupés", value: 120 },
   213→   213→      { name: "Hors service", value: 8 },
   214→   214→    ];
   215→   215→  }, [stats]);

   223→   223→        type: "admission",
   224→   224→        patientName: a.patient_id || "Patient",
   225→   225→        department: a.department_id || "Service",
   226→   226→        timestamp: a.admitted_at || new Date().toISOString(),
   227→   227→      });
   228→   228→    });
   229→   229→    // Add mock discharge/transfer to show variety
   230→   230→    const mockEvents: PatientFlowEvent[] = [
   231→   231→      {
   232→   232→        id: "mock-d1",
   233→   233→        type: "discharge",
   234→   234→        patientName: "Diallo M.",
   235→   235→        department: "Médecine Interne",
   236→   236→        timestamp: new Date(Date.now() - 3600000).toISOString(),
   237→   237→      },
   238→   238→      {
   239→   239→        id: "mock-t1",
   240→   240→        type: "transfer",
   241→   241→        patientName: "Camara F.",
   242→   242→        department: "Réanimation → Chirurgie",
   243→   243→        timestamp: new Date(Date.now() - 7200000).toISOString(),
   244→   244→      },
   245→   245→      {
   246→   246→        id: "mock-e1",
   247→   247→        type: "emergency",
   248→   248→        patientName: "Touré A.",
   249→   249→        department: "Urgences",
   250→   250→        timestamp: new Date(Date.now() - 900000).toISOString(),
   251→   251→      },
   252→   252→    ];
   253→   253→    return [...events, ...mockEvents].sort(
   254→   254→      (a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime()
   255→   255→    );
   256→   256→  }, [recentAdmissions]);
   257→   257→
   258→   258→  // Priority alerts from real data
   259→   259→  const priorityAlerts: PriorityAlert[] = useMemo(() => {
   260→   260→    const alerts: PriorityAlert[] = [];
   261→   261→    if (stats && stats.emergencyVisits > 0) {
   262→   262→      alerts.push({
   263→   263→        id: "alert-er",
   264→   264→        severity: "critical",
   265→   265→        title: `${stats.emergencyVisits} patient(s) en urgence`,
   266→   266→        description: "Patients nécessitant une prise en charge immédiate",
   267→   267→        time: "Maintenant",
   268→   268→      });
   269→   269→    }
   270→   270→    if (stats && stats.pendingLabOrders > 3) {
   271→   271→      alerts.push({
   272→   272→        id: "alert-lab",
   273→   273→        severity: "high",
   274→   274→        title: "Examens en attente",
   275→   275→        description: `${stats.pendingLabOrders} résultats de laboratoire en attente`,
   276→   276→        time: "Il y a 15 min",
   277→   277→      });
   278→   278→    }
   279→   279→    if (stats && stats.activeAlerts > 0) {
   280→   280→      alerts.push({
   281→   281→        id: "alert-epi",
   282→   282→        severity: "warning",
   283→   283→        title: "Alertes épidémiologiques",
   284→   284→        description: `${stats.activeAlerts} alerte(s) active(s) détectée(s)`,
   285→   285→        time: "Il y a 1h",
   286→   286→      });
   287→   287→    }
   288→   288→    // Add mock low stock alert if none
   289→   289→    if (alerts.length < 2) {
   290→   290→      alerts.push({
   291→   291→        id: "alert-stock",
   292→   292→        severity: "high",
   293→   293→        title: "Stock faible — Paracétamol",
   294→   294→        description: "Seuil minimal atteint, commande urgente requise",
   295→   295→        time: "Il y a 30 min",
   296→   296→      });
   297→   297→      alerts.push({
   298→   298→        id: "alert-bed",
   299→   299→        severity: "warning",
   300→   300→        title: "Occupation lits élevée",
   301→   301→        description: "Taux d'occupation supérieur à 85%",
   302→   302→        time: "Il y a 2h",
   303→   303→      });
   304→   304→    }
   305→   305→    return alerts;
   306→   306→  }, [stats]);
   307→   307→
   308→   308→  const bedOccupancy =
   309→   309→    stats && stats.totalBeds > 0
   310→   310→      ? Math.round(((stats.totalBeds - stats.availableBeds) / stats.totalBeds) * 100)
   311→   311→      : 0;
   312→   312→
   313→   313→  if (loading && !stats) {
   314→   314→    return (
   315→   315→      <section>
   316→   316→        <div className="dashboard-header">
   317→   317→          <div>
   318→   318→            <h1 className="dashboard-title">Tableau de bord</h1>
   319→   319→            <p className="muted">Vue d'ensemble de l'activité hospitalière</p>
   320→   320→          </div>
   321→   321→        </div>
   322→   322→        <div className="card" style={{ textAlign: "center", padding: "48px" }}>
   323→   323→          <div className="spinner" />
   324→   324→          <p className="muted" style={{ marginTop: "16px" }}>
   325→   325→            Chargement des statistiques...
   326→   326→          </p>
   327→   327→        </div>
   328→   328→      </section>
   329→   329→    );
   330→   330→  }
   331→   331→
   332→   332→  return (
   333→   333→    <section>
   334→   334→      {/* Header */}
   335→   335→      <div className="dashboard-header">
   336→   336→        <div>
   337→   337→          <h1 className="dashboard-title">Tableau de bord</h1>
   338→   338→          <p className="muted">Vue d'ensemble de l'activité hospitalière en temps réel</p>
   339→   339→        </div>
   340→   340→        <div className="dashboard-header-actions">
   341→   341→          <span className="dashboard-timestamp">
   342→   342→            <Clock size={14} />
   343→   343→            {new Date().toLocaleDateString("fr-FR", {
   344→   344→              weekday: "long",
   345→   345→              year: "numeric",
   346→   346→              month: "long",
   347→   347→              day: "numeric",
   348→   348→            })}
   349→   349→          </span>
   350→   350→        </div>
   351→   351→      </div>
   352→   352→
   353→   353→      {/* Row 1 — 8 KPI Cards */}
   354→   354→      <div className="kpi-grid">
   355→   355→        <KpiCard
   356→   356→          title="Patients"
   357→   357→          value={stats?.patients ?? 0}
   358→   358→          icon={Users}
   359→   359→          color="#2563eb"

   361→   361→          onClick={() => navigate("/patients")}
   362→   362→        />
   363→   363→        <KpiCard
   364→   364→          title="Admissions"
   365→   365→          value={stats?.admissions ?? 0}
   366→   366→          icon={ClipboardList}
   367→   367→          color="#7c3aed"
   368→   368→          bgColor="#f5f3ff"
   369→   369→          onClick={() => navigate("/admissions")}
   370→   370→        />
   371→   371→        <KpiCard
   372→   372→          title="Urgences"
   373→   373→          value={stats?.emergencyVisits ?? 0}
   374→   374→          icon={AlertTriangle}
   375→   375→          color="#dc2626"
   376→   376→          bgColor="#fef2f2"
   377→   377→          onClick={() => navigate("/emergency")}
   378→   378→        />
   379→   379→        <KpiCard
   380→   380→          title="Hospitalisés"
   381→   381→          value={stats?.activeStays ?? 0}
   382→   382→          icon={BedDouble}
   383→   383→          color="#0f6b3e"
   384→   384→          bgColor="#e8f5ee"
   385→   385→          onClick={() => navigate("/hospitalization")}
   386→   386→        />
   387→   387→        <KpiCard
   388→   388→          title="Lits dispo."
   389→   389→          value={`${stats?.availableBeds ?? 0}/${stats?.totalBeds ?? 0}`}
   390→   390→          icon={Bed}
   391→   391→          color={bedOccupancy > 85 ? "#dc2626" : "#0f6b3e"}
   392→   392→          bgColor={bedOccupancy > 85 ? "#fef2f2" : "#e8f5ee"}
   393→   393→          subtitle={stats && stats.totalBeds > 0 ? `Occupation : ${bedOccupancy}%` : undefined}
   394→   394→          onClick={() => navigate("/hospitalization")}
   395→   395→        />
   396→   396→        <KpiCard
   397→   397→          title="Examens attente"
   398→   398→          value={(stats?.pendingLabOrders ?? 0) + (stats?.pendingImaging ?? 0)}
   399→   399→          icon={TestTube2}
   400→   400→          color="#d97706"
   401→   401→          bgColor="#fffbeb"
   402→   402→          onClick={() => navigate("/lab")}
   403→   403→        />
   404→   404→        <KpiCard
   405→   405→          title="Alertes"
   406→   406→          value={stats?.activeAlerts ?? 0}
   407→   407→          icon={AlertCircle}
   408→   408→          color={stats && stats.activeAlerts > 0 ? "#dc2626" : "#0f6b3e"}
   409→   409→          bgColor={stats && stats.activeAlerts > 0 ? "#fef2f2" : "#e8f5ee"}
   410→   410→        />
   411→   411→        <KpiCard
   412→   412→          title="Rapports"
   413→   413→          value={stats?.draftReports ?? 0}
   414→   414→          icon={FileText}
   415→   415→          color="#6366f1"
   416→   416→          bgColor="#eef2ff"
   417→   417→          onClick={() => navigate("/reporting")}
   418→   418→        />
   419→   419→      </div>
   420→   420→
   421→   421→      {/* Row 2 — Charts */}
   422→   422→      <div className="dashboard-charts-row">
   423→   423→        <div className="card chart-container">
   424→   424→          <div className="chart-header">
   425→   425→            <h3 className="chart-title">
   426→   426→              <TrendingUp size={16} />
   427→   427→              Admissions — 7 derniers jours
   428→   428→            </h3>
   429→   429→          </div>
   430→   430→          <div className="chart-body">
   431→   431→            <ResponsiveContainer width="100%" height={260}>
   432→   432→              <BarChart data={admissionChartData} margin={{ top: 10, right: 20, left: 0, bottom: 0 }}>
   433→   433→                <CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" vertical={false} />
   434→   434→                <XAxis dataKey="name" tick={{ fontSize: 12, fill: "#64748b" }} axisLine={false} tickLine={false} />
   435→   435→                <YAxis tick={{ fontSize: 12, fill: "#64748b" }} axisLine={false} tickLine={false} />
   436→   436→                <Tooltip
   437→   437→                  contentStyle={{
   438→   438→                    backgroundColor: "#fff",
   439→   439→                    border: "1px solid #e2e8f0",
   440→   440→                    borderRadius: "8px",
   441→   441→                    fontSize: "13px",
   442→   442→                    boxShadow: "0 4px 12px rgba(0,0,0,0.08)",
   443→   443→                  }}
   444→   444→                />
