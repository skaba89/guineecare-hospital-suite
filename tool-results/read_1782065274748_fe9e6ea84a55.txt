   125→
   126→function StockTab({ lookups }: { lookups: LookupData }) {
   127→  const [stock, setStock] = useState<Row[]>([]);
   128→  const [products, setProducts] = useState<Row[]>([]);
   129→  const [loading, setLoading] = useState(true);
   130→  const [search, setSearch] = useState("");
   131→  const [categoryFilter, setCategoryFilter] = useState("");
   132→
   133→  const options = buildOptions(lookups);
   134→  const facilityId = firstValue(options.facilities);
   135→
   136→  const loadData = useCallback(async () => {
   137→    setLoading(true);
   138→    try {
   139→      const [stockRes, prodRes] = await Promise.all([
   140→        apiRequest<any>("/pharmacy/stock?page_size=1000"),
   141→        apiRequest<any>("/pharmacy/products?page_size=1000"),

   143→      setStock(Array.isArray(stockRes.data) ? stockRes.data : []);
   144→      setProducts(Array.isArray(prodRes.data) ? prodRes.data : []);
   145→    } catch {
   146→      showToast("Erreur de chargement des stocks.", "error");
   147→    } finally {
   148→      setLoading(false);
   149→    }
   150→  }, []);
   151→
   152→  useEffect(() => {
   153→    loadData();
   154→    const handler = () => loadData();
   155→    window.addEventListener("refresh-resource", handler);
   156→    return () => window.removeEventListener("refresh-resource", handler);
   157→  }, [loadData]);
   158→
   159→  // Build enriched stock items by merging stock with product info
   160→  const enrichedStock: Row[] = stock.map((s) => {
   161→    const prod = products.find((p: Row) => p.id === s.product_id) || {};
   162→    return { ...s, ...prod, product_name: prod.name, product_code: prod.code };
   163→  });
   164→
   165→  // Apply filters
   166→  const filtered = enrichedStock.filter((item: Row) => {
   167→    const matchSearch =
   168→      !search ||
   169→      (item.product_name || "")
   170→        .toLowerCase()
   171→        .includes(search.toLowerCase()) ||
   172→      (item.product_code || "")
   173→        .toLowerCase()
   174→        .includes(search.toLowerCase());
   175→    const matchCategory =
   176→      !categoryFilter || item.category === categoryFilter;
   177→    return matchSearch && matchCategory;
   178→  });
   179→
   180→  // KPIs
   181→  const totalProducts = products.length;
   182→  const totalUnits = stock.reduce(
   183→    (sum, s) => sum + (s.quantity_available || 0),
   184→    0
   185→  );
   186→  const lowStockCount = stock.filter(
   187→    (s) =>
   188→      s.quantity_available > 0 &&
   189→      s.quantity_available <= (s.min_threshold || 0)
   190→  ).length;

   192→    (s) => s.quantity_available === 0
   193→  ).length;
   194→  const alertCount = lowStockCount + emptyStockCount;
   195→  const estimatedValue = stock.reduce(
   196→    (sum, s) => sum + (s.quantity_available || 0) * 1500,
   197→    0
   198→  );

   202→    const threshold = Number(item.min_threshold) || 0;
   203→    if (qty === 0) return "epuise";
   204→    if (qty <= threshold) return "bas";
   205→    return "ok";
   206→  }
   207→
   208→  if (loading) {
   209→    return (

   211→        <div className="spinner" />
   212→        <p className="muted" style={{ marginTop: "12px" }}>
   213→          Chargement des stocks...
   214→        </p>

   216→    );
   217→  }
   218→
   219→  return (

   223→        <div className="kpi-card">

   225→            className="kpi-card-icon"

   229→          </div>
   230→          <div className="kpi-card-content">
   231→            <div className="kpi-card-value">{totalProducts}</div>
   232→            <div className="kpi-card-title">Total produits</div>
   233→          </div>
   234→        </div>
   235→        <div className="kpi-card">
   236→          <div
   237→            className="kpi-card-icon"
   238→            style={{ background: "var(--info-light)", color: "var(--info)" }}
   239→          >
   240→            <Pill size={22} />
   241→          </div>
   242→          <div className="kpi-card-content">
   243→            <div className="kpi-card-value">{totalUnits.toLocaleString("fr-FR")}</div>
   244→            <div className="kpi-card-title">Stock total unités</div>
   245→          </div>
   246→        </div>
   247→        <div className="kpi-card">
   248→          <div
   249→            className="kpi-card-icon"
   250→            style={{ background: "var(--danger-light)", color: "var(--danger)" }}
   251→          >
   252→            <AlertTriangle size={22} />
   253→          </div>
   254→          <div className="kpi-card-content">
   255→            <div className="kpi-card-value" style={{ color: "var(--danger)" }}>
   256→              {alertCount}
   257→            </div>
   258→            <div className="kpi-card-title">Alertes stock bas</div>
   259→          </div>
   260→        </div>
   261→        <div className="kpi-card">
   262→          <div
   263→            className="kpi-card-icon"
   264→            style={{ background: "var(--success-light)", color: "var(--success)" }}
   265→          >
   266→            <TrendingUp size={22} />
   267→          </div>
   268→          <div className="kpi-card-content">
   269→            <div className="kpi-card-value">
   270→              {estimatedValue.toLocaleString("fr-FR")} GNF
   271→            </div>
   272→            <div className="kpi-card-title">Valeur stock estimée</div>
   273→          </div>
   274→        </div>
   275→      </div>
   276→
   277→      {/* Toolbar */}
   278→      <div className="section-header">
   279→        <div style={{ display: "flex", gap: "12px", alignItems: "center", flex: 1 }}>
   280→          <div style={{ position: "relative", flex: 1, maxWidth: "360px" }}>

   288→                color: "var(--muted)",
   289→              }}
   290→            />
   291→            <input
   292→              type="text"

   297→            />
   298→          </div>
   299→          <select

   302→            style={{ maxWidth: "220px" }}
   303→          >
   304→            {CATEGORY_OPTIONS.map((o) => (
   305→              <option key={o.value} value={o.value}>
   306→                {o.label}
   307→              </option>
   308→            ))}
   309→          </select>
   310→        </div>
   311→        <button className="primary-button" style={{ display: "flex", alignItems: "center", gap: "6px" }}>
   312→          <ShoppingCart size={16} />
   313→          Commande fournisseur
   314→        </button>
   315→      </div>
   316→
   317→      {/* Stock Table */}
   318→      <div className="card" style={{ padding: 0, overflow: "hidden" }}>
   319→        <div className="table-wrapper">
   320→          <table className="table">
   321→            <thead>
   322→              <tr>
   323→                <th>Produit</th>
   324→                <th>Catégorie</th>
   325→                <th>Forme</th>
   326→                <th>Dosage</th>
   327→                <th style={{ textAlign: "right" }}>Qté dispo</th>
   328→                <th style={{ textAlign: "right" }}>Seuil min</th>
   329→                <th>Statut</th>
   330→                <th>Actions</th>
   331→              </tr>
   332→            </thead>
   333→            <tbody>
   334→              {filtered.length === 0 ? (
   335→                <tr>
   336→                  <td colSpan={8} style={{ textAlign: "center", padding: "24px" }}>
   337→                    <span className="muted">Aucun produit en stock.</span>
   338→                  </td>
   339→                </tr>
   340→              ) : (
   341→                filtered.map((item: Row) => {
   342→                  const status = getStockStatus(item);
   343→                  return (
   344→                    <tr
   345→                      key={item.id}
   346→                      style={{
   347→                        background:
   348→                          status === "epuise"
   349→                            ? "var(--danger-light)"
   350→                            : status === "bas"
   351→                            ? "var(--warning-light)"
   352→                            : "inherit",
   353→                      }}
   354→                    >
   355→                      <td style={{ fontWeight: 600 }}>
   356→                        {item.product_name || item.product_id || "—"}
   357→                        {item.product_code && (
   358→                          <span
   359→                            className="muted"
   360→                            style={{ fontWeight: 400, marginLeft: "8px", fontSize: "12px" }}

   372→                      <td style={{ textAlign: "right" }}>
   373→                        {item.min_threshold ?? 0}
   374→                      </td>
   375→                      <td>
   376→                        <span
   377→                          className={`badge ${
   378→                            status === "ok"
   379→                              ? "badge-green"
   380→                              : status === "bas"
   381→                              ? "badge-yellow"
   382→                              : "badge-red"
   383→                          }`}
   384→                        >
   385→                          {status === "ok"
   386→                            ? "OK"
   387→                            : status === "bas"
   388→                            ? "Bas"
   389→                            : "Épuisé"}
   390→                        </span>
   391→                      </td>
   392→                      <td>

   398→                              `Détail produit: ${item.product_name || item.product_id}`

   409→            </tbody>

   412→      </div>

   414→      {/* Chart */}
   415→      <div className="card chart-container" style={{ marginTop: "20px" }}>

   419→            Top 5 produits les plus dispensés

   447→/* ═════════════════════════════════════════════════════════════════
   448→   Dispensation Tab
   449→   ═════════════════════════════════════════════════════════════════ */
   450→
   451→interface DispensationLine {
   452→  id: string;
   453→  product_id: string;
   454→  quantity: number;
   455→}
   456→
   457→function DispensationTab({
   458→  lookups,
   459→  onCreated,
   460→}: {
   461→  lookups: LookupData;
   462→  onCreated: () => void;
   463→}) {
   464→  const options = buildOptions(lookups);
   465→  const facilityId = firstValue(options.facilities);
   466→  const doctors = lookups.staff.filter((s) => s.role === "DOCTOR");
   467→  const doctorOptions = doctors.map((d) => ({
   468→    value: d.id,
   469→    label: `${d.last_name || ""} ${d.first_name || ""} — ${d.role || ""}`.trim(),
   470→  }));
   471→
   472→  const [patientId, setPatientId] = useState("");
   473→  const [prescription, setPrescription] = useState("");
   474→  const [prescriberId, setPrescriberId] = useState("");
   475→  const [lines, setLines] = useState<DispensationLine[]>([]);
   476→  const [submitting, setSubmitting] = useState(false);
   477→
   478→  const [stock, setStock] = useState<Row[]>([]);
   479→  const [recentDispensations, setRecentDispensations] = useState<Row[]>([]);
   480→
   481→  const loadStock = useCallback(async () => {
   482→    try {
   483→      const res = await apiRequest<any>("/pharmacy/stock?page_size=1000");
   484→      setStock(Array.isArray(res.data) ? res.data : []);
   485→    } catch {
   486→      /* silent */
   487→    }
   488→  }, []);
   489→
   490→  const loadMovements = useCallback(async () => {
   491→    try {
   492→      const res = await apiRequest<any>("/pharmacy/stock/movements?page_size=1000");
   493→      const movements = Array.isArray(res.data) ? res.data : [];
   494→      setRecentDispensations(movements.filter((m: Row) => m.movement_type === "OUT"));
   495→    } catch {
   496→      /* silent */
   497→    }
   498→  }, []);
   499→
   500→  useEffect(() => {
   501→    loadStock();
   502→    loadMovements();
   503→    const handler = () => {
   504→      loadStock();
   505→      loadMovements();
   506→    };
   507→    window.addEventListener("refresh-resource", handler);
   508→    return () => window.removeEventListener("refresh-resource", handler);
   509→  }, [loadStock, loadMovements]);
   510→
   511→  function addLine() {
   512→    setLines([
   513→      ...lines,
   514→      { id: `line-${Date.now()}`, product_id: "", quantity: 1 },
   515→    ]);
   516→  }
   517→
   518→  function removeLine(lineId: string) {
   519→    setLines(lines.filter((l) => l.id !== lineId));
   520→  }
   521→
   522→  function updateLine(lineId: string, field: "product_id" | "quantity", value: string | number) {
   523→    setLines(
   524→      lines.map((l) => (l.id === lineId ? { ...l, [field]: value } : l))
   525→    );
   526→  }
   527→
   528→  function getStockForProduct(productId: string): number {
   529→    const item = stock.find((s) => s.product_id === productId);
   530→    return item?.quantity_available ?? 0;
   531→  }
   532→
   533→  async function handleDispense(e: React.FormEvent) {
   534→    e.preventDefault();
   535→    if (!patientId || lines.length === 0) {
   536→      showToast("Veuillez sélectionner un patient et au moins un produit.", "error");
   537→      return;
   538→    }
   539→
   540→    const invalidLines = lines.filter(
   541→      (l) => !l.product_id || l.quantity <= 0
   542→    );
   543→    if (invalidLines.length > 0) {
   544→      showToast("Veuillez remplir toutes les lignes de prescription.", "error");
   545→      return;
   546→    }
   547→
   548→    setSubmitting(true);
   549→    try {
   550→      for (const line of lines) {
   551→        await apiRequest("/pharmacy/stock/movements", {
   552→          method: "POST",
   553→          body: JSON.stringify({
   554→            facility_id: facilityId,
   555→            product_id: line.product_id,
   556→            movement_type: "OUT",
   557→            quantity: line.quantity,
   558→            reason: `Dispensation — Patient: ${patientId} — Prescripteur: ${prescriberId || "N/A"} — Ordonnance: ${prescription || "N/A"}`,
   559→          }),
   560→        });
   561→      }
   562→      showToast("Dispensation enregistrée avec succès.", "success");
   563→      setPatientId("");
   564→      setPrescription("");
   565→      setPrescriberId("");
   566→      setLines([]);
   567→      onCreated();
   568→      loadStock();
   569→      loadMovements();
   570→    } catch (err: any) {
   571→      showToast(err.message || "Erreur lors de la dispensation.", "error");
   572→    } finally {
   573→      setSubmitting(false);
   574→    }
   575→  }
   576→
   577→  return (
   578→    <>
   579→      {/* Dispensation Form */}
   580→      <div className="card form-card">
   581→        <h3 style={{ marginBottom: "16px" }}>Nouvelle dispensation</h3>
   582→        <form onSubmit={handleDispense}>
   583→          <div className="form-grid" style={{ marginBottom: "16px" }}>
   584→            <label className="form-control">
   585→              Patient
   586→              <select
   587→                value={patientId}
   588→                onChange={(e) => setPatientId(e.target.value)}
   589→                required
   590→              >
